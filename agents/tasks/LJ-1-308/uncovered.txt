76	module _ {n : ℕ} where
77	  binForm : ℕ → Formula S (4 + n) → Formula S (suc n)
78	  binForm k rel = ∃̇ (∃̇ (∃̇ (arityTagPairAtL
79	    (suc (suc (suc zero))) (suc (suc zero)) k (suc zero) zero ∧̇ rel)))
81	  unForm : ℕ → Formula S (3 + n) → Formula S (suc n)
82	  unForm k rel = ∃̇ (∃̇ (arityTagAtL (suc (suc zero)) (suc zero) k zero ∧̇ rel))
84	isTmAt : ∀ {n} → Fin n → Fin n → Fin n → Formula S n
85	isTmAt t N A = ∃̇ (tagAtL (suc t) 0 zero ∧̇ (var zero ∈̇ var (suc A)))
86	            ∨̇ ∃̇ (tagAtL (suc t) 1 zero ∧̇ (var zero ∈̇ var (suc N)))
88	module _ {n : ℕ} where
89	  bothTm fstTm : Fin n → Formula S (4 + n)
90	  bothTm A = isTmAt (suc zero) (suc (suc zero)) (suc (suc (suc (suc A))))
91	          ∧̇ isTmAt zero (suc (suc zero)) (suc (suc (suc (suc A))))
92	  fstTm  A = isTmAt (suc zero) (suc (suc zero)) (suc (suc (suc (suc A))))
94	  noneB : Formula S (4 + n)
95	  noneB = ⊤̇ {n = 4 + n}
97	  zeroPay noneU : Formula S (3 + n)
98	  zeroPay = var zero ≐ con (numeralL 0)
99	  noneU   = ⊤̇ {n = 3 + n}
101	  shapes : Fin n → Formula S (suc n)
102	  shapes A = binForm 0 (bothTm A) ∨̇ (binForm 1 (bothTm A)
103	           ∨̇ (binForm 2 noneB ∨̇ (binForm 3 noneB ∨̇ (binForm 4 noneB
104	           ∨̇ (unForm 5 noneU ∨̇ (unForm 6 zeroPay ∨̇ (unForm 7 zeroPay
105	           ∨̇ (unForm 8 noneU ∨̇ (unForm 9 noneU
106	           ∨̇ (binForm 10 (fstTm A) ∨̇ binForm 11 (fstTm A)))))))))))
108	  shapedAt : Fin n → Fin n → Formula S n
109	  shapedAt C A = ∀̇∈ (var C) (shapes A)
