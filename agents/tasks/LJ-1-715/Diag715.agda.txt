-- the union clause stated bare: the failing shape

IsLimitA : S → Type (ℓ-suc ℓ)
IsLimitA α =
    IsOrd α
  × ((x : S) → ⟨ x ∈ˢ α ⟩ → ⟨ sucV x ∈ˢ α ⟩)
  × ((X : Type ℓ) (f : X → S)
      → ((i : X) → ⟨ f i ∈ˢ α ⟩)
      → ⟨ ⋃ (sett X f) ∈ˢ α ⟩)
