-- the successor clause alone

succClOnly : (α : S) → Type (ℓ-suc ℓ)
succClOnly α = (x : S) → ⟨ x ∈ˢ α ⟩ → ⟨ sucV x ∈ˢ α ⟩

-- does S unfold to V (ell)?

-- the union clause alone, concrete family

unionClOnly : (α : S) → Type (ℓ-suc ℓ)
unionClOnly α =
  (X : Type ℓ) (f : X → S)
    → ((i : X) → ⟨ f i ∈ˢ α ⟩)
    → ⟨ ⋃ (sett X f) ∈ˢ α ⟩
