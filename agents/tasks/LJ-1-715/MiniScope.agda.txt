-- [LJ-1.715] MINIATURE, kept as .agda.txt because it cannot typecheck by
-- design: it measures the NotInScope.  Evidence run: runs/mini-scope.out.
-- Original question (one): is a
-- clause's where-bound function, lambda-lifted as <fn>.<name> over the
-- clause context, referenceable by that qualified name from a SIBLING
-- definition in the same module?  The 710 probe measured NotInScope for
-- L.Ordinal.bound2.f from OUTSIDE the module (runs/t-paths2.out); this
-- file measures the INSIDE case at the same shape.

open import Agda.Primitive using ( Level )

module LJ-1-715.MiniScope {ℓ : Level} where

data Two : Set where
  tt ff : Two

-- The bound2 shape: a definition whose where block defines a
-- clause-indexed family over the clause context (A u v, used or not)
-- and hands the result out through one tuple.

bar : (A : Set ℓ) (u v : A) → A
bar A u v = fam tt
  where
  f : Two → A
  f tt = u
  f ff = v

  fam : A
  fam = f tt

-- The sibling, same module, qualified reference, then a conversion
-- check at a CONCRETE index: sibling's result must judge equal to v,
-- which forces the checker to unfold bar.f through its clauses.

sib : (A : Set ℓ) (u v : A) → A
sib A u v = bar.f A u v ff

sib-check : (A : Set ℓ) (u v : A) → sib A u v ≡ v
sib-check A u v = refl

-- And a variable-index use: the family handed to a quantifier at a
-- variable, clause-split on the index first (the 710 cure, premise 5).

sib-split : (A : Set ℓ) (u v : A) (b : Two) → A
sib-split A u v tt = bar.f A u v tt
sib-split A u v ff = bar.f A u v ff
