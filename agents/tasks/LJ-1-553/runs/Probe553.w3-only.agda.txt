--
-- THE WIDEST UNMEASURED TERM, as the brief names it: the membership
-- itself, at `KValue`'s frame, INHABITED.
--
--   ⟨ fst (lookup (suc zero) γ') ∈ fst K ⟩   at KValue's frame
--
-- `[LJ-1.71]` shipped a module whose telescope was uninhabited at EVERY
-- frame (archive/dev/LJ-dispatch-index.md:135), so the supplied
-- hypothesis is guarded here before a field is touched.
--
-- THE GUARD IS WEAKER HERE THAN IT WAS FOR `[LJ-1.551]`, AND I SAY SO.
-- `[LJ-1.71]`'s `tagEq` was uninhabited because its slot was PINNED to a
-- value that could not satisfy it.  The graph slot, `lookup (suc zero)
-- γ'`, is a FREE cell of the frame (agents/tasks/LJ-1-551/Probe551.agda
-- :341, cell `g1`), so it can be instantiated.  A single instantiation
-- would therefore be a cheap pass.  I give TWO, and the second is the
-- one that carries weight:
--
--   (a) `g1 := numeralL 0`, discharged by `KValue.facts .numK0`.  This
--       is the cheap one and it proves only non-vacuity.
--   (b) `g1 := SE.B₀ = LsetS gam ordγ`, THE CARRIER STAGE ITSELF,
--       discharged from `SupplyEnv.B₀∈σ` and `Lset-mono`.  A value graph
--       over the carrier is a set of that scale, so this is the witness
--       at a set the graph slot could actually hold.
--
-- `[LJ-1.508]` measured that NOTHING in `src/` states this membership:
-- `grep -rnE "fst \(lookup T γ'?\) ∈ fst \(lookup .*K" src/` returns 0
-- (agents/tasks/LJ-1-508/lj-1.508-report.md:360-366).  So it is supplied
-- here as a hypothesis and inhabited here as a witness, and it is
-- claimed nowhere else.
--
-- ONE Agda process per run, GHCRTS="-A64m -I0 -M8g", the wide caliber,
-- set on the pane by the program and untouched here.  Nothing lands in
-- src/.  I postulate nothing.

-- =====================================================================
-- W3.  THE SUPPLIED MEMBERSHIP, INHABITED AT `KValue`'s FRAME.
-- =====================================================================

module W3
  (lam : V ℓ) (ordλ : IsOrd lam)
  (succλ : (d : V ℓ) → ⟨ d ∈ lam ⟩ → ⟨ sucV d ∈ lam ⟩)
  (∅∈λ : ⟨ ∅ ∈ lam ⟩)
  (gam : V ℓ) (ordγ : IsOrd gam) (γ∈λ : ⟨ gam ∈ lam ⟩)
  (ω∈σ : ⟨ ω ∈ sucV gam ⟩) where

  module KV = KValue lam ordλ succλ ∅∈λ gam ordγ γ∈λ
  module SE = SupplyEnv lam ordλ succλ ∅∈λ gam ordγ γ∈λ ω∈σ

  -- The frame is `[LJ-1.551]`'s, unchanged: cell 0 PINNED to `B₀` and
  -- five free cells above `KValue.Kenv`
  -- (agents/tasks/LJ-1-551/Probe551.agda:338-339).
  γ‡ : (g1 g2 g3 g4 g5 : S) → Vec S 20
  γ‡ g1 g2 g3 g4 g5 = SE.B₀ ∷ g1 ∷ g2 ∷ g3 ∷ g4 ∷ g5 ∷ KV.Kenv

  Kslot : (g1 g2 g3 g4 g5 : S) → S
  Kslot g1 g2 g3 g4 g5 =
    lookup (suc (suc (suc (suc (suc (suc KV.iK)))))) (γ‡ g1 g2 g3 g4 g5)

  -- THE HYPOTHESIS AS A TYPE, read at the record's own graph index.
  -- `lookup (suc zero) (γ‡ g1 g2 g3 g4 g5)` IS `g1`.
  GraphInK : (g1 g2 g3 g4 g5 : S) → Type (ℓ-suc ℓ)
  GraphInK g1 g2 g3 g4 g5 =
    ⟨ fst (lookup (suc zero) (γ‡ g1 g2 g3 g4 g5))
      ∈ fst (Kslot g1 g2 g3 g4 g5) ⟩

  -- (a) THE CHEAP WITNESS.  Non-vacuity and nothing more.
  graph-in-K-numeral : (g2 g3 g4 g5 : S) → GraphInK (numeralL 0) g2 g3 g4 g5
  graph-in-K-numeral g2 g3 g4 g5 = KV.facts .numK0

  -- (b) THE WITNESS THAT CARRIES WEIGHT: the CARRIER STAGE in `K`.
  --     `SE.B₀∈σ` puts `Lset gam` in `Lset σ`
  --     (src/L/Coding/EnvSupply.lagda.md:127) and `SE.σ∈λ` puts `σ` in
  --     `lam` (:123), so `Lset-mono` lands it in `Lset lam`, which IS
  --     `fst (Kslot ...)`.
  graph-in-K-carrier : (g2 g3 g4 g5 : S) → GraphInK SE.B₀ g2 g3 g4 g5
  graph-in-K-carrier g2 g3 g4 g5 =
    Lset-mono {α = lam} {β = SE.σ} SE.σ∈λ {x = fst SE.B₀} SE.B₀∈σ

  -- THE W3 TERM: the two witnesses in ONE value, at one frame.
  membership-inhabited : (g2 g3 g4 g5 : S)
    → GraphInK (numeralL 0) g2 g3 g4 g5 × GraphInK SE.B₀ g2 g3 g4 g5
  membership-inhabited g2 g3 g4 g5 =
    graph-in-K-numeral g2 g3 g4 g5 , graph-in-K-carrier g2 g3 g4 g5

membership-inhabited = W3.membership-inhabited
