module At (lam : S) (ordλ : IsOrd lam)
  (succλ : (d : S) → ⟨ d ∈ˢ lam ⟩ → ⟨ sucV d ∈ˢ lam ⟩)
  (X : S) (X⊆Lλ : (z : S) → ⟨ z ∈ˢ X ⟩ → ⟨ z ∈ˢ Lset lam ⟩)

  open W3.At lam ordλ succλ public

  -- Verbatim agents/tasks/LJ-1-697/Probe697.agda:72-74.
  Below : (δ : CS.S) (oδ : IsOrd (fst δ)) → Type (ℓ-suc ℓ)
  Below δ oδ =
    ⟨ fst (hierL (fst δ) (δ .snd) oδ) ∈ˢ Lset (W3.step 3 (fst δ)) ⟩

  -- THE ATTEMPT.  The only rule that CONCLUDES a stage membership is
  -- Lset-mono.  Fired with α = step 3 (fst δ), it leaves exactly two
  -- residuals: an index-membership ⟨ β ∈ˢ step 3 (fst δ) ⟩ and a base
  -- membership ⟨ fst (hierL ...) ∈ˢ Lset β ⟩.  Both undersupplied by
  -- the triple; Agda records them as unsolved interaction metas.
  below-direct : (δ : CS.S) (oδ : IsOrd (fst δ)) → Below δ oδ
  below-direct δ oδ =
    Lset-mono {α = W3.step 3 (fst δ)} {β = _} _ _
