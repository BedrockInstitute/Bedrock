--     conclusions at `Lset lam`, not at a slot.
-- The record states BOTH at one `γ'` and one `K`, six successors above
-- the tag indices (src/L/Condensation/TwelveAgree.lagda.md:133, :262).

-- 37-field value is refuted before it is written.

-- =====================================================================
-- W3.  THE CLEAN SET'S FRAME, INHABITED, AT ONE `γ'` AND ONE `K`.

module W3
  (lam : V ℓ) (ordλ : IsOrd lam)
  (succλ : (d : V ℓ) → ⟨ d ∈ lam ⟩ → ⟨ sucV d ∈ lam ⟩)
  (∅∈λ : ⟨ ∅ ∈ lam ⟩)
  (gam : V ℓ) (ordγ : IsOrd gam) (γ∈λ : ⟨ gam ∈ lam ⟩)
  (ω∈σ : ⟨ ω ∈ sucV gam ⟩) where

  module KV = KValue lam ordλ succλ ∅∈λ gam ordγ γ∈λ
  module SE = SupplyEnv lam ordλ succλ ∅∈λ gam ordγ γ∈λ ω∈σ

  -- The env family PINS cell 0, exactly as `[LJ-1.538]` pins it and
  -- `[LJ-1.545]` re-pins it (agents/tasks/LJ-1-545/Probe545.agda:416).
  γ‡ : (g1 g2 g3 g4 g5 : S) → Vec S 20
  γ‡ g1 g2 g3 g4 g5 = SE.B₀ ∷ g1 ∷ g2 ∷ g3 ∷ g4 ∷ g5 ∷ KV.Kenv

  Kslot : (g1 g2 g3 g4 g5 : S) → S
  Kslot g1 g2 g3 g4 g5 =
    lookup (suc (suc (suc (suc (suc (suc KV.iK)))))) (γ‡ g1 g2 g3 g4 g5)

  -- (a) the `KFacts` side, at the record's shifted index.
  TagAtFrame : (g1 g2 g3 g4 g5 : S) → Type (ℓ-suc ℓ)
  TagAtFrame g1 g2 g3 g4 g5 =
    fst (lookup (suc (suc (suc (suc (suc (suc KV.i0)))))) (γ‡ g1 g2 g3 g4 g5))
      ≡ fst (numeralL 0)

  -- (b) the record's `transK`, which is `arityK` with the binders swapped.
  TransAtFrame : (g1 g2 g3 g4 g5 : S) → Type (ℓ-suc ℓ)
  TransAtFrame g1 g2 g3 g4 g5 =
      (x a : S) → ⟨ fst x ∈ fst a ⟩
    → ⟨ fst a ∈ fst (Kslot g1 g2 g3 g4 g5) ⟩
    → ⟨ fst x ∈ fst (Kslot g1 g2 g3 g4 g5) ⟩

  -- (c) the `EnvSupply` side, at the record's own `K` slot.  This is the
  --     conjunct that could have failed: `SE.envK-gen` concludes at
  --     `Lset lam` (src/L/Coding/EnvSupply.lagda.md:281) and the record
  --     concludes at `lookup (suc^6 K) γ'`.
  EnvAtFrame : (g1 g2 g3 g4 g5 : S) → Type (ℓ-suc ℓ)
  EnvAtFrame g1 g2 g3 g4 g5 =
      (yc b a ar c E : S)
    → ∥ Σ[ n ∈ ℕ ] (fst ar ≡ # n) ∥₁
    → ⟨ (E ∷ yc ∷ b ∷ a ∷ ar ∷ c ∷ γ‡ g1 g2 g3 g4 g5) ⊨
        envSetAt zero (suc (suc (suc (suc zero))))
                      (suc (suc (suc (suc (suc (suc zero)))))) ⟩
    → ⟨ fst E ∈ fst (Kslot g1 g2 g3 g4 g5) ⟩

  -- THE W3 TERM.  The three, at ONE frame, in ONE value.
  clean-frame-inhabited : (g1 g2 g3 g4 g5 : S)
    → TagAtFrame g1 g2 g3 g4 g5
      × TransAtFrame g1 g2 g3 g4 g5
      × EnvAtFrame g1 g2 g3 g4 g5
  clean-frame-inhabited g1 g2 g3 g4 g5 =
      KV.facts .tagEq0
    , (λ x a hx ha → KV.facts .arityK a x hx ha)
    , (λ yc b a ar c E arNum h →
        SE.envK-gen (E ∷ yc ∷ b ∷ a ∷ ar ∷ c ∷ γ‡ g1 g2 g3 g4 g5)
          zero (suc (suc (suc (suc zero))))
          (suc (suc (suc (suc (suc (suc zero))))))
          refl arNum h)

clean-frame-inhabited = W3.clean-frame-inhabited
