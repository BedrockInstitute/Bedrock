  PT.rec Empty.isProp⊥ kill (subst ⟨_⟩ (spec z) x∈H)
  where
  H : CS.S
  H = hierL ∅ h0 o0
  xL : ⟨ isL x ⟩
  xL = isL-trans x∈H (H .snd)
  z : CS.S
  z = x , xL
  spec : IsHier ∅ H

  kill : Σ[ c ∈ CS.S ]
           (⟨ fst c ∈ ∅ ⟩ × (x ≡ pr (fst c) (Lset (fst c)))) → Empty.⊥
