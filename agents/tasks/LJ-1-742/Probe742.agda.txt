-- SECTION 4.  THE CLOSED OPERATIONS AT GAMMA.  sgl, binary cup and pr

module OpsAt (γ : V ℓ) (oγ : IsOrd γ) (clγ : closedω γ) (ω∈γ : ⟨ ω ∈ˢᵥ γ ⟩) where
  open Γ γ oγ clγ ω∈γ
  private
    module SE = SupplyEnv γ oγ (suc∈γ γ oγ clγ) (∅∈γ γ oγ ω∈γ)
                    ω ω-ord ω∈γ (self∈sucV ω)

  sgl∈γ : (x : V ℓ) → ⟨ x ∈ˢᵥ Lset γ ⟩ → ⟨ ⁅ x ⁆s ∈ˢᵥ Lset γ ⟩
  sgl∈γ x hx = PT.rec (snd (⁅ x ⁆s ∈ˢᵥ Lset γ)) step (Lset-out′ lem γ x hx)
    where
    step : Σ[ δ ∈ V ℓ ] (⟨ δ ∈ˢᵥ γ ⟩ × ⟨ x ∈ˢᵥ Lset (sucV δ) ⟩)
         → ⟨ ⁅ x ⁆s ∈ˢᵥ Lset γ ⟩
    step (δ , δ∈γ , hxδ) =
      Lset-mono {α = γ} {β = sucV (sucV δ)}
        (suc∈γ γ oγ clγ (sucV δ) (suc∈γ γ oγ clγ δ δ∈γ))
        (sgl∈Lset-suc (sucV δ) x hxδ)

  cup∈γ : (x y : V ℓ) → ⟨ x ∈ˢᵥ Lset γ ⟩ → ⟨ y ∈ˢᵥ Lset γ ⟩
        → ⟨ (x ∪ y) ∈ˢᵥ Lset γ ⟩
  cup∈γ x y hx hy =
    PT.rec (snd ((x ∪ y) ∈ˢᵥ Lset γ)) (λ xst →
      PT.rec (snd ((x ∪ y) ∈ˢᵥ Lset γ)) (step xst) (Lset-out′ lem γ y hy))
      (Lset-out′ lem γ x hx)
    where
    step : Σ[ δ ∈ V ℓ ] (⟨ δ ∈ˢᵥ γ ⟩ × ⟨ x ∈ˢᵥ Lset (sucV δ) ⟩)
         → Σ[ ε ∈ V ℓ ] (⟨ ε ∈ˢᵥ γ ⟩ × ⟨ y ∈ˢᵥ Lset (sucV ε) ⟩)
         → ⟨ (x ∪ y) ∈ˢᵥ Lset γ ⟩
    step (δ , δ∈γ , hxδ) (ε , ε∈γ , hyε) =
      close (ord-tri (sucV δ) oδ (sucV ε) oε)
      where
      oδ : IsOrd (sucV δ)
      oδ = suc-ord (mem-ord {A = γ} oγ δ δ∈γ)
      oε : IsOrd (sucV ε)
      oε = suc-ord (mem-ord {A = γ} oγ ε ε∈γ)

      at : (τ : V ℓ) → IsOrd τ → ⟨ τ ∈ˢᵥ γ ⟩
         → ⟨ x ∈ˢᵥ Lset τ ⟩ → ⟨ y ∈ˢᵥ Lset τ ⟩
         → ⟨ (x ∪ y) ∈ˢᵥ Lset γ ⟩
      at τ oτ τ∈γ hxe hye =
        Lset-mono {α = γ} {β = sucV (sucV τ)}
          (suc∈γ γ oγ clγ (sucV τ) (suc∈γ γ oγ clγ τ τ∈γ))
          (SE.union∈Lset-suc (sucV τ) ⁅ x , y ⁆ (pair∈Lset-suc τ x y hxe hye))

      close : Tri (sucV δ) (sucV ε) → ⟨ (x ∪ y) ∈ˢᵥ Lset γ ⟩
      close (inl p) = at (sucV ε) oε
        (suc∈γ γ oγ clγ ε ε∈γ)
        (Lset-mono {α = sucV ε} {β = sucV δ} p hxδ) hyε
      close (inr (inl e)) = at (sucV δ) oδ
        (suc∈γ γ oγ clγ δ δ∈γ) hxδ
        (subst (λ w → ⟨ y ∈ˢᵥ Lset w ⟩) (sym e) hyε)
      close (inr (inr p)) = at (sucV δ) oδ
        (suc∈γ γ oγ clγ δ δ∈γ) hxδ
        (Lset-mono {α = sucV δ} {β = sucV ε} p hyε)

-- ---------------------------------------------------------------------
-- SECTION 5.  THE Delta0 TRANSPORT.  For a Delta0 formula the bounded
-- semantics differs from the standard one only inside the recursion,
-- and every bounded-quantifier site keeps its own term-membership
-- filter, so a standard proof transports by recursion on the Delta0
-- witness.  The unbounded quantifiers are NOT covered here;  each of
-- them is discharged by hand in Section 7 with a witness in the bound.
-- ---------------------------------------------------------------------

module D0 (c : S) where
  module CR = Correct (hPropAlgebra (ℓ-suc ℓ)) 𝒮ʟ (λ x → x) c
  open CR using ( _⊨ᴬ_ )

  -- For a Delta0 formula the two readings are POINTWISE EQUAL: the
  -- bounded quantifiers keep their term filters, and only the
  -- unbounded quantifiers differ.  One induction gives the path;  both
  -- transport directions read off it.
  d0eq : ∀ {n} (θ : Formula S n) → Δ₀ θ
       → (δ : S ^ n) → (δ ⊨ θ) ≡ (δ ⊨ᴬ θ)
  d0eq (t ∈̇ u) δ-∈ δ = refl
  d0eq (t ≐ u) δ-≐ δ = refl
  d0eq (φ ∧̇ ψ) (δ-∧ cφ dψ) δ = cong₂ _⊓_ (d0eq φ cφ δ) (d0eq ψ dψ δ)
  d0eq (φ ∨̇ ψ) (δ-∨ cφ dψ) δ = cong₂ _⊔_ (d0eq φ cφ δ) (d0eq ψ dψ δ)
  d0eq (φ ⇒̇ ψ) (δ-⇒ cφ dψ) δ = cong₂ _⇒_ (d0eq φ cφ δ) (d0eq ψ dψ δ)
  d0eq (¬̇ φ) (δ-¬ cφ) δ = cong ¬_ (d0eq φ cφ δ)
  d0eq ⊤̇ δ-⊤ δ = refl
  d0eq ⊥̇ δ-⊥ δ = refl
  d0eq (∀̇∈ t φ) (δ-∀∈ cφ) δ = cong (⋀ S) (funExt (λ x →
    cong (λ q → (x ∈ˢ AbsL742.⟦ t ⟧ᵐ δ) ⇒ q) (d0eq φ cφ (x ∷ δ))))
  d0eq (∃̇∈ t φ) (δ-∃∈ cφ) δ = cong (⋁ S) (funExt (λ x →
    cong (λ q → (x ∈ˢ AbsL742.⟦ t ⟧ᵐ δ) ⊓ q) (d0eq φ cφ (x ∷ δ))))

  d0 : ∀ {n} (θ : Formula S n) → Δ₀ θ
     → (δ : S ^ n) → ⟨ δ ⊨ θ ⟩ → ⟨ δ ⊨ᴬ θ ⟩
  d0 θ dθ δ h = subst (λ R → ⟨ R ⟩) (d0eq θ dθ δ) h

  d0r : ∀ {n} (θ : Formula S n) → Δ₀ θ
      → (δ : S ^ n) → ⟨ δ ⊨ᴬ θ ⟩ → ⟨ δ ⊨ θ ⟩
  d0r θ dθ δ h = subst (λ R → ⟨ R ⟩) (sym (d0eq θ dθ δ)) h

  code∈γ ⊤̇ = pr∈γ (# 6) (# 0) (num∈γ 6) (num∈γ 0)
  code∈γ ⊥̇ = pr∈γ (# 7) (# 0) (num∈γ 7) (num∈γ 0)
  code∈γ (∃̇ a) = pr∈γ (# 8) (VCode.⌜ mapFo ι a ⌝) (num∈γ 8) (code∈γ a)
  code∈γ (∀̇ a) = pr∈γ (# 9) (VCode.⌜ mapFo ι a ⌝) (num∈γ 9) (code∈γ a)
  code∈γ (∀̇∈ t a) = pr∈γ (# 10) (pr (VCode.⌜ mapTm ι t ⌝ᵗ) (VCode.⌜ mapFo ι a ⌝))
      (num∈γ 10) (pr∈γ _ _ (codeTm∈γ t) (code∈γ a))
  code∈γ (∃̇∈ t a) = pr∈γ (# 11) (pr (VCode.⌜ mapTm ι t ⌝ᵗ) (VCode.⌜ mapFo ι a ⌝))
      (num∈γ 11) (pr∈γ _ _ (codeTm∈γ t) (code∈γ a))

  -- the subformula closure (InL's `closure`), staged at gamma
  clo∈γ : ∀ {n} (φ : Formula ⟪ fst A ⟫ n)
        → ⟨ closure ι ιL φ ∈ˢᵥ Lset γ ⟩
  clo∈γ (t ∈̇ u) = sgl∈γ (key ι ιL (t ∈̇ u)) (pr∈γ _ _ (num∈γ _) (code∈γ (t ∈̇ u)))
  clo∈γ (t ≐ u) = sgl∈γ (key ι ιL (t ≐ u)) (pr∈γ _ _ (num∈γ _) (code∈γ (t ≐ u)))
  clo∈γ {n} ⊤̇ =
    let φt : Formula ⟪ fst A ⟫ n
        φt = ⊤̇
    in sgl∈γ (key ι ιL φt)
         (pr∈γ (# n) (VCode.⌜ mapFo ι φt ⌝) (num∈γ n) (code∈γ φt))
  clo∈γ {n} ⊥̇ =
    let φf : Formula ⟪ fst A ⟫ n
        φf = ⊥̇
    in sgl∈γ (key ι ιL φf)
         (pr∈γ (# n) (VCode.⌜ mapFo ι φf ⌝) (num∈γ n) (code∈γ φf))
  clo∈γ (a ∧̇ b) = cup∈γ _ _ (sgl∈γ (key ι ιL (a ∧̇ b)) (pr∈γ _ _ (num∈γ _) (code∈γ (a ∧̇ b))))
                    (cup∈γ _ _ (clo∈γ a) (clo∈γ b))
  clo∈γ (a ∨̇ b) = cup∈γ _ _ (sgl∈γ (key ι ιL (a ∨̇ b)) (pr∈γ _ _ (num∈γ _) (code∈γ (a ∨̇ b))))
                    (cup∈γ _ _ (clo∈γ a) (clo∈γ b))
  clo∈γ (a ⇒̇ b) = cup∈γ _ _ (sgl∈γ (key ι ιL (a ⇒̇ b)) (pr∈γ _ _ (num∈γ _) (code∈γ (a ⇒̇ b))))
                    (cup∈γ _ _ (clo∈γ a) (clo∈γ b))
  clo∈γ (¬̇ a)   = cup∈γ _ _ (sgl∈γ (key ι ιL (¬̇ a)) (pr∈γ _ _ (num∈γ _) (code∈γ (¬̇ a)))) (clo∈γ a)
  clo∈γ (∃̇ a)   = cup∈γ _ _ (sgl∈γ (key ι ιL (∃̇ a)) (pr∈γ _ _ (num∈γ _) (code∈γ (∃̇ a)))) (clo∈γ a)
  clo∈γ (∀̇ a)   = cup∈γ _ _ (sgl∈γ (key ι ιL (∀̇ a)) (pr∈γ _ _ (num∈γ _) (code∈γ (∀̇ a)))) (clo∈γ a)
  clo∈γ (∀̇∈ t a) = cup∈γ _ _ (sgl∈γ (key ι ιL (∀̇∈ t a)) (pr∈γ _ _ (num∈γ _) (code∈γ (∀̇∈ t a)))) (clo∈γ a)
  clo∈γ (∃̇∈ t a) = cup∈γ _ _ (sgl∈γ (key ι ιL (∃̇∈ t a)) (pr∈γ _ _ (num∈γ _) (code∈γ (∃̇∈ t a)))) (clo∈γ a)

-- ---------------------------------------------------------------------
-- SECTION 8.  THE L-SIDE STAGES.  The L-side key of a relabelled
-- formula carries the V-side key's stage across keyBridge;  the slot
-- and the table are staged by one structural fold mirroring `tree`'s
-- own clauses, with sgl and cup closed at gamma.
-- ---------------------------------------------------------------------

module LStages
  (γ : V ℓ) (oγ : IsOrd γ) (clγ : closedω γ) (ω∈γ : ⟨ ω ∈ˢᵥ γ ⟩)
  (A : S) (hA : ⟨ fst A ∈ Lset γ ⟩)
  (Hkey : keyS-in-carrier-lim)
  (Hsat : Sat-at-asConst)
  where
  open OpsAt γ oγ clγ ω∈γ
  open VStages γ oγ clγ ω∈γ A hA

  toS' : ∀ {n} → Formula ⟪ fst A ⟫ n → Formula S n
  toS' = mapFo (asConst A)

  key∈γ : ∀ {n} (χ : Formula ⟪ fst A ⟫ n) → ⟨ fst (keyS A χ) ∈ˢᵥ Lset γ ⟩
  key∈γ χ = Hkey γ oγ clγ ω∈γ A hA χ

  -- keyBridge: fst (keyS A χ) is the V-level value of fst (keyʟ (toS' χ))
  keyʟ∈γ : ∀ {n} (χ : Formula ⟪ fst A ⟫ n)
         → ⟨ fst (keyʟ (toS' χ)) ∈ˢᵥ Lset γ ⟩
  keyʟ∈γ χ = subst (λ w → ⟨ w ∈ˢᵥ Lset γ ⟩) (keyBridge A χ) (key∈γ χ)

  -- the Sat leaf: the hypothesis covers every arity of the carrier
  Sat∈γ : ∀ {n} (χ : Formula ⟪ fst A ⟫ n)
        → ⟨ fst (Sat A (toS' χ)) ∈ˢᵥ Lset γ ⟩
  Sat∈γ χ = Hsat γ oγ clγ ω∈γ A hA χ

  ent∈γ : ∀ {n} (χ : Formula ⟪ fst A ⟫ n)
        → ⟨ fst (ent A (toS' χ)) ∈ˢᵥ Lset γ ⟩
  ent∈γ χ =
    subst (λ w → ⟨ w ∈ˢᵥ Lset γ ⟩)
      (sym (prʟ-fst (keyʟ (toS' χ)) (Sat A (toS' χ))))
      (pr∈γ (fst (keyʟ (toS' χ))) (fst (Sat A (toS' χ)))
        (keyʟ∈γ χ) (Sat∈γ χ))

  -- sglʟ x and cupʟ x y staged at gamma, through their fst equations
  sglʟ∈γ : (x : S) → ⟨ fst x ∈ˢᵥ Lset γ ⟩ → ⟨ fst (sglʟ x) ∈ˢᵥ Lset γ ⟩
  sglʟ∈γ x h = subst (λ w → ⟨ w ∈ˢᵥ Lset γ ⟩) (sym (sglʟ-fst x)) (sgl∈γ (fst x) h)

  cupʟ∈γ : (x y : S) → ⟨ fst x ∈ˢᵥ Lset γ ⟩ → ⟨ fst y ∈ˢᵥ Lset γ ⟩
         → ⟨ fst (cupʟ x y) ∈ˢᵥ Lset γ ⟩
  cupʟ∈γ x y hx hy =
      subst (λ w → ⟨ w ∈ˢᵥ Lset γ ⟩) (sym (cupʟ-fst x y)) (cup∈γ (fst x) (fst y) hx hy)

  -- one fold for both `tree keyʟ` (the slot) and `tree ent` (the table)
  fold : (f : ∀ {m} → Formula S m → S)
       → (∀ {m} (χ : Formula ⟪ fst A ⟫ m) → ⟨ fst (f (toS' χ)) ∈ˢᵥ Lset γ ⟩)
       → ∀ {n} (χ : Formula ⟪ fst A ⟫ n)
       → ⟨ fst (tree A f (toS' χ)) ∈ˢᵥ Lset γ ⟩
  fold f leaf∈γ (t ∈̇ u) = sglʟ∈γ (f (toS' (t ∈̇ u))) (leaf∈γ (t ∈̇ u))
  fold f leaf∈γ (t ≐ u) = sglʟ∈γ (f (toS' (t ≐ u))) (leaf∈γ (t ≐ u))
  fold f leaf∈γ ⊤̇      = sglʟ∈γ (f (toS' ⊤̇)) (leaf∈γ ⊤̇)
  fold f leaf∈γ ⊥̇      = sglʟ∈γ (f (toS' ⊥̇)) (leaf∈γ ⊥̇)
  fold f leaf∈γ (a ∧̇ b) =
      cupʟ∈γ (sglʟ (f (toS' (a ∧̇ b)))) (cupʟ (tree A f (toS' a)) (tree A f (toS' b)))
      (sglʟ∈γ (f (toS' (a ∧̇ b))) (leaf∈γ (a ∧̇ b)))
      (cupʟ∈γ _ _ (fold f leaf∈γ a) (fold f leaf∈γ b))
  fold f leaf∈γ (a ∨̇ b) =
      cupʟ∈γ (sglʟ (f (toS' (a ∨̇ b)))) (cupʟ (tree A f (toS' a)) (tree A f (toS' b)))
      (sglʟ∈γ (f (toS' (a ∨̇ b))) (leaf∈γ (a ∨̇ b)))
      (cupʟ∈γ _ _ (fold f leaf∈γ a) (fold f leaf∈γ b))
  fold f leaf∈γ (a ⇒̇ b) =
      cupʟ∈γ (sglʟ (f (toS' (a ⇒̇ b)))) (cupʟ (tree A f (toS' a)) (tree A f (toS' b)))
      (sglʟ∈γ (f (toS' (a ⇒̇ b))) (leaf∈γ (a ⇒̇ b)))
      (cupʟ∈γ _ _ (fold f leaf∈γ a) (fold f leaf∈γ b))
  fold f leaf∈γ (¬̇ a) = cupʟ∈γ (sglʟ (f (toS' (¬̇ a)))) (tree A f (toS' a))
      (sglʟ∈γ (f (toS' (¬̇ a))) (leaf∈γ (¬̇ a))) (fold f leaf∈γ a)
  fold f leaf∈γ (∃̇ a) = cupʟ∈γ (sglʟ (f (toS' (∃̇ a)))) (tree A f (toS' a))
      (sglʟ∈γ (f (toS' (∃̇ a))) (leaf∈γ (∃̇ a))) (fold f leaf∈γ a)
  fold f leaf∈γ (∀̇ a) = cupʟ∈γ (sglʟ (f (toS' (∀̇ a)))) (tree A f (toS' a))
      (sglʟ∈γ (f (toS' (∀̇ a))) (leaf∈γ (∀̇ a))) (fold f leaf∈γ a)
  fold f leaf∈γ (∀̇∈ t a) = cupʟ∈γ (sglʟ (f (toS' (∀̇∈ t a)))) (tree A f (toS' a))
      (sglʟ∈γ (f (toS' (∀̇∈ t a))) (leaf∈γ (∀̇∈ t a))) (fold f leaf∈γ a)
  fold f leaf∈γ (∃̇∈ t a) = cupʟ∈γ (sglʟ (f (toS' (∃̇∈ t a)))) (tree A f (toS' a))
      (sglʟ∈γ (f (toS' (∃̇∈ t a))) (leaf∈γ (∃̇∈ t a))) (fold f leaf∈γ a)

  slot∈γ : ∀ {n} (χ : Formula ⟪ fst A ⟫ n) → ⟨ fst (slot A (toS' χ)) ∈ˢᵥ Lset γ ⟩
  slot∈γ = fold keyʟ keyʟ∈γ

  satTable∈γ : ∀ {n} (χ : Formula ⟪ fst A ⟫ n)
             → ⟨ fst (satTable A (toS' χ)) ∈ˢᵥ Lset γ ⟩
  satTable∈γ = fold (ent A) ent∈γ

-- ---------------------------------------------------------------------
-- SECTION 9.  THE DESCENT PACKAGE.  The bounded walks' staging moves,
-- landed as INSTANTIATIONS of EnvSupply's Fact module at the limit
-- stage (EnvSupply.lagda.md:450, prK at :453, valK at :463; the
-- transitivity is layer-trans (Lset-layer gamma), Constructible:183,
-- :246).  pr-down stages the components of a staged pair;  key-down
-- stages the three components of a member key through its arityTag
-- shape;  val-down stages a table entry's value.  The next dispatch's
-- closedAt / shapedAt / twelveAt walks call these;  nothing here is a
-- new construction.

module Desc (γ : V ℓ) (oγ : IsOrd γ) where
  open Fact (LsetS γ oγ) (layer-trans (Lset-layer γ)) public

  pr↓ : (u v : V ℓ) → ⟨ pr u v ∈ˢᵥ Lset γ ⟩
      → ⟨ u ∈ˢᵥ Lset γ ⟩ × ⟨ v ∈ˢᵥ Lset γ ⟩
  pr↓ = prK

  key↓ : (ar a b : S) (k : ℕ)
       → ⟨ pr (fst ar) (pr (# k) (pr (fst a) (fst b))) ∈ˢᵥ Lset γ ⟩
       → ⟨ fst ar ∈ˢᵥ Lset γ ⟩ × ⟨ fst a ∈ˢᵥ Lset γ ⟩ × ⟨ fst b ∈ˢᵥ Lset γ ⟩
  key↓ ar a b k hx = step1 .fst , step3 .fst , step3 .snd
    where
    step1 : ⟨ fst ar ∈ˢᵥ Lset γ ⟩ × ⟨ pr (# k) (pr (fst a) (fst b)) ∈ˢᵥ Lset γ ⟩
    step1 = pr↓ (fst ar) (pr (# k) (pr (fst a) (fst b))) hx
    step2 : ⟨ (# k) ∈ˢᵥ Lset γ ⟩ × ⟨ pr (fst a) (fst b) ∈ˢᵥ Lset γ ⟩
    step2 = pr↓ (# k) (pr (fst a) (fst b)) (step1 .snd)
    step3 : ⟨ fst a ∈ˢᵥ Lset γ ⟩ × ⟨ fst b ∈ˢᵥ Lset γ ⟩
    step3 = pr↓ (fst a) (fst b) (step2 .snd)

  val↓ : (C T : S) → ⟨ fst T ∈ˢᵥ Lset γ ⟩
       → (k : ℕ) (c ar a b yc : S)
       → ⟨ fst c ∈ˢᵥ fst C ⟩
       → fst c ≡ pr (fst ar) (pr (# k) (pr (fst a) (fst b)))
       → ⟨ pr (fst c) (fst yc) ∈ˢᵥ fst T ⟩
       → ⟨ fst yc ∈ˢᵥ Lset γ ⟩
  val↓ C T hT k c ar a b yc c∈ shape hc =
    valK C T hT k c ar a b yc c∈ shape hc
