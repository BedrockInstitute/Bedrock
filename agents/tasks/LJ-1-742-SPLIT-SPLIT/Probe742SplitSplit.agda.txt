-- ---------------------------------------------------------------------
-- SECTION 5.  THE Delta0 TRANSPORT.  For a Delta0 formula the bounded
-- semantics differs from the standard one only inside the recursion,
-- and every bounded-quantifier site keeps its own term-membership
-- filter, so a standard proof transports by recursion on the Delta0
-- witness.  The unbounded quantifiers are NOT covered here;  each of
-- them is discharged by hand in Section 7 with a witness in the bound.
-- ---------------------------------------------------------------------

module D0 (c : S) where
  module CR = Correct (hPropAlgebra (ℓ-suc ℓ)) 𝒮ʟ (λ x → x) c
  open CR using ( _⊨ᴬ_ )

  -- For a Delta0 formula the two readings are POINTWISE EQUAL: the
  -- bounded quantifiers keep their term filters, and only the
  -- unbounded quantifiers differ.  One induction gives the path;  both
  -- transport directions read off it.
  d0eq : ∀ {n} (θ : Formula S n) → Δ₀ θ
       → (δ : S ^ n) → (δ ⊨ θ) ≡ (δ ⊨ᴬ θ)
  d0eq (t ∈̇ u) δ-∈ δ = refl
  d0eq (t ≐ u) δ-≐ δ = refl
  d0eq (φ ∧̇ ψ) (δ-∧ cφ dψ) δ = cong₂ _⊓_ (d0eq φ cφ δ) (d0eq ψ dψ δ)
  d0eq (φ ∨̇ ψ) (δ-∨ cφ dψ) δ = cong₂ _⊔_ (d0eq φ cφ δ) (d0eq ψ dψ δ)
  d0eq (φ ⇒̇ ψ) (δ-⇒ cφ dψ) δ = cong₂ _⇒_ (d0eq φ cφ δ) (d0eq ψ dψ δ)
  d0eq (¬̇ φ) (δ-¬ cφ) δ = cong ¬_ (d0eq φ cφ δ)
  d0eq ⊤̇ δ-⊤ δ = refl
  d0eq ⊥̇ δ-⊥ δ = refl
  d0eq (∀̇∈ t φ) (δ-∀∈ cφ) δ = cong (⋀ S) (funExt (λ x →
    cong (λ q → (x ∈ˢ AbsL742.⟦ t ⟧ᵐ δ) ⇒ q) (d0eq φ cφ (x ∷ δ))))
  d0eq (∃̇∈ t φ) (δ-∃∈ cφ) δ = cong (⋁ S) (funExt (λ x →
    cong (λ q → (x ∈ˢ AbsL742.⟦ t ⟧ᵐ δ) ⊓ q) (d0eq φ cφ (x ∷ δ))))

  d0 : ∀ {n} (θ : Formula S n) → Δ₀ θ
     → (δ : S ^ n) → ⟨ δ ⊨ θ ⟩ → ⟨ δ ⊨ᴬ θ ⟩
  d0 θ dθ δ h = subst (λ R → ⟨ R ⟩) (d0eq θ dθ δ) h

  d0r : ∀ {n} (θ : Formula S n) → Δ₀ θ
      → (δ : S ^ n) → ⟨ δ ⊨ᴬ θ ⟩ → ⟨ δ ⊨ θ ⟩
  d0r θ dθ δ h = subst (λ R → ⟨ R ⟩) (sym (d0eq θ dθ δ)) h

  ι∈γ : (m : ⟪ fst A ⟫) → ⟨ ι m ∈ˢᵥ Lset γ ⟩
  ι∈γ m = Lset-trans′ lem γ (ι∈ m) hA

  num∈γ : (k : ℕ) → ⟨ (# k) ∈ˢᵥ Lset γ ⟩
  num∈γ k = Lset-mono {α = γ} {β = ω} ω∈γ (numeral∈limit k)

  pr∈γ : (x y : V ℓ) → ⟨ x ∈ˢᵥ Lset γ ⟩ → ⟨ y ∈ˢᵥ Lset γ ⟩
       → ⟨ pr x y ∈ˢᵥ Lset γ ⟩
  pr∈γ = pr∈λ

  codeTm∈γ : ∀ {n} (t : Term ⟪ fst A ⟫ n)
           → ⟨ VCode.⌜ mapTm ι t ⌝ᵗ ∈ˢᵥ Lset γ ⟩
  codeTm∈γ (con m) = pr∈γ (# zero) (ι m) (num∈γ zero) (ι∈γ m)
  codeTm∈γ (var i) = pr∈γ (# 1) (# (toℕ i)) (num∈γ 1) (num∈γ (toℕ i))

  code∈γ : ∀ {n} (φ : Formula ⟪ fst A ⟫ n)
         → ⟨ VCode.⌜ mapFo ι φ ⌝ ∈ˢᵥ Lset γ ⟩
  code∈γ (t ∈̇ u) = pr∈γ (# zero) (pr (VCode.⌜ mapTm ι t ⌝ᵗ) (VCode.⌜ mapTm ι u ⌝ᵗ))
      (num∈γ zero) (pr∈γ _ _ (codeTm∈γ t) (codeTm∈γ u))
  code∈γ (t ≐ u) = pr∈γ (# 1) (pr (VCode.⌜ mapTm ι t ⌝ᵗ) (VCode.⌜ mapTm ι u ⌝ᵗ))
      (num∈γ 1) (pr∈γ _ _ (codeTm∈γ t) (codeTm∈γ u))
  code∈γ (a ∧̇ b) = pr∈γ (# 2) (pr (VCode.⌜ mapFo ι a ⌝) (VCode.⌜ mapFo ι b ⌝))
      (num∈γ 2) (pr∈γ _ _ (code∈γ a) (code∈γ b))
  code∈γ (a ∨̇ b) = pr∈γ (# 3) (pr (VCode.⌜ mapFo ι a ⌝) (VCode.⌜ mapFo ι b ⌝))
      (num∈γ 3) (pr∈γ _ _ (code∈γ a) (code∈γ b))
  code∈γ (a ⇒̇ b) = pr∈γ (# 4) (pr (VCode.⌜ mapFo ι a ⌝) (VCode.⌜ mapFo ι b ⌝))
      (num∈γ 4) (pr∈γ _ _ (code∈γ a) (code∈γ b))
  code∈γ (¬̇ a) = pr∈γ (# 5) (VCode.⌜ mapFo ι a ⌝) (num∈γ 5) (code∈γ a)
  code∈γ ⊤̇ = pr∈γ (# 6) (# 0) (num∈γ 6) (num∈γ 0)
  code∈γ ⊥̇ = pr∈γ (# 7) (# 0) (num∈γ 7) (num∈γ 0)
  code∈γ (∃̇ a) = pr∈γ (# 8) (VCode.⌜ mapFo ι a ⌝) (num∈γ 8) (code∈γ a)
  code∈γ (∀̇ a) = pr∈γ (# 9) (VCode.⌜ mapFo ι a ⌝) (num∈γ 9) (code∈γ a)
  code∈γ (∀̇∈ t a) = pr∈γ (# 10) (pr (VCode.⌜ mapTm ι t ⌝ᵗ) (VCode.⌜ mapFo ι a ⌝))
      (num∈γ 10) (pr∈γ _ _ (codeTm∈γ t) (code∈γ a))
  code∈γ (∃̇∈ t a) = pr∈γ (# 11) (pr (VCode.⌜ mapTm ι t ⌝ᵗ) (VCode.⌜ mapFo ι a ⌝))
      (num∈γ 11) (pr∈γ _ _ (codeTm∈γ t) (code∈γ a))

  -- the subformula closure (InL's `closure`), staged at gamma
  clo∈γ : ∀ {n} (φ : Formula ⟪ fst A ⟫ n)
        → ⟨ closure ι ιL φ ∈ˢᵥ Lset γ ⟩
  clo∈γ (t ∈̇ u) = sgl∈γ (key ι ιL (t ∈̇ u)) (pr∈γ _ _ (num∈γ _) (code∈γ (t ∈̇ u)))
  clo∈γ (t ≐ u) = sgl∈γ (key ι ιL (t ≐ u)) (pr∈γ _ _ (num∈γ _) (code∈γ (t ≐ u)))
  clo∈γ {n} ⊤̇ =
    let φt : Formula ⟪ fst A ⟫ n
        φt = ⊤̇
    in sgl∈γ (key ι ιL φt)
         (pr∈γ (# n) (VCode.⌜ mapFo ι φt ⌝) (num∈γ n) (code∈γ φt))
  clo∈γ {n} ⊥̇ =
    let φf : Formula ⟪ fst A ⟫ n
        φf = ⊥̇
    in sgl∈γ (key ι ιL φf)
         (pr∈γ (# n) (VCode.⌜ mapFo ι φf ⌝) (num∈γ n) (code∈γ φf))
  clo∈γ (a ∧̇ b) = cup∈γ _ _ (sgl∈γ (key ι ιL (a ∧̇ b)) (pr∈γ _ _ (num∈γ _) (code∈γ (a ∧̇ b))))
                    (cup∈γ _ _ (clo∈γ a) (clo∈γ b))
  clo∈γ (a ∨̇ b) = cup∈γ _ _ (sgl∈γ (key ι ιL (a ∨̇ b)) (pr∈γ _ _ (num∈γ _) (code∈γ (a ∨̇ b))))
                    (cup∈γ _ _ (clo∈γ a) (clo∈γ b))
  clo∈γ (a ⇒̇ b) = cup∈γ _ _ (sgl∈γ (key ι ιL (a ⇒̇ b)) (pr∈γ _ _ (num∈γ _) (code∈γ (a ⇒̇ b))))
                    (cup∈γ _ _ (clo∈γ a) (clo∈γ b))
  clo∈γ (¬̇ a)   = cup∈γ _ _ (sgl∈γ (key ι ιL (¬̇ a)) (pr∈γ _ _ (num∈γ _) (code∈γ (¬̇ a)))) (clo∈γ a)
  clo∈γ (∃̇ a)   = cup∈γ _ _ (sgl∈γ (key ι ιL (∃̇ a)) (pr∈γ _ _ (num∈γ _) (code∈γ (∃̇ a)))) (clo∈γ a)
  clo∈γ (∀̇ a)   = cup∈γ _ _ (sgl∈γ (key ι ιL (∀̇ a)) (pr∈γ _ _ (num∈γ _) (code∈γ (∀̇ a)))) (clo∈γ a)
  clo∈γ (∀̇∈ t a) = cup∈γ _ _ (sgl∈γ (key ι ιL (∀̇∈ t a)) (pr∈γ _ _ (num∈γ _) (code∈γ (∀̇∈ t a)))) (clo∈γ a)
  clo∈γ (∃̇∈ t a) = cup∈γ _ _ (sgl∈γ (key ι ιL (∃̇∈ t a)) (pr∈γ _ _ (num∈γ _) (code∈γ (∃̇∈ t a)))) (clo∈γ a)

-- ---------------------------------------------------------------------
-- SECTION 8.  THE L-SIDE STAGES.  The L-side key of a relabelled
-- formula carries the V-side key's stage across keyBridge;  the slot
-- and the table are staged by one structural fold mirroring `tree`'s
-- own clauses, with sgl and cup closed at gamma.
-- ---------------------------------------------------------------------

module LStages
  (γ : V ℓ) (oγ : IsOrd γ) (clγ : closedω γ) (ω∈γ : ⟨ ω ∈ˢᵥ γ ⟩)
  (A : S) (hA : ⟨ fst A ∈ Lset γ ⟩)
  (Hkey : keyS-in-carrier-lim)
  (Hsat : Sat-at-asConst)
  where
  open OpsAt γ oγ clγ ω∈γ
  open VStages γ oγ clγ ω∈γ A hA

  toS' : ∀ {n} → Formula ⟪ fst A ⟫ n → Formula S n
  toS' = mapFo (asConst A)

  key∈γ : ∀ {n} (χ : Formula ⟪ fst A ⟫ n) → ⟨ fst (keyS A χ) ∈ˢᵥ Lset γ ⟩
  key∈γ χ = Hkey γ oγ clγ ω∈γ A hA χ

  -- keyBridge: fst (keyS A χ) is the V-level value of fst (keyʟ (toS' χ))
  keyʟ∈γ : ∀ {n} (χ : Formula ⟪ fst A ⟫ n)
         → ⟨ fst (keyʟ (toS' χ)) ∈ˢᵥ Lset γ ⟩
  keyʟ∈γ χ = subst (λ w → ⟨ w ∈ˢᵥ Lset γ ⟩) (keyBridge A χ) (key∈γ χ)

  -- the Sat leaf: the hypothesis covers every arity of the carrier
  Sat∈γ : ∀ {n} (χ : Formula ⟪ fst A ⟫ n)
        → ⟨ fst (Sat A (toS' χ)) ∈ˢᵥ Lset γ ⟩
  Sat∈γ χ = Hsat γ oγ clγ ω∈γ A hA χ

  ent∈γ : ∀ {n} (χ : Formula ⟪ fst A ⟫ n)
        → ⟨ fst (ent A (toS' χ)) ∈ˢᵥ Lset γ ⟩
  ent∈γ χ =
    subst (λ w → ⟨ w ∈ˢᵥ Lset γ ⟩)
      (sym (prʟ-fst (keyʟ (toS' χ)) (Sat A (toS' χ))))
      (pr∈γ (fst (keyʟ (toS' χ))) (fst (Sat A (toS' χ)))
        (keyʟ∈γ χ) (Sat∈γ χ))

  -- sglʟ x and cupʟ x y staged at gamma, through their fst equations
  sglʟ∈γ : (x : S) → ⟨ fst x ∈ˢᵥ Lset γ ⟩ → ⟨ fst (sglʟ x) ∈ˢᵥ Lset γ ⟩
  sglʟ∈γ x h = subst (λ w → ⟨ w ∈ˢᵥ Lset γ ⟩) (sym (sglʟ-fst x)) (sgl∈γ (fst x) h)

  cupʟ∈γ : (x y : S) → ⟨ fst x ∈ˢᵥ Lset γ ⟩ → ⟨ fst y ∈ˢᵥ Lset γ ⟩
         → ⟨ fst (cupʟ x y) ∈ˢᵥ Lset γ ⟩
  cupʟ∈γ x y hx hy =
      subst (λ w → ⟨ w ∈ˢᵥ Lset γ ⟩) (sym (cupʟ-fst x y)) (cup∈γ (fst x) (fst y) hx hy)

  -- one fold for both `tree keyʟ` (the slot) and `tree ent` (the table)
  fold : (f : ∀ {m} → Formula S m → S)
       → (∀ {m} (χ : Formula ⟪ fst A ⟫ m) → ⟨ fst (f (toS' χ)) ∈ˢᵥ Lset γ ⟩)
       → ∀ {n} (χ : Formula ⟪ fst A ⟫ n)
       → ⟨ fst (tree A f (toS' χ)) ∈ˢᵥ Lset γ ⟩
  fold f leaf∈γ (t ∈̇ u) = sglʟ∈γ (f (toS' (t ∈̇ u))) (leaf∈γ (t ∈̇ u))
  fold f leaf∈γ (t ≐ u) = sglʟ∈γ (f (toS' (t ≐ u))) (leaf∈γ (t ≐ u))
  fold f leaf∈γ ⊤̇      = sglʟ∈γ (f (toS' ⊤̇)) (leaf∈γ ⊤̇)
  fold f leaf∈γ ⊥̇      = sglʟ∈γ (f (toS' ⊥̇)) (leaf∈γ ⊥̇)
  fold f leaf∈γ (a ∧̇ b) =
      cupʟ∈γ (sglʟ (f (toS' (a ∧̇ b)))) (cupʟ (tree A f (toS' a)) (tree A f (toS' b)))
      (sglʟ∈γ (f (toS' (a ∧̇ b))) (leaf∈γ (a ∧̇ b)))
      (cupʟ∈γ _ _ (fold f leaf∈γ a) (fold f leaf∈γ b))
  fold f leaf∈γ (a ∨̇ b) =
      cupʟ∈γ (sglʟ (f (toS' (a ∨̇ b)))) (cupʟ (tree A f (toS' a)) (tree A f (toS' b)))
      (sglʟ∈γ (f (toS' (a ∨̇ b))) (leaf∈γ (a ∨̇ b)))
      (cupʟ∈γ _ _ (fold f leaf∈γ a) (fold f leaf∈γ b))
  fold f leaf∈γ (a ⇒̇ b) =
      cupʟ∈γ (sglʟ (f (toS' (a ⇒̇ b)))) (cupʟ (tree A f (toS' a)) (tree A f (toS' b)))
      (sglʟ∈γ (f (toS' (a ⇒̇ b))) (leaf∈γ (a ⇒̇ b)))
      (cupʟ∈γ _ _ (fold f leaf∈γ a) (fold f leaf∈γ b))
  fold f leaf∈γ (¬̇ a) = cupʟ∈γ (sglʟ (f (toS' (¬̇ a)))) (tree A f (toS' a))
      (sglʟ∈γ (f (toS' (¬̇ a))) (leaf∈γ (¬̇ a))) (fold f leaf∈γ a)
  fold f leaf∈γ (∃̇ a) = cupʟ∈γ (sglʟ (f (toS' (∃̇ a)))) (tree A f (toS' a))
      (sglʟ∈γ (f (toS' (∃̇ a))) (leaf∈γ (∃̇ a))) (fold f leaf∈γ a)
  fold f leaf∈γ (∀̇ a) = cupʟ∈γ (sglʟ (f (toS' (∀̇ a)))) (tree A f (toS' a))
      (sglʟ∈γ (f (toS' (∀̇ a))) (leaf∈γ (∀̇ a))) (fold f leaf∈γ a)
  fold f leaf∈γ (∀̇∈ t a) = cupʟ∈γ (sglʟ (f (toS' (∀̇∈ t a)))) (tree A f (toS' a))
      (sglʟ∈γ (f (toS' (∀̇∈ t a))) (leaf∈γ (∀̇∈ t a))) (fold f leaf∈γ a)
  fold f leaf∈γ (∃̇∈ t a) = cupʟ∈γ (sglʟ (f (toS' (∃̇∈ t a)))) (tree A f (toS' a))
      (sglʟ∈γ (f (toS' (∃̇∈ t a))) (leaf∈γ (∃̇∈ t a))) (fold f leaf∈γ a)

  slot∈γ : ∀ {n} (χ : Formula ⟪ fst A ⟫ n) → ⟨ fst (slot A (toS' χ)) ∈ˢᵥ Lset γ ⟩
  slot∈γ = fold keyʟ keyʟ∈γ

  satTable∈γ : ∀ {n} (χ : Formula ⟪ fst A ⟫ n)
             → ⟨ fst (satTable A (toS' χ)) ∈ˢᵥ Lset γ ⟩
  satTable∈γ = fold (ent A) ent∈γ

-- ---------------------------------------------------------------------
-- SECTION 9.  THE DESCENT PACKAGE.  The bounded walks' staging moves,
-- landed as INSTANTIATIONS of EnvSupply's Fact module at the limit
-- stage (EnvSupply.lagda.md:450, prK at :453, valK at :463; the
-- transitivity is layer-trans (Lset-layer gamma), Constructible:183,
-- :246).  pr-down stages the components of a staged pair;  key-down
-- stages the three components of a member key through its arityTag
-- shape;  val-down stages a table entry's value.  The next dispatch's
-- closedAt / shapedAt / twelveAt walks call these;  nothing here is a
-- new construction.
-- ---------------------------------------------------------------------

module Desc (γ : V ℓ) (oγ : IsOrd γ) where
  open Fact (LsetS γ oγ) (layer-trans (Lset-layer γ)) public

  pr↓ : (u v : V ℓ) → ⟨ pr u v ∈ˢᵥ Lset γ ⟩
      → ⟨ u ∈ˢᵥ Lset γ ⟩ × ⟨ v ∈ˢᵥ Lset γ ⟩
  pr↓ = prK

  key↓ : (ar a b : S) (k : ℕ)
       → ⟨ pr (fst ar) (pr (# k) (pr (fst a) (fst b))) ∈ˢᵥ Lset γ ⟩
       → ⟨ fst ar ∈ˢᵥ Lset γ ⟩ × ⟨ fst a ∈ˢᵥ Lset γ ⟩ × ⟨ fst b ∈ˢᵥ Lset γ ⟩
  key↓ ar a b k hx = step1 .fst , step3 .fst , step3 .snd
    where
    step1 : ⟨ fst ar ∈ˢᵥ Lset γ ⟩ × ⟨ pr (# k) (pr (fst a) (fst b)) ∈ˢᵥ Lset γ ⟩
    step1 = pr↓ (fst ar) (pr (# k) (pr (fst a) (fst b))) hx
    step2 : ⟨ (# k) ∈ˢᵥ Lset γ ⟩ × ⟨ pr (fst a) (fst b) ∈ˢᵥ Lset γ ⟩
    step2 = pr↓ (# k) (pr (fst a) (fst b)) (step1 .snd)
    step3 : ⟨ fst a ∈ˢᵥ Lset γ ⟩ × ⟨ fst b ∈ˢᵥ Lset γ ⟩
    step3 = pr↓ (fst a) (fst b) (step2 .snd)


-- ---------------------------------------------------------------------
-- SECTION 10.  THE BOUNDED READING.  The Correct instance at the bound
-- c := LsetS gamma ogamma, the staging predicate, and the two quantifier
-- rules the walks are written with.  Under the bounded reading an
-- unbounded existential needs its witness IN the bound and an unbounded
-- universal ranges over the bound only;  atoms and bounded quantifiers
-- read as the standard ones do.
-- ---------------------------------------------------------------------

module Bnd (γ : V ℓ) (oγ : IsOrd γ) (clγ : closedω γ) (ω∈γ : ⟨ ω ∈ˢᵥ γ ⟩) where
  bnd : S
  bnd = LsetS γ oγ

  module CR = Correct (hPropAlgebra (ℓ-suc ℓ)) 𝒮ʟ (λ x → x) bnd
  open CR using ( _⊨ᴬ_ ) renaming ( A to Abnd ) public

  module D0B = D0 bnd

  Stg : S → Type (ℓ-suc ℓ)
  Stg x = ⟨ fst x ∈ˢᵥ Lset γ ⟩

  -- transitivity:  a member of a staged set is staged
  trv : (x : V ℓ) (Y : S) → Stg Y → ⟨ x ∈ˢᵥ fst Y ⟩ → ⟨ x ∈ˢᵥ Lset γ ⟩
  trv x Y hY hx = Lset-trans′ lem γ hx hY

  sg : (x : V ℓ) → ⟨ isL x ⟩ → S
  sg x h = x , h

  -- the staging of numerals, and of successors at the succ-closed limit
  num∈γ' : (k : ℕ) → ⟨ (# k) ∈ˢᵥ Lset γ ⟩
  num∈γ' k = Lset-mono {α = γ} {β = ω} ω∈γ (numeral∈limit k)

  numStg : (k : ℕ) → Stg (numeralL k)
  numStg k = subst (λ w → ⟨ w ∈ˢᵥ Lset γ ⟩) (sym (numeralL-fst k)) (num∈γ' k)

  module SM = SupplyMerge γ oγ (suc∈γ γ oγ clγ) (∅∈γ γ oγ ω∈γ)

  -- the two quantifier rules of the bounded reading
  ∃ᴬ-in : ∀ {n} (φ : Formula S (suc n)) (δ : S ^ n) (x : S)
        → ⟨ x ∈ˢ Abnd ⟩ → ⟨ (x ∷ δ) ⊨ᴬ φ ⟩ → ⟨ δ ⊨ᴬ ∃̇ φ ⟩
  ∃ᴬ-in φ δ x hx h = ∣ x , (hx , h) ∣₁

  ∀ᴬ-in : ∀ {n} (φ : Formula S (suc n)) (δ : S ^ n)
        → ((x : S) → ⟨ x ∈ˢ Abnd ⟩ → ⟨ (x ∷ δ) ⊨ᴬ φ ⟩)
        → ⟨ δ ⊨ᴬ ∀̇ φ ⟩
  ∀ᴬ-in φ δ k = k

  ∀ᴬ-out : ∀ {n} (φ : Formula S (suc n)) (δ : S ^ n)
          → ⟨ δ ⊨ᴬ ∀̇ φ ⟩ → (x : S) → ⟨ x ∈ˢ Abnd ⟩ → ⟨ (x ∷ δ) ⊨ᴬ φ ⟩
  ∀ᴬ-out φ δ h x hx = h x hx

-- ---------------------------------------------------------------------
-- SECTION 11.  THE BOUNDED READER INTROS.  Each landed reader, read
-- under the bounded semantics.  For the Delta0 readers (prAtL, sucAtL,
-- consAtL) the D0 transport moves the landed adequacy across;  for the
-- existential readers (tagAtL, tagPairAtL, arityTag*, appAt, subVal*)
-- the witness is built staged.  The ᴬ-out directions extract the meta
-- readings back, with the witnesses staged.
-- ---------------------------------------------------------------------

module Rd (γ : V ℓ) (oγ : IsOrd γ) (clγ : closedω γ) (ω∈γ : ⟨ ω ∈ˢᵥ γ ⟩) where
  open Bnd γ oγ clγ ω∈γ
  open OpsAt γ oγ clγ ω∈γ
  open Γ γ oγ clγ ω∈γ

  private
    module SE = SupplyEnv γ oγ (suc∈γ γ oγ clγ) (∅∈γ γ oγ ω∈γ)
                    ω ω-ord ω∈γ (self∈sucV ω)

  -- The BoundedFo certificates the D0 transport needs.  Every reader
  -- here carries no constants, so each leaf is the unit certificate.
  bddSglAt : ∀ {n} (k i : Fin n)
           → BoundedFo InL ((var i ∈̇ var k) ∧̇ (∀̇∈ (var k) (var zero ≐ var (suc i))))
  bddSglAt k i = ((_ , _) , (_ , (_ , _)))

  bddPairAt : ∀ {n} (k i j : Fin n)
            → BoundedFo InL ((var i ∈̇ var k)
                ∧̇ ((var j ∈̇ var k)
                  ∧̇ (∀̇∈ (var k) ((var zero ≐ var (suc i)) ∨̇ (var zero ≐ var (suc j))))))
  bddPairAt k i j = (_ , _) , ((_ , _) , (_ , ((_ , _) , (_ , _))))

  bddPrAt : ∀ {n} (q u v : Fin n) → BoundedFo InL (prAt q u v)
  bddPrAt q u v =
      ( _ , bddSglAt zero (suc u) )
    , ( ( _ , bddPairAt zero (suc u) (suc v) )
      , ( _ , ( bddSglAt zero (suc u) , bddPairAt zero (suc u) (suc v) ) ) )

  bddSucAt' : ∀ {n} (i j : Fin n) → BoundedFo InL (sucAt i j)
  bddSucAt' i j =
      (_ , _)
    , ( (_ , (_ , _)) , (_ , ((_ , _) , (_ , _))) )

  bddSgl0' : ∀ {n} (k : Fin n) → BoundedFo InL (sgl0At k)
  bddSgl0' k = (_ , (_ , _)) , (_ , (_ , _))

  bddPair0' : ∀ {n} (k j : Fin n) → BoundedFo InL (pair0At k j)
  bddPair0' k j = (_ , _) , ((_ , _) , (_ , ((_ , _) , (_ , _))))

  bddTag0' : ∀ {n} (s x : Fin n) → BoundedFo InL (tag0At s x)
  bddTag0' {n} s x =
      (_ , bddSgl0' {suc n} zero)
    , ( (_ , bddPair0' {suc n} zero (suc x))
      , (_ , (bddSgl0' {suc n} zero , bddPair0' {suc n} zero (suc x))) )

  bddShift' : ∀ {n} (p' p : Fin n) → BoundedFo InL (shiftPairAt p' p)
  bddShift' p' p = (_ , (_ , (_ , (_ , (_ , _)))))

  bddCons' : ∀ {n} (e' m e : Fin n) → BoundedFo InL (consAt e' m e)
  bddCons' {n} e' m e =
      (_ , bddTag0' {suc n} zero (suc m))
    , ( (_ , (_ , bddShift' {suc (suc n)} zero (suc zero)))
      , (_ , ( bddTag0' {suc n} zero (suc m)
             , (_ , bddShift' {suc (suc n)} (suc zero) zero) )) )

  -- inside the bounded reading, membership in the bound IS staging
  ab→st : (x : S) → ⟨ x ∈ˢ Abnd ⟩ → Stg x
  ab→st x h = h

  st→ab : (x : S) → Stg x → ⟨ x ∈ˢ Abnd ⟩
  st→ab x h = h

  -- pairing stays inside Lset gamma (Bound's package at the limit)
  pr∈γ' : (x y : V ℓ) → ⟨ x ∈ˢᵥ Lset γ ⟩ → ⟨ y ∈ˢᵥ Lset γ ⟩
        → ⟨ pr x y ∈ˢᵥ Lset γ ⟩
  pr∈γ' = pr∈λ


  -- staging a model-level pair from staged components
  prʟStg : (x y : S) → Stg x → Stg y → Stg (prʟ x y)
  prʟStg x y hx hy =
    subst (λ w → ⟨ w ∈ˢᵥ Lset γ ⟩) (sym (prʟ-fst x y))
      (pr∈γ' (fst x) (fst y) hx hy)



  -- Delta0 readers: adequacy + D0 transport
  prAtLᴬ-in : ∀ {n} (q u v : Fin n) (δ : S ^ n)
            → fst (lookup q δ) ≡ pr (fst (lookup u δ)) (fst (lookup v δ))
            → ⟨ δ ⊨ᴬ prAtL q u v ⟩
  prAtLᴬ-in q u v δ e =
    D0B.d0 (prAtL q u v) (Δ₀-liftFo (bddPrAt q u v) (Δ₀-prAt q u v)) δ
      (subst ⟨_⟩ (sym (prAtL-adequate q u v δ)) e)

  prAtLᴬ-out : ∀ {n} (q u v : Fin n) (δ : S ^ n)
             → ⟨ δ ⊨ᴬ prAtL q u v ⟩
             → fst (lookup q δ) ≡ pr (fst (lookup u δ)) (fst (lookup v δ))
  prAtLᴬ-out q u v δ h =
    subst ⟨_⟩ (prAtL-adequate q u v δ)
      (D0B.d0r (prAtL q u v) (Δ₀-liftFo (bddPrAt q u v) (Δ₀-prAt q u v)) δ h)

  sucAtLᴬ-in : ∀ {n} (i j : Fin n) (δ : S ^ n)
             → fst (lookup j δ) ≡ sucV (fst (lookup i δ))
             → ⟨ δ ⊨ᴬ sucAtL i j ⟩
  sucAtLᴬ-in i j δ e =
    D0B.d0 (sucAtL i j) (Δ₀-liftFo (bddSucAt' i j) (Δ₀-sucAt i j)) δ
      (subst ⟨_⟩ (sym (sucAtL-adequate i j δ)) e)

  sucAtLᴬ-out : ∀ {n} (i j : Fin n) (δ : S ^ n)
              → ⟨ δ ⊨ᴬ sucAtL i j ⟩
              → fst (lookup j δ) ≡ sucV (fst (lookup i δ))
  sucAtLᴬ-out i j δ h =
    subst ⟨_⟩ (sucAtL-adequate i j δ)
      (D0B.d0r (sucAtL i j) (Δ₀-liftFo (bddSucAt' i j) (Δ₀-sucAt i j)) δ h)

  consAtLᴬ-in : ∀ {n} (e' m e : Fin n) (δ : S ^ n)
              {k : ℕ} (g : Fin k → V ℓ)
              → fst (lookup e δ) ≡ env g
              → fst (lookup e' δ) ≡ env (cons (fst (lookup m δ)) g)
              → ⟨ δ ⊨ᴬ consAtL e' m e ⟩
  consAtLᴬ-in e' m e δ g hE h =
    D0B.d0 (consAtL e' m e) (Δ₀-liftFo (bddCons' e' m e) (Δ₀-consAt e' m e)) δ
      (subst ⟨_⟩ (sym (consAtL-adequate e' m e δ g hE)) h)

  consAtLᴬ-out : ∀ {n} (e' m e : Fin n) (δ : S ^ n)
               {k : ℕ} (g : Fin k → V ℓ)
               → fst (lookup e δ) ≡ env g
               → ⟨ δ ⊨ᴬ consAtL e' m e ⟩
               → fst (lookup e' δ) ≡ env (cons (fst (lookup m δ)) g)
  consAtLᴬ-out e' m e δ g hE h =
    subst ⟨_⟩ (consAtL-adequate e' m e δ g hE)
      (D0B.d0r (consAtL e' m e) (Δ₀-liftFo (bddCons' e' m e) (Δ₀-consAt e' m e)) δ h)

  -- appAt: bounded existential over the function; no staging needed
  appAtᴬ-in : ∀ {n} (f x y : Fin n) (δ : S ^ n)
            → ⟨ pr (fst (lookup x δ)) (fst (lookup y δ)) ∈ fst (lookup f δ) ⟩
            → ⟨ δ ⊨ᴬ appAt f x y ⟩
  appAtᴬ-in f x y δ h =
    ∣ zS , ( subst (λ w → ⟨ w ∈ˢᵥ fst (lookup f δ) ⟩) (sym (prʟ-fst (lookup x δ) (lookup y δ))) h
           , prAtLᴬ-in zero (suc x) (suc y) (zS ∷ δ)
               (prʟ-fst (lookup x δ) (lookup y δ)) ) ∣₁
    where
    zS : S
    zS = prʟ (lookup x δ) (lookup y δ)

  appAtᴬ-out : ∀ {n} (f x y : Fin n) (δ : S ^ n)
             → ⟨ δ ⊨ᴬ appAt f x y ⟩
             → ⟨ pr (fst (lookup x δ)) (fst (lookup y δ)) ∈ fst (lookup f δ) ⟩
  appAtᴬ-out f x y δ h =
    PT.rec (snd (pr (fst (lookup x δ)) (fst (lookup y δ)) ∈ fst (lookup f δ)))
      step h
    where
    step : Σ[ z ∈ S ] (⟨ fst z ∈ˢᵥ fst (lookup f δ) ⟩
                       × ⟨ (z ∷ δ) ⊨ᴬ prAtL zero (suc x) (suc y) ⟩)
         → ⟨ pr (fst (lookup x δ)) (fst (lookup y δ)) ∈ fst (lookup f δ) ⟩
    step (z , z∈ , hb) =
      subst (λ w → ⟨ w ∈ fst (lookup f δ) ⟩) (prAtLᴬ-out zero (suc x) (suc y) (z ∷ δ) hb) z∈

  -- tagAtL: unbounded existential, witness the numeral, staged
  tagAtLᴬ-in : ∀ {n} (s : Fin n) (k : ℕ) (x : Fin n) (δ : S ^ n)
             → fst (lookup s δ) ≡ pr (# k) (fst (lookup x δ))
             → ⟨ δ ⊨ᴬ tagAtL s k x ⟩
  tagAtLᴬ-in s k x δ e =
    ∃ᴬ-in ((var zero ≐ con (numeralL k))
             ∧̇ prAtL (suc s) zero (suc x)) δ (numeralL k) (numStg k)
      ( refl
      , prAtLᴬ-in (suc s) zero (suc x) (numeralL k ∷ δ)
          ( e ∙ cong (λ w → pr w (fst (lookup x δ))) (sym (numeralL-fst k))) )

  tagAtLᴬ-out : ∀ {n} (s : Fin n) (k : ℕ) (x : Fin n) (δ : S ^ n)
              → ⟨ δ ⊨ᴬ tagAtL s k x ⟩
              → fst (lookup s δ) ≡ pr (# k) (fst (lookup x δ))
  tagAtLᴬ-out s k x δ h =
    PT.rec (setIsSet (fst (lookup s δ)) (pr (# k) (fst (lookup x δ)))) step h
    where
    step : Σ[ z ∈ S ] (⟨ z ∈ˢ Abnd ⟩
                       × ⟨ (z ∷ δ) ⊨ᴬ ((var zero ≐ con (numeralL k))
                                        ∧̇ prAtL (suc s) zero (suc x)) ⟩)
         → fst (lookup s δ) ≡ pr (# k) (fst (lookup x δ))
    step (z , hz , hz≐ , hb) =
      prAtLᴬ-out (suc s) zero (suc x) (z ∷ δ) hb
        ∙ cong (λ w → pr w (fst (lookup x δ))) (hz≐ ∙ numeralL-fst k)

  -- tagPairAtL: unbounded existential, witness the inner code, staged
  tagPairAtLᴬ-in : ∀ {n} (s : Fin n) (k : ℕ) (a b : Fin n) (δ : S ^ n)
                 → Stg (lookup a δ) → Stg (lookup b δ)
                 → fst (lookup s δ) ≡ pr (# k) (pr (fst (lookup a δ)) (fst (lookup b δ)))
                 → ⟨ δ ⊨ᴬ tagPairAtL s k a b ⟩
  tagPairAtLᴬ-in s k a b δ ha hb e =
    ∃ᴬ-in (prAtL zero (suc a) (suc b) ∧̇ tagAtL (suc s) k zero) δ zS zS∈
      ( prAtLᴬ-in zero (suc a) (suc b) (zS ∷ δ) (prʟ-fst (lookup a δ) (lookup b δ))
      , tagAtLᴬ-in (suc s) k zero (zS ∷ δ) e' )
    where
    zS : S
    zS = prʟ (lookup a δ) (lookup b δ)
    zS∈ : ⟨ zS ∈ˢ Abnd ⟩
    zS∈ = st→ab zS (prʟStg (lookup a δ) (lookup b δ) ha hb)
    h' : pr (fst (lookup a δ)) (fst (lookup b δ)) ≡ fst zS
    h' = sym (prʟ-fst (lookup a δ) (lookup b δ))
    e' : fst (lookup s δ) ≡ pr (# k) (fst zS)
    e' = e ∙ cong (pr (# k)) h'


  tagPairAtLᴬ-out : ∀ {n} (s : Fin n) (k : ℕ) (a b : Fin n) (δ : S ^ n)
                  → ⟨ δ ⊨ᴬ tagPairAtL s k a b ⟩
                  → fst (lookup s δ) ≡ pr (# k) (pr (fst (lookup a δ)) (fst (lookup b δ)))
  tagPairAtLᴬ-out s k a b δ h = PT.rec (setIsSet (fst (lookup s δ))
                   (pr (# k) (pr (fst (lookup a δ)) (fst (lookup b δ))))) step h
    where
    step : Σ[ z ∈ S ] (⟨ z ∈ˢ Abnd ⟩
                       × ⟨ (z ∷ δ) ⊨ᴬ (prAtL zero (suc a) (suc b)
                                       ∧̇ tagAtL (suc s) k zero) ⟩)
         → fst (lookup s δ) ≡ pr (# k) (pr (fst (lookup a δ)) (fst (lookup b δ)))
    step (z , hz , hp , ht) =
      tagAtLᴬ-out (suc s) k zero (z ∷ δ) ht
        ∙ cong (pr (# k)) (prAtLᴬ-out zero (suc a) (suc b) (z ∷ δ) hp)

  -- arityTagPairAtL: witness the inner code z = pr (ar) (tag-pair), staged
  arityTagPairAtLᴬ-in : ∀ {n} (c ar : Fin n) (k : ℕ) (a b : Fin n) (δ : S ^ n)
                      → Stg (lookup ar δ) → Stg (lookup a δ) → Stg (lookup b δ)
                      → fst (lookup c δ) ≡ pr (fst (lookup ar δ))
                            (pr (# k) (pr (fst (lookup a δ)) (fst (lookup b δ))))
                      → ⟨ δ ⊨ᴬ arityTagPairAtL c ar k a b ⟩
  arityTagPairAtLᴬ-in c ar k a b δ har ha hb e =
    ∃ᴬ-in (prAtL (suc c) (suc ar) zero ∧̇ tagPairAtL zero k (suc a) (suc b)) δ zS zS∈
      ( prAtLᴬ-in (suc c) (suc ar) zero (zS ∷ δ) e'
      , tagPairAtLᴬ-in zero k (suc a) (suc b) (zS ∷ δ) ha hb inner )
    where
    zS : S
    zS = prʟ (numeralL k) (prʟ (lookup a δ) (lookup b δ))
    zS∈ : ⟨ zS ∈ˢ Abnd ⟩
    zS∈ = st→ab zS
      (prʟStg (numeralL k) (prʟ (lookup a δ) (lookup b δ)) (numStg k)
        (prʟStg (lookup a δ) (lookup b δ) ha hb))
    inner : fst zS ≡ pr (# k) (pr (fst (lookup a δ)) (fst (lookup b δ)))
    inner = prʟ-fst (numeralL k) (prʟ (lookup a δ) (lookup b δ))
          ∙ cong₂ pr (numeralL-fst k) (prʟ-fst (lookup a δ) (lookup b δ))
    g'' : pr (# k) (pr (fst (lookup a δ)) (fst (lookup b δ))) ≡ fst zS
    g'' = sym inner
    e' : fst (lookup c δ) ≡ pr (fst (lookup ar δ)) (fst zS)
    e' = e ∙ cong (pr (fst (lookup ar δ))) g''

  -- the consumed reading of the arity-tag antecedent: the shape equation
  arityTagPairAtLᴬ-out : ∀ {n} (c ar : Fin n) (k : ℕ) (a b : Fin n) (δ : S ^ n)
                       → ⟨ δ ⊨ᴬ arityTagPairAtL c ar k a b ⟩
                       → fst (lookup c δ) ≡ pr (fst (lookup ar δ))
                             (pr (# k) (pr (fst (lookup a δ)) (fst (lookup b δ))))
  arityTagPairAtLᴬ-out c ar k a b δ h = PT.rec (setIsSet (fst (lookup c δ))
    (pr (fst (lookup ar δ)) (pr (# k) (pr (fst (lookup a δ)) (fst (lookup b δ)))))) step h
    where
    step : Σ[ z ∈ S ] (⟨ z ∈ˢ Abnd ⟩
                       × ⟨ (z ∷ δ) ⊨ᴬ (prAtL (suc c) (suc ar) zero
                                       ∧̇ tagPairAtL zero k (suc a) (suc b)) ⟩)
         → fst (lookup c δ) ≡ pr (fst (lookup ar δ))
                             (pr (# k) (pr (fst (lookup a δ)) (fst (lookup b δ))))
    step (z , hz , hp , ht) =
      prAtLᴬ-out (suc c) (suc ar) zero (z ∷ δ) hp
        ∙ cong (pr (fst (lookup ar δ))) (tagPairAtLᴬ-out zero k (suc a) (suc b) (z ∷ δ) ht)

  arityTagAtLᴬ-in : ∀ {n} (c ar : Fin n) (k : ℕ) (a : Fin n) (δ : S ^ n)
                  → Stg (lookup ar δ) → Stg (lookup a δ)
                  → fst (lookup c δ) ≡ pr (fst (lookup ar δ)) (pr (# k) (fst (lookup a δ)))
                  → ⟨ δ ⊨ᴬ arityTagAtL c ar k a ⟩
  arityTagAtLᴬ-in c ar k a δ har ha e =
    ∃ᴬ-in (prAtL (suc c) (suc ar) zero ∧̇ tagAtL zero k (suc a)) δ zS zS∈
      ( prAtLᴬ-in (suc c) (suc ar) zero (zS ∷ δ) e'
      , tagAtLᴬ-in zero k (suc a) (zS ∷ δ) refl' )
    where
    zS : S
    zS = prʟ (numeralL k) (lookup a δ)
    zS∈ : ⟨ zS ∈ˢ Abnd ⟩
    zS∈ = st→ab zS
      (prʟStg (numeralL k) (lookup a δ) (numStg k) ha)
    h4 : pr (# k) (fst (lookup a δ)) ≡ fst zS
    h4 = cong₂ pr (sym (numeralL-fst k)) refl
        ∙ sym (prʟ-fst (numeralL k) (lookup a δ))
    e' : fst (lookup c δ) ≡ pr (fst (lookup ar δ)) (fst zS)
    e' = e ∙ cong (pr (fst (lookup ar δ))) h4
    refl' : fst zS ≡ pr (# k) (fst (lookup a δ))
    refl' = prʟ-fst (numeralL k) (lookup a δ)
          ∙ cong₂ pr (numeralL-fst k) refl

  arityTagAtLᴬ-out : ∀ {n} (c ar : Fin n) (k : ℕ) (a : Fin n) (δ : S ^ n)
                   → ⟨ δ ⊨ᴬ arityTagAtL c ar k a ⟩
                   → fst (lookup c δ) ≡ pr (fst (lookup ar δ)) (pr (# k) (fst (lookup a δ)))
  arityTagAtLᴬ-out c ar k a δ h = PT.rec (setIsSet (fst (lookup c δ))
    (pr (fst (lookup ar δ)) (pr (# k) (fst (lookup a δ))))) step h
    where
    step : Σ[ z ∈ S ] (⟨ z ∈ˢ Abnd ⟩
                       × ⟨ (z ∷ δ) ⊨ᴬ (prAtL (suc c) (suc ar) zero
                                       ∧̇ tagAtL zero k (suc a)) ⟩)
         → fst (lookup c δ) ≡ pr (fst (lookup ar δ)) (pr (# k) (fst (lookup a δ)))
    step (z , hz , hp , ht) =
      prAtLᴬ-out (suc c) (suc ar) zero (z ∷ δ) hp
        ∙ cong (pr (fst (lookup ar δ))) (tagAtLᴬ-out zero k (suc a) (z ∷ δ) ht)

  -- subValAt / subValSuccAt: consumed readings
  subValAtᴬ-out : ∀ {n} (T ar a y : Fin n) (δ : S ^ n)
                → ⟨ δ ⊨ᴬ subValAt T ar a y ⟩
                → ⟨ pr (pr (fst (lookup ar δ)) (fst (lookup a δ))) (fst (lookup y δ))
                    ∈ fst (lookup T δ) ⟩
  subValAtᴬ-out T ar a y δ h = PT.rec
    (snd (pr (pr (fst (lookup ar δ)) (fst (lookup a δ))) (fst (lookup y δ))
        ∈ fst (lookup T δ))) step h
    where
    step : Σ[ z ∈ S ] (⟨ z ∈ˢ Abnd ⟩
                       × ⟨ (z ∷ δ) ⊨ᴬ (prAtL zero (suc ar) (suc a)
                                       ∧̇ appAt (suc T) zero (suc y)) ⟩)
         → ⟨ pr (pr (fst (lookup ar δ)) (fst (lookup a δ))) (fst (lookup y δ))
            ∈ fst (lookup T δ) ⟩
    step (z , hz , hp , ha) =
      subst (λ w → ⟨ pr w (fst (lookup y δ)) ∈ fst (lookup T δ) ⟩)
        (prAtLᴬ-out zero (suc ar) (suc a) (z ∷ δ) hp)
        (appAtᴬ-out (suc T) zero (suc y) (z ∷ δ) ha)

  subValSuccAtᴬ-out : ∀ {n} (T ar a y : Fin n) (δ : S ^ n)
                    → ⟨ δ ⊨ᴬ subValSuccAt T ar a y ⟩
                    → ⟨ pr (pr (sucV (fst (lookup ar δ))) (fst (lookup a δ)))
                          (fst (lookup y δ)) ∈ fst (lookup T δ) ⟩
  subValSuccAtᴬ-out T ar a y δ h = PT.rec
    (snd (pr (pr (sucV (fst (lookup ar δ))) (fst (lookup a δ))) (fst (lookup y δ))
        ∈ fst (lookup T δ))) step h
    where
    step : Σ[ z ∈ S ] (⟨ z ∈ˢ Abnd ⟩
                       × ⟨ (z ∷ δ) ⊨ᴬ (sucAtL (suc ar) zero
                                       ∧̇ subValAt (suc T) zero (suc a) (suc y)) ⟩)
         → ⟨ pr (pr (sucV (fst (lookup ar δ))) (fst (lookup a δ))) (fst (lookup y δ))
            ∈ fst (lookup T δ) ⟩
    step (z , hz , hs , hv) =
      subst (λ w → ⟨ pr (pr w (fst (lookup a δ))) (fst (lookup y δ)) ∈ fst (lookup T δ) ⟩)
        (sucAtLᴬ-out (suc ar) zero (z ∷ δ) hs)
        (subValAtᴬ-out (suc T) zero (suc a) (suc y) (z ∷ δ) hv)

  -- extAt under the bounded reading: both universal halves restrict
  extAtᴬ-in : ∀ {n} (y : Fin n) (φ : Formula S (suc n)) (δ : S ^ n)
            → ((z : S) → Stg z → ⟨ fst z ∈ fst (lookup y δ) ⟩ → ⟨ (z ∷ δ) ⊨ᴬ φ ⟩)
            → ((z : S) → Stg z → ⟨ (z ∷ δ) ⊨ᴬ φ ⟩ → ⟨ fst z ∈ fst (lookup y δ) ⟩)
            → ⟨ δ ⊨ᴬ extAt y φ ⟩
  extAtᴬ-in y φ δ fwd bwd = fwd , bwd

  extAtᴬ-fwd : ∀ {n} (y : Fin n) (φ : Formula S (suc n)) (δ : S ^ n)
             → ⟨ δ ⊨ᴬ extAt y φ ⟩ → (z : S) → Stg z
             → ⟨ fst z ∈ fst (lookup y δ) ⟩ → ⟨ (z ∷ δ) ⊨ᴬ φ ⟩
  extAtᴬ-fwd y φ δ h z hz zin = h .fst z hz zin

  extAtᴬ-bwd : ∀ {n} (y : Fin n) (φ : Formula S (suc n)) (δ : S ^ n)
             → ⟨ δ ⊨ᴬ extAt y φ ⟩ → (z : S) → Stg z
             → ⟨ (z ∷ δ) ⊨ᴬ φ ⟩ → ⟨ fst z ∈ fst (lookup y δ) ⟩
  extAtᴬ-bwd y φ δ h z hz hφ = h .snd z hz hφ

  -- domAt: unbounded forall over the bound; the forward half only
  domAtᴬ-out : ∀ {n} (f d : Fin n) (δ : S ^ n)
             → ⟨ δ ⊨ᴬ domAt f d ⟩
             → (x y : S) → Stg x → Stg y
             → ⟨ pr (fst x) (fst y) ∈ fst (lookup f δ) ⟩
             → ⟨ fst x ∈ fst (lookup d δ) ⟩
  domAtᴬ-out f d δ h x y hx hy hp = h x hx .fst antecedent
    where
    antecedent : ⟨ (x ∷ δ) ⊨ᴬ inDomAt (suc f) zero ⟩
    antecedent =
      ∣ y , st→ab y hy
      , appAtᴬ-in (suc (suc f)) (suc zero) zero (y ∷ x ∷ δ) hp ∣₁

  -- staging the successor witness at the succ-closed limit
  sucStg : (ar : S) → Stg ar → Stg (sucʟ ar)
  sucStg ar har =
    subst (λ w → ⟨ w ∈ˢᵥ Lset γ ⟩) (sym (sucʟ-fst ar)) (SE.sucK (fst ar) har)

  -- the clause frames under the bounded reading.  The ∀̇∈ filter reads
  -- as the standard one;  the ∀̇ binders arrive staged.
  binClauseᴬ-in : ∀ {n} (C T : Fin n) (k : ℕ) (rel : Formula S (5 + n)) (δ : S ^ n)
    → ((c ar a b yc : S)
       → ⟨ fst c ∈ fst (lookup C δ) ⟩
       → Stg ar → Stg a → Stg b → Stg yc
       → fst c ≡ pr (fst ar) (pr (# k) (pr (fst a) (fst b)))
       → ⟨ pr (fst c) (fst yc) ∈ fst (lookup T δ) ⟩
       → ⟨ (yc ∷ b ∷ a ∷ ar ∷ c ∷ δ) ⊨ᴬ rel ⟩)
    → ⟨ δ ⊨ᴬ binClauseAt C T k rel ⟩
  binClauseᴬ-in {n} C T k rel δ g =
    λ c c∈ ar har a ha b hb yc hyc sh hac →
      g c ar a b yc c∈ (ab→st ar har) (ab→st a ha) (ab→st b hb) (ab→st yc hyc)
        (arityTagPairAtLᴬ-out c5 n5 k a5 b5 (yc ∷ b ∷ a ∷ ar ∷ c ∷ δ) sh)
        (appAtᴬ-out (sh5 T) c5 yc5 (yc ∷ b ∷ a ∷ ar ∷ c ∷ δ) hac)
    where
    c5 n5 a5 b5 yc5 : Fin (5 + n)
    c5 = suc (suc (suc (suc zero)))
    n5 = suc (suc (suc zero))
    a5 = suc (suc zero)
    b5 = suc zero
    yc5 = zero
    sh5 : Fin _ → Fin _
    sh5 i = suc (suc (suc (suc (suc i))))

  unClauseᴬ-in : ∀ {n} (C T : Fin n) (k : ℕ) (rel : Formula S (4 + n)) (δ : S ^ n)
    → ((c ar a yc : S)
       → ⟨ fst c ∈ fst (lookup C δ) ⟩
       → Stg ar → Stg a → Stg yc
       → fst c ≡ pr (fst ar) (pr (# k) (fst a))
       → ⟨ pr (fst c) (fst yc) ∈ fst (lookup T δ) ⟩
       → ⟨ (yc ∷ a ∷ ar ∷ c ∷ δ) ⊨ᴬ rel ⟩)
    → ⟨ δ ⊨ᴬ unClauseAt C T k rel ⟩
  unClauseᴬ-in {n} C T k rel δ g =
    λ c c∈ ar har a ha yc hyc sh hac →
      g c ar a yc c∈ (ab→st ar har) (ab→st a ha) (ab→st yc hyc)
        (arityTagAtLᴬ-out c4 n4 k a4 (yc ∷ a ∷ ar ∷ c ∷ δ) sh)
        (appAtᴬ-out (sh4 T) c4 yc4 (yc ∷ a ∷ ar ∷ c ∷ δ) hac)
    where
    c4 n4 a4 yc4 : Fin (4 + n)
    c4 = suc (suc (suc zero))
    n4 = suc (suc zero)
    a4 = suc zero
    yc4 = zero
    sh4 : Fin _ → Fin _
    sh4 i = suc (suc (suc (suc i)))

module ClW (γ : V ℓ) (oγ : IsOrd γ) (clγ : closedω γ) (ω∈γ : ⟨ ω ∈ˢᵥ γ ⟩) where
  open Bnd γ oγ clγ ω∈γ
  open Rd γ oγ clγ ω∈γ

  -- the four shapes, bounded
  binSameClosedᴬ-in : ∀ {n} (C : Fin n) (k : ℕ) (δ : S ^ n)
    → ((c ar a b : S)
       → ⟨ fst c ∈ fst (lookup C δ) ⟩
       → fst c ≡ pr (fst ar) (pr (# k) (pr (fst a) (fst b)))
       → ⟨ pr (fst ar) (fst a) ∈ fst (lookup C δ) ⟩
         × ⟨ pr (fst ar) (fst b) ∈ fst (lookup C δ) ⟩)
    → ⟨ δ ⊨ᴬ binShapeAt C k (bothSameAt C) ⟩
  binSameClosedᴬ-in {n} C k δ use =
    λ c c∈ ar har a ha b hb sh →
      let (m1 , m2) =
            use c ar a b c∈
              (arityTagPairAtLᴬ-out c4 n4 k a4 b4 (b ∷ a ∷ ar ∷ c ∷ δ) sh)
      in appAtᴬ-in (sh4 C) n4 a4 (b ∷ a ∷ ar ∷ c ∷ δ) m1
       , appAtᴬ-in (sh4 C) n4 b4 (b ∷ a ∷ ar ∷ c ∷ δ) m2
    where
    c4 n4 a4 b4 : Fin (4 + n)
    c4 = suc (suc (suc zero))
    n4 = suc (suc zero)
    a4 = suc zero
    b4 = zero
    sh4 : Fin _ → Fin _
    sh4 i = suc (suc (suc (suc i)))

  unSameClosedᴬ-in : ∀ {n} (C : Fin n) (k : ℕ) (δ : S ^ n)
    → ((c ar a : S)
       → ⟨ fst c ∈ fst (lookup C δ) ⟩
       → fst c ≡ pr (fst ar) (pr (# k) (fst a))
       → ⟨ pr (fst ar) (fst a) ∈ fst (lookup C δ) ⟩)
    → ⟨ δ ⊨ᴬ unShapeAt C k (oneSameAt C) ⟩
  unSameClosedᴬ-in {n} C k δ use =
    λ c c∈ ar har a ha sh →
      appAtᴬ-in (sh3 C) n3 a3 (a ∷ ar ∷ c ∷ δ)
        (use c ar a c∈
          (arityTagAtLᴬ-out c3 n3 k a3 (a ∷ ar ∷ c ∷ δ) sh))
    where
    c3 n3 a3 : Fin (3 + n)
    c3 = suc (suc zero)
    n3 = suc zero
    a3 = zero
    sh3 : Fin _ → Fin _
    sh3 i = suc (suc (suc i))

  unSuccClosedᴬ-in : ∀ {n} (C : Fin n) (k : ℕ) (δ : S ^ n)
    → ((c ar a : S)
       → ⟨ fst c ∈ fst (lookup C δ) ⟩
       → fst c ≡ pr (fst ar) (pr (# k) (fst a))
       → ⟨ pr (sucV (fst ar)) (fst a) ∈ fst (lookup C δ) ⟩)
    → ⟨ δ ⊨ᴬ unShapeAt C k (oneSuccAt C) ⟩
  unSuccClosedᴬ-in {n} C k δ use =
    λ c c∈ ar har a ha sh →
      ∣ sucʟ ar , st→ab (sucʟ ar) (sucStg ar har)
      , ( sucAtLᴬ-in (suc n3) zero (sucʟ ar ∷ a ∷ ar ∷ c ∷ δ) (sucʟ-fst ar)
        , appAtᴬ-in (suc (sh3 C)) zero (suc a3) (sucʟ ar ∷ a ∷ ar ∷ c ∷ δ)
            (subst (λ w → ⟨ pr w (fst a) ∈ fst (lookup C δ) ⟩)
              (sym (sucʟ-fst ar))
              (use c ar a c∈
                (arityTagAtLᴬ-out c3 n3 k a3 (a ∷ ar ∷ c ∷ δ) sh))) ) ∣₁
    where
    c3 n3 a3 : Fin (3 + n)
    c3 = suc (suc zero)
    n3 = suc zero
    a3 = zero
    sh3 : Fin _ → Fin _
    sh3 i = suc (suc (suc i))

  binSuccClosedᴬ-in : ∀ {n} (C : Fin n) (k : ℕ) (δ : S ^ n)
    → ((c ar a b : S)
       → ⟨ fst c ∈ fst (lookup C δ) ⟩
       → fst c ≡ pr (fst ar) (pr (# k) (pr (fst a) (fst b)))
       → ⟨ pr (sucV (fst ar)) (fst b) ∈ fst (lookup C δ) ⟩)
    → ⟨ δ ⊨ᴬ binShapeAt C k (succSndAt C) ⟩
  binSuccClosedᴬ-in {n} C k δ use =
    λ c c∈ ar har a ha b hb sh →
      ∣ sucʟ ar , st→ab (sucʟ ar) (sucStg ar har)
      , ( sucAtLᴬ-in (suc n4) zero (sucʟ ar ∷ b ∷ a ∷ ar ∷ c ∷ δ) (sucʟ-fst ar)
        , appAtᴬ-in (suc (sh4 C)) zero (suc b4) (sucʟ ar ∷ b ∷ a ∷ ar ∷ c ∷ δ)
            (subst (λ w → ⟨ pr w (fst b) ∈ fst (lookup C δ) ⟩)
              (sym (sucʟ-fst ar))
              (use c ar a b c∈
                (arityTagPairAtLᴬ-out c4 n4 k a4 b4 (b ∷ a ∷ ar ∷ c ∷ δ) sh))) ) ∣₁
    where
    c4 n4 a4 b4 : Fin (4 + n)
    c4 = suc (suc (suc zero))
    n4 = suc (suc zero)
    a4 = suc zero
    b4 = zero
    sh4 : Fin _ → Fin _
    sh4 i = suc (suc (suc (suc i)))

-- ---------------------------------------------------------------------
-- SECTION 14.  THE CLOSEDNESS INSTANCE.  One generic closedOf over any
-- peeling set (Closed.lagda.md's theorem, re-read under the bounded
-- semantics), and the closure instance the code predicate consumes.
-- ---------------------------------------------------------------------

module ClOf {K : Type ℓ} (f : K → V ℓ) (h : (k : K) → ⟨ isL (f k) ⟩)
            (γ : V ℓ) (oγ : IsOrd γ) (clγ : closedω γ) (ω∈γ : ⟨ ω ∈ˢᵥ γ ⟩) where
  open Bnd γ oγ clγ ω∈γ
  open Rd γ oγ clγ ω∈γ
  open ClW γ oγ clγ ω∈γ

  private
    Cl : ∀ {m} → Formula K m → V ℓ
    Cl = closure f h

  Peel : (D : S) → Type (ℓ-suc ℓ)
  Peel D = (x : V ℓ) → ⟨ x ∈ fst D ⟩
           → ∥ (Σ[ m ∈ ℕ ] Σ[ ψ ∈ Formula K m ]
                 ((x ≡ key f h ψ)
                  × ((z : V ℓ) → ⟨ z ∈ Cl ψ ⟩ → ⟨ z ∈ fst D ⟩))) ∥₁

  closedOfᴬ : (D : S) → Peel D
            → ∀ {m} (δ : S ^ m) (i : Fin m) (p : fst (lookup i δ) ≡ fst D)
            → ⟨ δ ⊨ᴬ closedAt i ⟩
  closedOfᴬ D peel {m} δ i p =
      same i 2 p (λ ar a b r →
        ( subst (λ w → ⟨ pr ar a ∈ w ⟩) (sym p) (r a b refl .fst)
        , subst (λ w → ⟨ pr ar b ∈ w ⟩) (sym p) (r a b refl .snd) ))
    , ( same i 3 p (λ ar a b r →
          ( subst (λ w → ⟨ pr ar a ∈ w ⟩) (sym p) (r a b refl .fst)
          , subst (λ w → ⟨ pr ar b ∈ w ⟩) (sym p) (r a b refl .snd) ))
    , ( same i 4 p (λ ar a b r →
            ( subst (λ w → ⟨ pr ar a ∈ w ⟩) (sym p) (r a b refl .fst)
            , subst (λ w → ⟨ pr ar b ∈ w ⟩) (sym p) (r a b refl .snd) ))
    , ( one i 5 p (λ ar a r →
            subst (λ w → ⟨ pr ar a ∈ w ⟩) (sym p) r)
    , ( up i 8 p (λ ar a r →
            subst (λ w → ⟨ pr (sucV ar) a ∈ w ⟩) (sym p) r)
    , ( up i 9 p (λ ar a r →
            subst (λ w → ⟨ pr (sucV ar) a ∈ w ⟩) (sym p) r)
    , ( sndUp i 10 p (λ ar a b r →
            subst (λ w → ⟨ pr (sucV ar) b ∈ w ⟩) (sym p) (r a b refl))
    , sndUp i 11 p (λ ar a b r →
            subst (λ w → ⟨ pr (sucV ar) b ∈ w ⟩) (sym p) (r a b refl)) ))))))
    where
    cin : ∀ {j : Fin m} → fst (lookup j δ) ≡ fst D
        → (x : V ℓ) → ⟨ x ∈ fst (lookup j δ) ⟩ → ⟨ x ∈ fst D ⟩
    cin q x h = subst (λ w → ⟨ x ∈ w ⟩) q h

    viaKey : (k : ℕ) (c : S) (ar pv : V ℓ)
           → ⟨ fst c ∈ fst D ⟩ → fst c ≡ pr ar (pr (# k) pv)
           → (T : Type (ℓ-suc ℓ)) → isProp T
           → (Concl f h (fst D) k ar pv → T) → T
    viaKey k c ar pv c∈ sh T pT g = PT.rec pT
      (λ { (mm , ψ , q , incl) →
        g (byTag f h (fst D) ψ k ar pv incl (sym q ∙ sh)) })
      (peel (fst c) c∈)

    same : ∀ (j : Fin m) (k : ℕ) (q : fst (lookup j δ) ≡ fst D)
         → ((ar a b : V ℓ) → Concl f h (fst D) k ar (pr a b)
            → ⟨ pr ar a ∈ fst (lookup j δ) ⟩ × ⟨ pr ar b ∈ fst (lookup j δ) ⟩)
         → ⟨ δ ⊨ᴬ binShapeAt j k (bothSameAt j) ⟩
    same j k q use = binSameClosedᴬ-in j k δ
      (λ c ar a b c∈ sh →
        viaKey k c (fst ar) (pr (fst a) (fst b)) (subst (λ w → ⟨ fst c ∈ w ⟩) q c∈) sh
          (⟨ pr (fst ar) (fst a) ∈ fst (lookup j δ) ⟩
             × ⟨ pr (fst ar) (fst b) ∈ fst (lookup j δ) ⟩)
          (isProp× (snd (pr (fst ar) (fst a) ∈ fst (lookup j δ)))
                   (snd (pr (fst ar) (fst b) ∈ fst (lookup j δ))))
          (λ r → use (fst ar) (fst a) (fst b) r))
    one : ∀ (j : Fin m) (k : ℕ) (q : fst (lookup j δ) ≡ fst D)
        → ((ar a : V ℓ) → Concl f h (fst D) k ar a
           → ⟨ pr ar a ∈ fst (lookup j δ) ⟩)
        → ⟨ δ ⊨ᴬ unShapeAt j k (oneSameAt j) ⟩
    one j k q use = unSameClosedᴬ-in j k δ
      (λ c ar a c∈ sh →
        viaKey k c (fst ar) (fst a) (subst (λ w → ⟨ fst c ∈ w ⟩) q c∈) sh
          (⟨ pr (fst ar) (fst a) ∈ fst (lookup j δ) ⟩)
          (snd (pr (fst ar) (fst a) ∈ fst (lookup j δ)))
          (λ r → use (fst ar) (fst a) r))
    up : ∀ (j : Fin m) (k : ℕ) (q : fst (lookup j δ) ≡ fst D)
       → ((ar a : V ℓ) → Concl f h (fst D) k ar a
          → ⟨ pr (sucV ar) a ∈ fst (lookup j δ) ⟩)
       → ⟨ δ ⊨ᴬ unShapeAt j k (oneSuccAt j) ⟩
    up j k q use = unSuccClosedᴬ-in j k δ
      (λ c ar a c∈ sh →
        viaKey k c (fst ar) (fst a) (subst (λ w → ⟨ fst c ∈ w ⟩) q c∈) sh
          (⟨ pr (sucV (fst ar)) (fst a) ∈ fst (lookup j δ) ⟩)
          (snd (pr (sucV (fst ar)) (fst a) ∈ fst (lookup j δ)))
          (λ r → use (fst ar) (fst a) r))
    sndUp : ∀ (j : Fin m) (k : ℕ) (q : fst (lookup j δ) ≡ fst D)
          → ((ar a b : V ℓ) → Concl f h (fst D) k ar (pr a b)
             → ⟨ pr (sucV ar) b ∈ fst (lookup j δ) ⟩)
          → ⟨ δ ⊨ᴬ binShapeAt j k (succSndAt j) ⟩
    sndUp j k q use = binSuccClosedᴬ-in j k δ
      (λ c ar a b c∈ sh →
        viaKey k c (fst ar) (pr (fst a) (fst b)) (subst (λ w → ⟨ fst c ∈ w ⟩) q c∈) sh
          (⟨ pr (sucV (fst ar)) (fst b) ∈ fst (lookup j δ) ⟩)
          (snd (pr (sucV (fst ar)) (fst b) ∈ fst (lookup j δ)))
          (λ r → use (fst ar) (fst a) (fst b) r))

  closureClosedᴬ : ∀ {n m} (φ : Formula K n) (δ : S ^ (suc m))
                 → ⟨ (clo f h φ ∷ δ) ⊨ᴬ closedAt zero ⟩
  closureClosedᴬ φ δ = closedOfᴬ (clo f h φ) (closure-inv f h φ)
    (clo f h φ ∷ δ) zero refl

-- ---------------------------------------------------------------------
-- SECTION 15.  THE SHAPE WALK UNDER THE BOUNDED READING.  Shape.lagda.md's
-- isTmAt introduction and the twelve-alternative walk, re-read under the
-- bounded semantics: the existential witnesses are constructed staged
-- (numerals, term codes, subcodes) and the leaf relations are the
-- bounded tag intros.
-- ---------------------------------------------------------------------

module ShW {K : Type ℓ} (f : K → V ℓ) (h : (k : K) → ⟨ isL (f k) ⟩)
            (γ : V ℓ) (oγ : IsOrd γ) (clγ : closedω γ) (ω∈γ : ⟨ ω ∈ˢᵥ γ ⟩) where
  open Bnd γ oγ clγ ω∈γ
  open Rd γ oγ clγ ω∈γ
  open ClW γ oγ clγ ω∈γ

  open import Cubical.Data.FinData.Properties using ( toℕ<n )
  open import Cubical.Data.Sum using ( inl; inr )
  open import Cubical.Data.Unit using ( tt* )
  open import L.Coding.InL {ℓ} using ( codeL; codeTmL )
  open import L.Coding.Shape {ℓ}
    using ( isTmAt; binForm; unForm; shapes; bothTm; fstTm; shapedAt )
  open import FOL.Manipulation.Relabelling using ( mapTm; mapFo )

  TmWit' : ℕ → V ℓ → Type (ℓ-suc ℓ)
  TmWit' n x = Σ[ t ∈ Term K n ] (VCode.⌜ mapTm f t ⌝ᵗ ≡ x)

  nn : ℕ → S
  nn m = # m , numL m

  isTmAtᴬ-in : ∀ {m} (t N A : Fin m) (γ' : S ^ m) (k : ℕ)
             → fst (lookup N γ') ≡ # k
             → Stg (lookup A γ')
             → ((c : K) → ⟨ f c ∈ˢᵥ fst (lookup A γ') ⟩)
             → TmWit' k (fst (lookup t γ')) → ⟨ γ' ⊨ᴬ isTmAt t N A ⟩
  isTmAtᴬ-in t N A γ' k qN astg intoc (con c , e) =
    ∣ inl ∣ y , ( st→ab y (trv (fst y) (lookup A γ') astg (intoc c))
                , ( tagAtLᴬ-in (suc t) 0 zero (y ∷ γ') (sym e)
                  , intoc c ) ) ∣₁ ∣₁
    where
    y : S
    y = f c , h c
  isTmAtᴬ-in t N A γ' k qN astg intoc (var i , e) =
    ∣ inr ∣ z , ( st→ab z (numStg (toℕ i))
                , ( tagAtLᴬ-in (suc t) 1 zero (z ∷ γ')
                      (sym e ∙ cong (VCode.mkTag 1) (sym (numeralL-fst (toℕ i))))
                  , subst (λ w → ⟨ fst z ∈ w ⟩) (sym qN)
                      (subst (λ w → ⟨ w ∈ (# k) ⟩) (sym (numeralL-fst (toℕ i)))
                        (#mono (toℕ i) k (toℕ<n i))) ) ) ∣₁ ∣₁
    where
    z : S
    z = numeralL (toℕ i)

  binFormᴬ-wit : ∀ (k : ℕ) {n} (rel : Formula S (4 + n)) (δ : S ^ n)
               → (c0 b a N : S) → Stg b → Stg a → Stg N
               → fst c0 ≡ pr (fst N) (pr (# k) (pr (fst a) (fst b)))
               → ⟨ (b ∷ a ∷ N ∷ c0 ∷ δ) ⊨ᴬ rel ⟩
               → ⟨ (c0 ∷ δ) ⊨ᴬ binForm k rel ⟩
  binFormᴬ-wit k {n} rel δ c0 b a N hb ha hN eq relᴬ =
    ∣ N , st→ab N hN , ∣ a , st→ab a ha , ∣ b , st→ab b hb
      , ( arityTagPairAtLᴬ-in s3 s2 k s1 s0 (b ∷ a ∷ N ∷ c0 ∷ δ) hN ha hb eq
        , relᴬ ) ∣₁ ∣₁ ∣₁
    where
    s3 s2 s1 s0 : Fin (4 + n)
    s3 = suc (suc (suc zero))
    s2 = suc (suc zero)
    s1 = suc zero
    s0 = zero

  unFormᴬ-wit : ∀ (k : ℕ) {n} (rel : Formula S (3 + n)) (δ : S ^ n)
              → (c0 a N : S) → Stg a → Stg N
              → fst c0 ≡ pr (fst N) (pr (# k) (fst a))
              → ⟨ (a ∷ N ∷ c0 ∷ δ) ⊨ᴬ rel ⟩
              → ⟨ (c0 ∷ δ) ⊨ᴬ unForm k rel ⟩
  unFormᴬ-wit k {n} rel δ c0 a N ha hN eq relᴬ =
    ∣ N , st→ab N hN , ∣ a , st→ab a ha
      , ( arityTagAtLᴬ-in s2 s1 k s0 (a ∷ N ∷ c0 ∷ δ) hN ha eq
        , relᴬ ) ∣₁ ∣₁
    where
    s2 s1 s0 : Fin (3 + n)
    s2 = suc (suc zero)
    s1 = suc zero
    s0 = zero

  cdS : ∀ {m} → Formula K m → S
  cdS ψ = VCode.⌜ mapFo f ψ ⌝ , codeL f h ψ

  ClS : ∀ {m} → Formula K m → V ℓ
  ClS = closure f h

  PeelD : (D : S) → Type (ℓ-suc ℓ)
  PeelD D = (x : V ℓ) → ⟨ x ∈ fst D ⟩
           → ∥ (Σ[ m ∈ ℕ ] Σ[ ψ ∈ Formula K m ]
                 ((x ≡ key f h ψ)
                  × ((z : V ℓ) → ⟨ z ∈ ClS ψ ⟩ → ⟨ z ∈ fst D ⟩))) ∥₁

  ctS : ∀ {m} → Term K m → S
  ctS t = VCode.⌜ mapTm f t ⌝ᵗ , codeTmL f h t

  shapedOfᴬ : (D : S) → PeelD D
            → ∀ {m} (A : Fin m) (δ : S ^ m)
            → ((k : K) → ⟨ f k ∈ fst (lookup A δ) ⟩)
            → Stg (lookup A δ)
            → (∀ {m'} (t : Term K m') → Stg (ctS t))
            → (∀ {m'} (χ : Formula K m') → Stg (cdS χ))
            → (cS : S) → ⟨ fst cS ∈ fst D ⟩
            → ⟨ (cS ∷ D ∷ δ) ⊨ᴬ shapes (suc A) ⟩
  shapedOfᴬ D peel A δ intoc astg ctst cdst cS c∈ =
    PT.rec (snd ((cS ∷ D ∷ δ) ⊨ᴬ shapes (suc A))) go'
      (peel (fst cS) c∈)
    where
    s4 : ∀ {m} → Fin m → Fin (5 + m)
    s4 i = suc (suc (suc (suc (suc i))))

    tm1 : ∀ {m} (t : Term K m) (b a0 N0 c0 : S)
        → fst N0 ≡ # m
        → fst a0 ≡ VCode.⌜ mapTm f t ⌝ᵗ
        → ⟨ (b ∷ a0 ∷ N0 ∷ c0 ∷ D ∷ δ)
            ⊨ᴬ isTmAt (suc zero) (suc (suc zero)) (s4 A) ⟩
    tm1 {m} t b a0 N0 c0 qN qa =
      isTmAtᴬ-in (suc zero) (suc (suc zero)) (s4 A)
        (b ∷ a0 ∷ N0 ∷ c0 ∷ D ∷ δ) m qN astg intoc (t , sym qa)

    tm0 : ∀ {m} (u : Term K m) (b a0 N0 c0 : S)
        → fst N0 ≡ # m
        → fst b ≡ VCode.⌜ mapTm f u ⌝ᵗ
        → ⟨ (b ∷ a0 ∷ N0 ∷ c0 ∷ D ∷ δ)
            ⊨ᴬ isTmAt zero (suc (suc zero)) (s4 A) ⟩
    tm0 {m} u b a0 N0 c0 qN qb =
      isTmAtᴬ-in zero (suc (suc zero)) (s4 A)
        (b ∷ a0 ∷ N0 ∷ c0 ∷ D ∷ δ) m qN astg intoc (u , sym qb)

    bothᴬ : ∀ {m} (t u : Term K m) (b a0 N0 c0 : S)
          → fst N0 ≡ # m
          → fst a0 ≡ VCode.⌜ mapTm f t ⌝ᵗ
          → fst b ≡ VCode.⌜ mapTm f u ⌝ᵗ
          → ⟨ (b ∷ a0 ∷ N0 ∷ c0 ∷ D ∷ δ) ⊨ᴬ bothTm (suc A) ⟩
    bothᴬ t u b a0 N0 c0 qN qa qb =
      tm1 t b a0 N0 c0 qN qa , tm0 u b a0 N0 c0 qN qb

    go : ∀ {m} (ψ : Formula K m) (c0 : S) → fst c0 ≡ key f h ψ
       → ⟨ (c0 ∷ D ∷ δ) ⊨ᴬ shapes (suc A) ⟩
    go {m} (t ∈̇ u) c0 q =
      ∣ inl ( binFormᴬ-wit 0 (bothTm (suc A)) (D ∷ δ)
            c0 (ctS u) (ctS t) (nn m) (ctst u) (ctst t) (num∈γ' m) q
            (bothᴬ t u (ctS u) (ctS t) (nn m) c0 refl refl refl) ) ∣₁
    go {m} (t ≐ u) c0 q =
      ∣ inr ∣ inl ( binFormᴬ-wit 1 (bothTm (suc A)) (D ∷ δ)
            c0 (ctS u) (ctS t) (nn m) (ctst u) (ctst t) (num∈γ' m) q
            (bothᴬ t u (ctS u) (ctS t) (nn m) c0 refl refl refl) ) ∣₁ ∣₁
    go {m} (a ∧̇ b) c0 q =
      ∣ inr ∣ inr ∣ inl ( binFormᴬ-wit 2 ⊤̇ (D ∷ δ)
            c0 (cdS b) (cdS a) (nn m) (cdst b) (cdst a) (num∈γ' m) q tt* ) ∣₁ ∣₁ ∣₁
    go {m} (a ∨̇ b) c0 q =
      ∣ inr ∣ inr ∣ inr ∣ inl ( binFormᴬ-wit 3 ⊤̇ (D ∷ δ)
            c0 (cdS b) (cdS a) (nn m) (cdst b) (cdst a) (num∈γ' m) q tt* ) ∣₁ ∣₁ ∣₁ ∣₁
    go {m} (a ⇒̇ b) c0 q =
      ∣ inr ∣ inr ∣ inr ∣ inr ∣ inl ( binFormᴬ-wit 4 ⊤̇ (D ∷ δ)
            c0 (cdS b) (cdS a) (nn m) (cdst b) (cdst a) (num∈γ' m) q tt* ) ∣₁ ∣₁ ∣₁ ∣₁ ∣₁
    go {m} (¬̇ a) c0 q =
      ∣ inr ∣ inr ∣ inr ∣ inr ∣ inr ∣ inl ( unFormᴬ-wit 5 ⊤̇ (D ∷ δ)
            c0 (cdS a) (nn m) (cdst a) (num∈γ' m) q tt* ) ∣₁ ∣₁ ∣₁ ∣₁ ∣₁ ∣₁
    go {m} ⊤̇ c0 q =
      ∣ inr ∣ inr ∣ inr ∣ inr ∣ inr ∣ inr ∣ inl ( unFormᴬ-wit 6
            (var zero ≐ con (numeralL 0)) (D ∷ δ)
            c0 (nn 0) (nn m) (num∈γ' 0) (num∈γ' m)
            q (sym (numeralL-fst 0)) ) ∣₁ ∣₁ ∣₁ ∣₁ ∣₁ ∣₁ ∣₁
    go {m} ⊥̇ c0 q =
      ∣ inr ∣ inr ∣ inr ∣ inr ∣ inr ∣ inr ∣ inr ∣ inl ( unFormᴬ-wit 7
            (var zero ≐ con (numeralL 0)) (D ∷ δ)

-- ---------------------------------------------------------------------
-- SECTION 16.  THE WALK.  The bounded DefBody reading assembled at the
-- fixed environment delta = Sat A (toS A psi) :: keyS A psi :: z :: env.
-- Conjunct 1 (isCodeAt) and conjunct 3 (DefinesAt) are built.  Conjunct
-- 2 (satGraphAt) is the standing hole: its closedAt half is the slot
-- instance of Section 14, its domAt/appAt halves are Section 11, and
-- what remains open is the bounded twelve-clause walk.
-- ---------------------------------------------------------------------

module Walk (γ : V ℓ) (oγ : IsOrd γ) (clγ : closedω γ) (ω∈γ : ⟨ ω ∈ˢᵥ γ ⟩)
            (A : S) (hA : ⟨ fst A ∈ Lset γ ⟩)
            (Hkey : keyS-in-carrier-lim) (Hsat : Sat-at-asConst) where
  open Bnd γ oγ clγ ω∈γ
  open Rd γ oγ clγ ω∈γ
  open VStages γ oγ clγ ω∈γ A hA
  open LStages γ oγ clγ ω∈γ A hA Hkey Hsat
  open ShW ι ιL γ oγ clγ ω∈γ

  -- the entry reading of a one-entry environment
  envOne' : V ℓ → V ℓ
  envOne' v = env {1} (λ _ → v)

  envOneOut : (v w : V ℓ) → ⟨ w ∈ envOne' v ⟩ → w ≡ pr (# 0) v
  envOneOut v w = PT.rec (setIsSet w (pr (# 0) v))
    (λ { (lift zero , q) → sym q })

  envOneIn : (v w : V ℓ) → w ≡ pr (# 0) v → ⟨ w ∈ envOne' v ⟩
  envOneIn v w q = ∣ lift zero , sym q ∣₁



  private
    sh3 : ∀ {m} → Fin m → Fin (3 + m)
    sh3 i = suc (suc (suc i))

  module ClI = ClOf ι ιL γ oγ clγ ω∈γ
  module ShI = ShW ι ιL γ oγ clγ ω∈γ

  -- CONJUNCT 1a.  keyArityAtL under the bounded reading: the witness is
  -- the code itself, staged by the V-side climb;  the tag equation is
  -- the keyS/codeS identification, definitional at arity one.
  keyArityᴬ : ∀ {n} (z : S) (env : S ^ n) (ψ : Formula ⟪ fst A ⟫ 1)
            → ⟨ (Sat A (mapFo (asConst A) ψ) ∷ keyS A ψ ∷ z ∷ env) ⊨ᴬ keyArityAtL (suc zero) 1 ⟩
  keyArityᴬ z env ψ =
    ∣ codeS A ψ , st→ab (codeS A ψ) (code∈γ ψ)
    , tagAtLᴬ-in (suc (suc zero)) 1 zero (codeS A ψ ∷ (Sat A (mapFo (asConst A) ψ) ∷ keyS A ψ ∷ z ∷ env))
        refl ∣₁

  -- CONJUNCT 1b.  hasWitnessAt under the bounded reading: the witness is
  -- the subformula closure, staged by clo∈γ;  closedAt and shapedAt are
  -- the bounded walks of Sections 14 and 15 at that closure.
  hasWitᴬ : ∀ {n} (w : Fin n) (z : S) (env : S ^ n) (ψ : Formula ⟪ fst A ⟫ 1)
          → fst (lookup w env) ≡ fst A
          → ⟨ (Sat A (mapFo (asConst A) ψ) ∷ keyS A ψ ∷ z ∷ env)
              ⊨ᴬ hasWitnessAt (sh3 w) (suc zero) ⟩
  hasWitᴬ {n} w z env ψ qw =
    ∣ cloS , st→ab cloS (clo∈γ ψ)
    , ( subst (λ u → ⟨ u ∈ fst cloS ⟩) refl (key∈closure ι ιL ψ)
      , ( ClI.closureClosedᴬ ψ δ₀
        , ShI.shapedCloᴬ ψ (sh3 w) δ₀ intoc astg codeTm∈γ code∈γ ) ) ∣₁
    where
    cloS : S
    cloS = clo ι ιL ψ
    δ₀ : S ^ (3 + n)
    δ₀ = Sat A (mapFo (asConst A) ψ) ∷ keyS A ψ ∷ z ∷ env
    intoc : (m : ⟪ fst A ⟫) → ⟨ ι m ∈ fst (lookup (sh3 w) δ₀) ⟩
    intoc m = subst (λ u → ⟨ ι m ∈ u ⟩) (sym qw) (ι∈ m)
    astg : Stg (lookup (sh3 w) δ₀)
    astg = subst (λ (u : S) → ⟨ fst u ∈ˢᵥ Lset γ ⟩)
      (sym (Σ≡Prop (λ v → snd (isL v)) qw)) hA

  -- CONJUNCT 1.  isCodeAt under the bounded reading.
  isCodeAtᴬ : ∀ {n} (w : Fin n) (z : S) (env : S ^ n) (ψ : Formula ⟪ fst A ⟫ 1)
            → fst (lookup w env) ≡ fst A
            → ⟨ (Sat A (mapFo (asConst A) ψ) ∷ keyS A ψ ∷ z ∷ env) ⊨ᴬ isCodeAt (suc zero) (sh3 w) ⟩
  isCodeAtᴬ w z env ψ qw = keyArityᴬ z env ψ , hasWitᴬ w z env ψ qw

-- ---------------------------------------------------------------------

-- (DefinesAt) of remaining-goal, at the fixed environment delta =

module Fill (γ : V ℓ) (oγ : IsOrd γ) (clγ : closedω γ) (ω∈γ : ⟨ ω ∈ˢᵥ γ ⟩)
            (A : S) (hA : ⟨ fst A ∈ Lset γ ⟩)
            (Hkey : keyS-in-carrier-lim) (Hsat : Sat-at-asConst) where
  open Bnd γ oγ clγ ω∈γ
  open Rd γ oγ clγ ω∈γ
  open Γ γ oγ clγ ω∈γ
  open Desc γ oγ
  open LStages γ oγ clγ ω∈γ A hA Hkey Hsat

  private

    K : S

    topRelᶠ {n} B =

        (Δ₀-liftFo (bddPrAt (suc (suc zero)) (suc zero) zero)

  -- 17A.  STAGING HELPERS.
  private
    -- an environment is definitionally the finite set of its pairs, so
    -- FiniteSup's finSetK stages it outright.
    envStg : {k : ℕ} (g : Fin k → V ℓ)
           → ((i : Fin k) → ⟨ g i ∈ˢᵥ Lset γ ⟩)
           → ⟨ env g ∈ˢᵥ Lset γ ⟩
    envStg g hg =
      FS.finSetK _ (λ i → pr (# (toℕ i)) (g i))
        (λ i → pr∈λ (# (toℕ i)) (g i) (num∈γ (toℕ i)) (hg i))

    -- a staged set with the shape pr (# 0) v has v staged
    stgVal₀ : (v : V ℓ) (s : S) → Stg s → fst s ≡ pr (# 0) v
            → ⟨ v ∈ˢᵥ Lset γ ⟩
    stgVal₀ v s hs e = prK (# 0) v (subst (λ w → ⟨ w ∈ˢᵥ Lset γ ⟩) e hs) .snd

    -- a value paired into a staged set is staged
    stgIn : (x : V ℓ) (Z : V ℓ) → ⟨ Z ∈ˢᵥ Lset γ ⟩ → ⟨ x ∈ Z ⟩
          → ⟨ x ∈ˢᵥ Lset γ ⟩
    stgIn x Z hZ hx = Lset-trans′ lem γ hx hZ

  -- the bounded readings of tmValAt's two disjuncts, both directions
  private
    tmValVarᴬ-in : ∀ {m} (t e v : Fin m) (δ' : S ^ m) (i : ℕ)
                 → fst (lookup t δ') ≡ pr (# 1) (# i)
                 → ⟨ pr (# i) (fst (lookup v δ')) ∈ fst (lookup e δ') ⟩
                 → ⟨ δ' ⊨ᴬ tmValAt t e v ⟩
    tmValVarᴬ-in t e v δ' i qc hp =
      ∣ inl (∣ ( w , st→ab w (numStg i)
        , ( tagAtLᴬ-in (suc t) 1 zero (w ∷ δ')
              (qc ∙ cong (pr (# 1)) (sym (numeralL-fst i)))
          , appAtᴬ-in (suc e) zero (suc v) (w ∷ δ')
              (subst (λ u → ⟨ pr u (fst (lookup v δ')) ∈ fst (lookup e δ') ⟩)
                (sym (numeralL-fst i)) hp) ) ) ∣₁) ∣₁
      where
      w : S
      w = numeralL i

    tmValConᴬ-in : ∀ {m} (t e v : Fin m) (δ' : S ^ m) (c₀ : V ℓ)
                 → fst (lookup t δ') ≡ pr (# 0) c₀
                 → fst (lookup v δ') ≡ c₀
                 → ⟨ δ' ⊨ᴬ tmValAt t e v ⟩
    tmValConᴬ-in t e v δ' c₀ qc qv =
      ∣ inr (tagAtLᴬ-in t 0 v δ' (qc ∙ cong (pr (# 0)) (sym qv))) ∣₁

    tmValVarOfᴬ : ∀ {m} (t e v : Fin m) (δ' : S ^ m) (i : ℕ)
                → fst (lookup t δ') ≡ pr (# 1) (# i)
                → ⟨ δ' ⊨ᴬ tmValAt t e v ⟩
                → ⟨ pr (# i) (fst (lookup v δ')) ∈ fst (lookup e δ') ⟩
    tmValVarOfᴬ t e v δ' i qc = PT.rec targetP branch
      where
      target : Type (ℓ-suc ℓ)
      target = ⟨ pr (# i) (fst (lookup v δ')) ∈ fst (lookup e δ') ⟩
      targetP : isProp target
      targetP = snd (pr (# i) (fst (lookup v δ')) ∈ fst (lookup e δ'))
      branch : (⟨ δ' ⊨ᴬ (∃̇ (tagAtL (suc t) 1 zero ∧̇ appAt (suc e) zero (suc v))) ⟩
                ⊎ ⟨ δ' ⊨ᴬ tagAtL t 0 v ⟩) → target
      stepIn : (Σ[ w ∈ S ] (⟨ w ∈ˢ Abnd ⟩
                 × ⟨ (w ∷ δ') ⊨ᴬ (tagAtL (suc t) 1 zero
                     ∧̇ appAt (suc e) zero (suc v)) ⟩))
             → target
      stepIn (w , hw , htag , happ) =
        subst (λ u → ⟨ pr u (fst (lookup v δ')) ∈ fst (lookup e δ') ⟩)
          (sym (pr-inj (sym qc ∙ tagAtLᴬ-out (suc t) 1 zero (w ∷ δ') htag) .snd))
          (appAtᴬ-out (suc e) zero (suc v) (w ∷ δ') happ)

      branch (inl wit) = PT.rec targetP stepIn wit
      branch (inr htag') =
        Empty.rec (snotz (#-inj 1 0
          (pr-inj (sym qc ∙ tagAtLᴬ-out t 0 v δ' htag') .fst)))
    tmValConOfᴬ : ∀ {m} (t e v : Fin m) (δ' : S ^ m) (c₀ : V ℓ)
                → fst (lookup t δ') ≡ pr (# 0) c₀
                → ⟨ δ' ⊨ᴬ tmValAt t e v ⟩
                → fst (lookup v δ') ≡ c₀
    tmValConOfᴬ t e v δ' c₀ qc = PT.rec (setIsSet (fst (lookup v δ')) c₀) branch
      where
      branch : (⟨ δ' ⊨ᴬ (∃̇ (tagAtL (suc t) 1 zero ∧̇ appAt (suc e) zero (suc v))) ⟩
                ⊎ ⟨ δ' ⊨ᴬ tagAtL t 0 v ⟩) → fst (lookup v δ') ≡ c₀
      stepIn : (Σ[ w ∈ S ] (⟨ w ∈ˢ Abnd ⟩
                 × ⟨ (w ∷ δ') ⊨ᴬ (tagAtL (suc t) 1 zero
                     ∧̇ appAt (suc e) zero (suc v)) ⟩))
             → fst (lookup v δ') ≡ c₀
      stepIn (w , hw , htag , happ) =
        Empty.rec (znots (#-inj 0 1
          (pr-inj (sym qc ∙ tagAtLᴬ-out (suc t) 1 zero (w ∷ δ') htag) .fst)))
      branch (inl wit) = PT.rec (setIsSet (fst (lookup v δ')) c₀) stepIn wit
      branch (inr htag) =
        sym (pr-inj (sym qc ∙ tagAtLᴬ-out t 0 v δ' htag) .snd)
  -- its two readings are joined by the Delta0 transport on the certificate
  -- Δ₀-pairsIn.
  private
    svAtᴬ-out : ∀ {m} (f : Fin m) (δ' : S ^ m)
              → ⟨ δ' ⊨ᴬ svAt f ⟩
              → (x y y' : S) → ⟨ x ∈ˢ Abnd ⟩ → ⟨ y ∈ˢ Abnd ⟩ → ⟨ y' ∈ˢ Abnd ⟩
              → ⟨ pr (fst x) (fst y) ∈ fst (lookup f δ') ⟩
              → ⟨ pr (fst x) (fst y') ∈ fst (lookup f δ') ⟩
              → fst y ≡ fst y'
    svAtᴬ-out f δ' h x y y' hx hy hy' p q =
      h x hx y hy y' hy'
        (appAtᴬ-in (suc (suc (suc f))) (suc (suc zero)) (suc zero)
           (y' ∷ y ∷ x ∷ δ') p)
        (appAtᴬ-in (suc (suc (suc f))) (suc (suc zero)) zero
           (y' ∷ y ∷ x ∷ δ') q)

    domAtᴬ-in : ∀ {m} (f d : Fin m) (δ' : S ^ m)
              → ⟨ δ' ⊨ᴬ domAt f d ⟩
              → (x : S) → ⟨ x ∈ˢ Abnd ⟩
              → ⟨ fst x ∈ fst (lookup d δ') ⟩
              → ∥ (Σ[ y ∈ S ] (⟨ y ∈ˢ Abnd ⟩
                   × ⟨ pr (fst x) (fst y) ∈ fst (lookup f δ') ⟩)) ∥₁
    domAtᴬ-in f d δ' h x hx m' = PT.rec squash₁
      (λ { (y , hy , hp) →
        ∣ y , hy
        , appAtᴬ-out (suc (suc f)) (suc zero) zero (y ∷ x ∷ δ') hp ∣₁ })
      (h x hx .snd m')

    domAtᴬ-in′ : ∀ {m} (f d : Fin m) (δ' : S ^ m)
               → ⟨ δ' ⊨ᴬ domAt f d ⟩
               → (x : S) → ⟨ x ∈ˢ Abnd ⟩
               → ⟨ fst x ∈ fst (lookup d δ') ⟩
               → ⟨ (x ∷ δ') ⊨ᴬ inDomAt (suc f) zero ⟩
    domAtᴬ-in′ f d δ' h x hx m' = h x hx .snd m'

    valuesInAtᴬ-out : ∀ {m} (f b : Fin m) (δ' : S ^ m)
                    → ⟨ δ' ⊨ᴬ valuesInAt f b ⟩
                    → (x y : S) → ⟨ x ∈ˢ Abnd ⟩ → ⟨ y ∈ˢ Abnd ⟩
                    → ⟨ pr (fst x) (fst y) ∈ fst (lookup f δ') ⟩
                    → ⟨ fst y ∈ fst (lookup b δ') ⟩
    valuesInAtᴬ-out f b δ' h x y hx hy p =
      h x hx y hy (appAtᴬ-in (suc (suc f)) (suc zero) zero (y ∷ x ∷ δ') p)

    ovᴬ-in : ∀ {m} (e d b : Fin m) (δ' : S ^ m) (stE : Stg (lookup e δ'))
           → ⟨ δ' ⊨ envOverAt e d b ⟩ → ⟨ δ' ⊨ᴬ envOverAt e d b ⟩
    ovᴬ-in e d b δ' stE h =
      ( λ x hx y hy y' hy' pA qA →
          svAt-out e δ' (h .fst) x y y'
            (appAtᴬ-out (suc (suc (suc e))) (suc (suc zero)) (suc zero)
               (y' ∷ y ∷ x ∷ δ') pA)
            (appAtᴬ-out (suc (suc (suc e))) (suc (suc zero)) zero
               (y' ∷ y ∷ x ∷ δ') qA) )
      , ( λ x hx →
            ( λ hIn → PT.rec (snd (fst x ∈ fst (lookup d δ')))
                (λ { (y , (hy , hp)) →
                  domAt-out e d δ' (h .snd .fst) x y
                    (appAtᴬ-out (suc (suc e)) (suc zero) zero (y ∷ x ∷ δ') hp) })
                hIn )
          , λ m' → PT.rec squash₁
                (λ { (y , hp) →
                  ∣ y , st→ab y
                      (prK (fst x) (fst y)
                        (trv (pr (fst x) (fst y)) (lookup e δ') stE hp) .snd)
                  , appAtᴬ-in (suc (suc e)) (suc zero) zero (y ∷ x ∷ δ') hp
                    ∣₁ })
                (domAt-in e d δ' (h .snd .fst) x m') )
      , ( λ x hx y hy pA →
          valuesInAt-out e b δ' (h .snd .snd .fst) x y
            (appAtᴬ-out (suc (suc e)) (suc zero) zero (y ∷ x ∷ δ') pA) )
      , D0B.d0r (pairsInAt e d b) (Δ₀-pairsIn e d b) δ'
             (h .snd .snd .snd)

  -- 17B.  THE AMBIENT MOVES.  Sound's Ambient, replayed under the
  -- bounded reading:  the recovery consumes the bounded envOverAt at a
  -- staged environment, whose svAt / domAt / valuesInAt instances are all
  -- at staged sets, and whose pairsInAt is the Delta0-transported standard
  -- one.
  module Ambientᴬ (B : S) (stB : ⟨ fst B ∈ˢᵥ Lset γ ⟩)
                  {k : ℕ} (δg : S ^ k) (Ei di bi : Fin k) (m : ℕ)
                  (qd : fst (lookup di δg) ≡ # m)
                  (qb : fst (lookup bi δg) ≡ fst B)
                  (hEᴬ : ⟨ δg ⊨ᴬ envSetAt Ei di bi ⟩)
                  (stE : Stg (lookup Ei δg))
                  where
    nnS : ℕ → S
    nnS j = # j , numL j

    nnStg : (j : ℕ) → Stg (nnS j)
    nnStg j = num∈γ' j

    fwdᴬ : (z : S) → ⟨ z ∈ˢ Abnd ⟩
         → ⟨ fst z ∈ fst (lookup Ei δg) ⟩
         → ⟨ (z ∷ δg) ⊨ᴬ envOverAt zero (suc di) (suc bi) ⟩
    fwdᴬ z hz hZ = extAtᴬ-fwd Ei (envOverAt zero (suc di) (suc bi)) δg hEᴬ z hz hZ

    bwdᴬ : (z : S) → ⟨ z ∈ˢ Abnd ⟩
         → ⟨ (z ∷ δg) ⊨ᴬ envOverAt zero (suc di) (suc bi) ⟩
         → ⟨ fst z ∈ fst (lookup Ei δg) ⟩
    bwdᴬ z hz hov = extAtᴬ-bwd Ei (envOverAt zero (suc di) (suc bi)) δg hEᴬ z hz hov

    -- the recovery, at a staged environment
    recᴬ : (z : S) → ⟨ z ∈ˢ Abnd ⟩
         → ⟨ fst z ∈ fst (lookup Ei δg) ⟩
         → Σ[ g ∈ Ix B m ] (fst z ≡ fst (envS B g))
    recᴬ z hz hZ = g , recovers
      where
      ov : ⟨ (z ∷ δg) ⊨ᴬ envOverAt zero (suc di) (suc bi) ⟩
      ov = fwdᴬ z hz hZ

      svᴬ : ⟨ (z ∷ δg) ⊨ᴬ svAt zero ⟩
      svᴬ = ov .fst
      domᴬ : ⟨ (z ∷ δg) ⊨ᴬ domAt zero (suc di) ⟩
      domᴬ = ov .snd .fst
      valsᴬ : ⟨ (z ∷ δg) ⊨ᴬ valuesInAt zero (suc bi) ⟩
      valsᴬ = ov .snd .snd .fst
      prsᴬ : ⟨ (z ∷ δg) ⊨ᴬ pairsInAt zero (suc di) (suc bi) ⟩
      prsᴬ = ov .snd .snd .snd

      prs : ⟨ (z ∷ δg) ⊨ pairsInAt zero (suc di) (suc bi) ⟩
      prs = D0B.d0r (pairsInAt zero (suc di) (suc bi))
              (Δ₀-pairsIn zero (suc di) (suc bi)) (z ∷ δg) prsᴬ

      Entry : Fin m → Type (ℓ-suc ℓ)
      Entry i = Σ[ y ∈ S ] (⟨ y ∈ˢ Abnd ⟩
                 × ⟨ pr (# (toℕ i)) (fst y) ∈ fst z ⟩)

      isPropEntry : (i : Fin m) → isProp (Entry i)
      isPropEntry i (y , hy , p) (y' , hy' , p') =
        Σ≡Prop (λ w → isProp× (snd (w ∈ˢ Abnd))
                        (snd (pr (# (toℕ i)) (fst w) ∈ fst z)))
          (Σ≡Prop (λ v → snd (isL v))
            (svAtᴬ-out zero (z ∷ δg) svᴬ (nnS (toℕ i)) y y'
              (st→ab (nnS (toℕ i)) (num∈γ' (toℕ i))) hy hy'
              p p'))


      entry : (i : Fin m) → Entry i
      entry i = PT.rec (isPropEntry i) (λ w → w)
        (domAtᴬ-in zero (suc di) (z ∷ δg) domᴬ (nnS (toℕ i))
          (st→ab (nnS (toℕ i)) (num∈γ' (toℕ i)))
          (subst (λ w → ⟨ # (toℕ i) ∈ w ⟩) (sym qd)
            (#mono (toℕ i) m (toℕ<n i))))

      entry∈' : (i : Fin m)
              → ⟨ pr (fst (nnS (toℕ i))) (fst (entry i .fst)) ∈ fst z ⟩
      entry∈' i = entry i .snd .snd

      fib : (i : Fin m) → Σ[ mm ∈ ⟪ fst B ⟫ ] (⟪ fst B ⟫↪ mm ≡ fst (entry i .fst))
      fib i = ∈-asFiber {a = fst (entry i .fst)} {b = fst B}
        (subst (λ w → ⟨ fst (entry i .fst) ∈ w ⟩) qb
          (valuesInAtᴬ-out zero (suc bi) (z ∷ δg) valsᴬ (nnS (toℕ i))
            (entry i .fst) (st→ab (nnS (toℕ i)) (num∈γ' (toℕ i)))
            (entry i .snd .fst) (entry∈' i)))

      g : Ix B m
      g i = fib i .fst

      val≡ : (i : Fin m) → ⟪ fst B ⟫↪ (g i) ≡ fst (entry i .fst)
      val≡ i = fib i .snd

      fwd : (w : V ℓ) → ⟨ w ∈ fst (envS B g) ⟩ → ⟨ w ∈ fst z ⟩
      fwd w hw = PT.rec (snd (w ∈ fst z))
        (λ { (li , q) → subst (λ zz → ⟨ zz ∈ fst z ⟩)
               (cong (pr (# (toℕ (lower li))))
                 (sym (val≡ (lower li))) ∙ q)
               (entry (lower li) .snd .snd) })
        hw

      bwd : (w : V ℓ) → ⟨ w ∈ fst z ⟩ → ⟨ w ∈ fst (envS B g) ⟩
      bwd w hw = PT.rec squash₁
        (λ { (u , (v , (u∈ , (v∈ , eq)))) →
          PT.rec squash₁
          (λ { (jm , (m<n , um)) →
            let i = fromℕ' m jm m<n
                iu : # (toℕ i) ≡ fst u
                iu = cong #_ (toFromId' m jm m<n) ∙ sym um
                hv : ⟨ pr (# (toℕ i)) (fst v) ∈ fst z ⟩
                hv = subst (λ zz → ⟨ zz ∈ fst z ⟩)
                       (eq ∙ cong (λ zz → pr zz (fst v)) (sym iu)) hw
                hv' : ⟨ pr (fst (nnS (toℕ i))) (fst v) ∈ fst z ⟩
                hv' = hv
                same : fst v ≡ fst (entry i .fst)
                same = svAtᴬ-out zero (z ∷ δg) svᴬ (nnS (toℕ i)) v
                         (entry i .fst)
                         (st→ab (nnS (toℕ i)) (num∈γ' (toℕ i)))
                         (st→ab v (stgIn (fst v) (fst B) stB
                           (subst (λ w → ⟨ fst v ∈ w ⟩) qb v∈)))
                         (entry i .snd .fst) hv' (entry∈' i)
            in ∣ lift i , cong (pr (# (toℕ i))) (val≡ i ∙ sym same)
                        ∙ cong (λ zz → pr zz (fst v)) iu ∙ sym eq ∣₁ })
          (∈#-elim m (fst u) (subst (λ zz → ⟨ fst u ∈ zz ⟩) qd u∈)) })
        (pairsIn-out zero (suc di) (suc bi) (z ∷ δg) prs
          (w , isL-trans {x = fst z} {y = w} hw (snd z)) hw)

      recovers : fst z ≡ fst (envS B g)
      recovers = extensionalV (λ w → ⇔toPath (bwd w) (fwd w))

    intoᴬ : (z : S) → ⟨ z ∈ˢ Abnd ⟩
          → ⟨ fst z ∈ fst (lookup Ei δg) ⟩
          → ⟨ fst z ∈ fst (envSet B m) ⟩
    intoᴬ z hz hZ =
      subst (λ w → ⟨ w ∈ fst (envSet B m) ⟩)
        (sym (recᴬ z hz hZ .snd)) (envSet-in B (recᴬ z hz hZ .fst))

    outofᴬ : (z : S) → ⟨ z ∈ˢ Abnd ⟩
           → ⟨ fst z ∈ fst (envSet B m) ⟩
           → ⟨ fst z ∈ fst (lookup Ei δg) ⟩
    outofᴬ z hz hz∈ = PT.rec (snd (fst z ∈ fst (lookup Ei δg)))
      (λ { (g , eg) →
        bwdᴬ z hz
          (ovᴬ-in zero (suc di) (suc bi) (z ∷ δg) hz
            (envOverAt-transport (B ∷ nnS m ∷ envS B g ∷ []) (z ∷ δg)
              (suc (suc zero)) (suc zero) zero zero (suc di) (suc bi)
              (sym eg) (sym qd) (sym qb)
              (envOver B g))) })
      (envSet-out B m z hz∈)

    asEnvᴬ : (z : S) → ⟨ z ∈ˢ Abnd ⟩
           → ⟨ fst z ∈ fst (lookup Ei δg) ⟩
           → Σ[ g ∈ Ix B m ] (fst z ≡ fst (envS B g))
    asEnvᴬ = recᴬ

  -- 17C.  THE TWELVE, under the bounded reading, at the graph environment
  -- gEnv = A :: satTable A phi :: slot A phi :: deltaG.  The five clause
  -- families replay Sound's assemblies:  the meta inversions are Sound's
  -- own frames, the Sat-cond moves are unchanged, and every unbounded
  -- existential is discharged staged (descended from the staged carrier,
  -- table, slot and clause binders).
  module Twelve (psi : Formula ⟪ fst A ⟫ 1) (deltaG : S ^ 3) where
    phi : Formula S 1
    phi = toS A psi
    gEnv : S ^ 6
    gEnv = A ∷ satTable A phi ∷ slot A phi ∷ deltaG

    Ci Ti Bi : Fin 6
    Ci = suc (suc zero)
    Ti = suc zero
    Bi = zero

    stC : Stg (slot A phi)
    stC = slot∈γ psi
    stT : Stg (satTable A phi)
    stT = satTable∈γ psi

    c∈g : (c : S) → ⟨ fst c ∈ fst (lookup Ci gEnv) ⟩ → Stg c
    c∈g c h∈ = trv (fst c) (slot A phi) stC h∈
