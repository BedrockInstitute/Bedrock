{-# OPTIONS --cubical --safe --guardedness #-}

-- [LJ-1.685] W3 slice.  Does renameFo commute with LsetGraphAt by refl?
-- `.agda.txt` so a red run is not a verification target.

open import Base.Prelude
open import Base.Classical using ( LEM )

module LJ-1-685.runs.RENAME {ℓ : Level} (lem : LEM (ℓ-suc ℓ)) where

open import FOL.Syntax using ( Formula )
open import FOL.Manipulation.Renaming using ( renameFo )
open import L.Coding.Sequence {ℓ} lem using ( LsetGraphAt )

rename-graph : {n m : ℕ} (ρ : Fin n → Fin m) (w b : Fin n)
             → renameFo ρ (LsetGraphAt w b) ≡ LsetGraphAt (ρ w) (ρ b)
rename-graph ρ w b = refl
