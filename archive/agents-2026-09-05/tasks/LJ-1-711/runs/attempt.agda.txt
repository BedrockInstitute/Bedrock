IsLimit : S → Type (ℓ-suc ℓ)
IsLimit α =
    IsOrd α
  × ((x : S) → ⟨ x ∈ₛ α ⟩ → ⟨ sucV x ∈ₛ α ⟩)
  × ((X : Type ℓ) (h : X → S)
      → ((i : X) → ⟨ h i ∈ₛ α ⟩)
      → ⟨ ⋃ (sett X h) ∈ₛ α ⟩)

succFam : S → S → Lift {ℓ-zero} {ℓ} Bool → S
succFam σ₁ σ₂ (lift true)  = sucV σ₁
succFam σ₁ σ₂ (lift false) = sucV σ₂

-- THE ATTEMPT.  Close the merge membership through the limit's own
-- union clause instanced at the written-out family, exactly the route
-- [LJ-1.710] section 3 proved green for its own-named goal set.  The
-- difference: the goal SET here states fst (bound2 ...), which is the
-- form the placement row consumes.
attempt :
    (α : S) (Lim : IsLimit α)
    (σ₁ σ₂ : S) (o₁ : IsOrd σ₁) (o₂ : IsOrd σ₂)
    (h₁ : ⟨ σ₁ ∈ₛ α ⟩) (h₂ : ⟨ σ₂ ∈ₛ α ⟩)
  → ⟨ fst (bound2 σ₁ σ₂ o₁ o₂) ∈ₛ α ⟩
attempt α (_ , succCl , unionCl) σ₁ σ₂ o₁ o₂ h₁ h₂ =
  unionCl (Lift {ℓ-zero} {ℓ} Bool) (succFam σ₁ σ₂) mm
  where
  mm : (b : Lift {ℓ-zero} {ℓ} Bool) → ⟨ succFam σ₁ σ₂ b ∈ₛ α ⟩
  mm (lift true)  = succCl σ₁ h₁
  mm (lift false) = succCl σ₂ h₂
