-- =====================================================================
-- W3.  THE THREE EXTRA ASKS, INHABITED TOGETHER AT `KValue`'s FRAME.
--
-- The three fields ask for three DIFFERENT things and the brief allows
-- that.  Read at first hand:
--

--       honest form src/L/Coding/EnvSupply.lagda.md:712-725
--       EXTRA: {k : ℕ} (g : Fin k → V ℓ), fst z ≡ env g,
--              ⟨ fst z ∈ fst K ⟩, ⟨ fst x ∈ fst K ⟩
--   `consK-allin`   :332-335, honest form :727-740, the SAME extra at a
--       one-cell-deeper environment

--       EXTRA: ∥ Σ[ n ∈ ℕ ] (fst ar ≡ # n) ∥₁
--
-- SO W3 IS ONE VALUE CARRYING ALL THREE AT ONE INSTANTIATION, and the
-- instantiation is CONCRETE, not a free cell: `x = ar = numeralL 0` and
-- `z` the environment of the EMPTY family.
-- =====================================================================

module W3
  (lam : V ℓ) (ordλ : IsOrd lam)
  (succλ : (d : V ℓ) → ⟨ d ∈ lam ⟩ → ⟨ sucV d ∈ lam ⟩)
  (∅∈λ : ⟨ ∅ ∈ lam ⟩)
  (gam : V ℓ) (ordγ : IsOrd gam) (γ∈λ : ⟨ gam ∈ lam ⟩)
  (ω∈σ : ⟨ ω ∈ sucV gam ⟩) where

  module KV = KValue lam ordλ succλ ∅∈λ gam ordγ γ∈λ
  module SM = SupplyMerge lam ordλ succλ ∅∈λ

  -- The empty family, and the environment over it.
  g₀ : Fin 0 → V ℓ
  g₀ ()

  -- `env {0} g₀` IS `finSet 0 (λ i → pr (# (toℕ i)) (g₀ i))`, character
  -- for character: `env` at src/L/Coding/Environment.lagda.md:84-86 and
  -- `finSet` at src/L/Axioms/Basic.lagda.md:285-286 are the same `sett`.
  env₀∈ : ⟨ env {0} g₀ ∈ Lset lam ⟩
  env₀∈ = SM.finSetK 0 (λ i → pr (# (toℕ i)) (g₀ i)) (λ ())

  z₀ : S
  z₀ = env {0} g₀
     , isL-trans {x = Lset lam} {y = env {0} g₀} env₀∈ (isL-Lset lam ordλ)

  x₀ : S
  x₀ = numeralL 0

  x₀∈ : ⟨ fst x₀ ∈ Lset lam ⟩
  x₀∈ = KV.facts .numK0

  -- THE THREE EXTRAS, AT ONE FRAME, IN ONE VALUE.
  --   (1) and (2) are the consK pair's extra: an environment WITNESS for
  --       `z`, and `z` and `x` in `K`.
  --   (3) is `someEnv`'s extra: the truncated numeral arity, at the same
  --       `ar = x₀`.
  extras-inhabited :
      (Σ[ k ∈ ℕ ] Σ[ g ∈ (Fin k → V ℓ) ] (fst z₀ ≡ env g))
    × (⟨ fst z₀ ∈ Lset lam ⟩ × ⟨ fst x₀ ∈ Lset lam ⟩)
    × ∥ Σ[ m ∈ ℕ ] (fst x₀ ≡ # m) ∥₁
  extras-inhabited =
      (0 , (g₀ , refl))
    , (env₀∈ , x₀∈)
    , ∣ 0 , numeralL-fst 0 ∣₁

extras-inhabited = W3.extras-inhabited

-- =====================================================================
-- `statement-matches`: Agda's word that the three types above ARE the
-- master's own, weakened by the marked extras and by NOTHING else.
-- Every field is read off a `TFacts` value by projection, with no
-- coercion, no `subst` and no re-association; the extras are DROPPED on
-- the left of each `→`, which is what "weakened by exactly these" means.
-- It certifies the STATEMENT only; THE OBLIGATION NEVER CALLS IT, and no
-- `TFacts` field is used to prove a `TFacts` field.  This is
-- `[LJ-1.545]`'s device (agents/tasks/LJ-1-553/Probe553.agda:305-315
-- uses it at nine fields).
-- =====================================================================

statement-matches : {n : ℕ}
    (N0 N1 N2 N3 N4 N5 N6 N7 N8 N9 N10 N11 t0 t1 K : Fin (5 + n))
    (γ' : S ^ (11 + n))
  → TFacts N0 N1 N2 N3 N4 N5 N6 N7 N8 N9 N10 N11 t0 t1 K γ'
  → LastThree {n} K γ'
statement-matches N0 N1 N2 N3 N4 N5 N6 N7 N8 N9 N10 N11 t0 t1 K γ' tf = record
  { someEnv+ = λ ya yc b a ar c _ → TFacts.someEnv tf ya yc b a ar c
  ; consK-forall+ = λ ya yc a ar c E z x e' g _ _ _ →
      TFacts.consK-forall tf ya yc a ar c E z x e'
  ; consK-allin+ = λ E ya yc b a ar c z w x e' g _ _ _ →
      TFacts.consK-allin tf E ya yc b a ar c z w x e' }

-- The K slot at the record's own index, named once.
Kat : {n : ℕ} → Fin (5 + n) → S ^ (11 + n) → S
Kat K γ' = lookup (suc (suc (suc (suc (suc (suc K)))))) γ'

module Discharge {n : ℕ}
  (K : Fin (5 + n))
  (γ' : S ^ (11 + n))
  (Ktr : isTransV (fst (Kat K γ')))
  (numK0 : ⟨ # 0 ∈ fst (Kat K γ') ⟩)
  (sucK : (a : V ℓ) → ⟨ a ∈ fst (Kat K γ') ⟩ → ⟨ sucV a ∈ fst (Kat K γ') ⟩)
  (pairK : (a b : V ℓ) → ⟨ a ∈ fst (Kat K γ') ⟩ → ⟨ b ∈ fst (Kat K γ') ⟩
         → ⟨ pr a b ∈ fst (Kat K γ') ⟩)
  (finSetK : (m : ℕ) (h : Fin m → V ℓ)
           → ((i : Fin m) → ⟨ h i ∈ fst (Kat K γ') ⟩)
           → ⟨ finSet m h ∈ fst (Kat K γ') ⟩)
  -- The env set at a NUMERAL arity, over the carrier pinned at cell 0.
  -- `[LJ-1.173]` restricted this to a numeral because the general form
  -- is FALSE at the bound (src/L/Condensation/TwelveAgree.lagda.md
  -- :297-308, citing `[LJ-1.172]`).
  (envSetGenK : (ar : S) → ∥ Σ[ p ∈ ℕ ] (fst ar ≡ # p) ∥₁
              → ⟨ fst ar ∈ fst (Kat K γ') ⟩
              → ⟨ fst (Generic.envSetGen (lookup zero γ') ar)
                   ∈ fst (Kat K γ') ⟩)
  (envInKp : {m : ℕ} (γ₁ : Vec S m) (di bi : Fin m)
           → lookup bi γ₁ ≡ lookup zero γ'
           → ∥ Σ[ p ∈ ℕ ] (fst (lookup di γ₁) ≡ # p) ∥₁
           → ⟨ fst (lookup di γ₁) ∈ fst (Kat K γ') ⟩
           → (z : S) → ⟨ (z ∷ γ₁) ⊨ envOverAt zero (suc di) (suc bi) ⟩
           → ⟨ fst z ∈ fst (Kat K γ') ⟩)
  where

  module F = Fact (Kat K γ')
  module EC = F.EnvClosure Ktr numK0 sucK pairK finSetK

  -- ---------------------------------------------------------------
  -- THE SLOT-GENERIC consK SHELL.  `src/` HAS NO SUCH SHELL AND THAT
  -- IS THE FINDING.  `Fact.subK-gen` (src/L/Coding/EnvSupply.lagda.md
  -- :481) and `Fact.tmValK` (:575) are slot-generic, but the three
  -- consK honest forms are PINNED at `(γ' : Vec S 2)` (:631, :646,
  -- :661, and again at :712, :727, :742), so at the record's frame,
  -- where the tail is `S ^ (11 + n)`, they cannot be applied at all.
  -- The body below is `ConsKClosed.consK-forall`'s own body
  -- (src/L/Coding/EnvSupply.lagda.md:719-725) with the vector length
  -- and the three slots made variable; nothing else changed.
  -- ---------------------------------------------------------------
  consK-gen : {m : ℕ} (γ₁ : Vec S m) (ei xi zi : Fin m)
            → {k : ℕ} (g : Fin k → V ℓ) → fst (lookup zi γ₁) ≡ env g
            → ⟨ fst (lookup zi γ₁) ∈ fst (Kat K γ') ⟩
            → ⟨ fst (lookup xi γ₁) ∈ fst (Kat K γ') ⟩
            → ⟨ γ₁ ⊨ consAtL ei xi zi ⟩
            → ⟨ fst (lookup ei γ₁) ∈ fst (Kat K γ') ⟩
  consK-gen γ₁ ei xi zi g zg zK xK h =
    subst (λ w → ⟨ w ∈ fst (Kat K γ') ⟩) (sym e'eq)
      (EC.envConsK g (fst (lookup xi γ₁))
        (subst (λ w → ⟨ w ∈ fst (Kat K γ') ⟩) zg zK) xK)
    where
    e'eq : fst (lookup ei γ₁) ≡ env (cons (fst (lookup xi γ₁)) g)
    e'eq = subst ⟨_⟩ (consAtL-adequate ei xi zi γ₁ g zg) h

  -- ---------------------------------------------------------------
  -- THE SLOT-GENERIC someEnv SHELL.  `src/` HAS NO SUCH SHELL EITHER:
  -- `SupplyEnv.someEnv` (src/L/Coding/EnvSupply.lagda.md:417-445) is
  -- pinned at a FOUR-cell environment `E ∷ ar ∷ B₀ ∷ level ∷ []`
  -- (:431), while `someEnvDef` asks for satisfaction at
  -- `E ∷ ya ∷ yc ∷ b ∷ a ∷ ar ∷ c ∷ γ'`
  -- (src/L/Condensation/LowerAgree.lagda.md:58).  The two are the SAME
  -- formula constructor at different slots, and the shell below is the
  -- tree's own pair `EnvSet.back` (src/L/Condensation.lagda.md:3045)
  -- after `Generic.Holds.holds` (src/L/Coding/EnvSet.lagda.md:540),
  -- both already generic in the vector length.
  -- ---------------------------------------------------------------
  someEnv-gen : {m : ℕ} (γ₁ : Vec S m) (Ei di bi Ki : Fin m)
    → (EK : ⟨ fst (lookup Ei γ₁) ∈ fst (lookup Ki γ₁) ⟩)
    → (arK : ⟨ fst (lookup di γ₁) ∈ fst (lookup Ki γ₁) ⟩)
    → (arityK : (N v : S) → ⟨ fst v ∈ fst N ⟩
              → ⟨ fst N ∈ fst (lookup Ki γ₁) ⟩
              → ⟨ fst v ∈ fst (lookup Ki γ₁) ⟩)
    → (envInK : (z : S) → ⟨ (z ∷ γ₁) ⊨ envOverAt zero (suc di) (suc bi) ⟩
              → ⟨ fst z ∈ fst (lookup Ki γ₁) ⟩)
    → (qE : fst (lookup Ei γ₁)
          ≡ fst (Generic.envSetGen (lookup bi γ₁) (lookup di γ₁)))
    → ⟨ γ₁ ⊨ envSetB Ei di bi Ki ⟩
  someEnv-gen γ₁ Ei di bi Ki EK arK arityK envInK qE = ES.back H.holds
    where
    module G = Generic (lookup bi γ₁) (lookup di γ₁)
    module ES = EnvSet Ei di bi Ki γ₁ arityK EK arK envInK
    module H = G.Holds γ₁ Ei di bi qE refl refl

  -- ---------------------------------------------------------------
  -- THE THREE, COLLECTED.
  -- ---------------------------------------------------------------
  last-three : LastThree {n} K γ'
  last-three = record
    { someEnv+ = λ ya yc b a ar c arNum yaK ycK arK →
        Generic.envSetGen (lookup zero γ') ar
        , ( envSetGenK ar arNum arK
          , someEnv-gen
              (Generic.envSetGen (lookup zero γ') ar
                 ∷ ya ∷ yc ∷ b ∷ a ∷ ar ∷ c ∷ γ')
              zero
              (suc (suc (suc (suc (suc zero)))))
              (suc (suc (suc (suc (suc (suc (suc zero)))))))
              (suc (suc (suc (suc (suc (suc (suc (suc (suc (suc (suc (suc (suc K)))))))))))))
              (envSetGenK ar arNum arK) arK (λ N v → Ktr)
              (envInKp
                (Generic.envSetGen (lookup zero γ') ar
                   ∷ ya ∷ yc ∷ b ∷ a ∷ ar ∷ c ∷ γ')
                (suc (suc (suc (suc (suc zero)))))
                (suc (suc (suc (suc (suc (suc (suc zero)))))))
                refl arNum arK)
              refl )
    ; consK-forall+ = λ ya yc a ar c E z x e' g zg zK xK h →
        consK-gen (e' ∷ x ∷ z ∷ E ∷ ya ∷ yc ∷ a ∷ ar ∷ c ∷ γ')

    ; consK-allin+ = λ E ya yc b a ar c z w x e' g zg zK xK h →
        consK-gen (e' ∷ x ∷ w ∷ z ∷ E ∷ ya ∷ yc ∷ b ∷ a ∷ ar ∷ c ∷ γ')
          zero (suc zero) (suc (suc (suc zero))) g zg zK xK h }
