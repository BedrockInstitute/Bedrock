module Spend (lam : S) (ordλ : IsOrd lam)
  (succλ : (d : S) → ⟨ d ∈ˢ lam ⟩ → ⟨ sucV d ∈ˢ lam ⟩)
  (X : S) (X⊆Lλ : (z : S) → ⟨ z ∈ˢ X ⟩ → ⟨ z ∈ˢ Lset lam ⟩)
  (∅∈λ : ⟨ ∅ ∈ˢ lam ⟩)
  (elem : P652.Frame652.A.Elementary lam ordλ succλ X X⊆Lλ ∅∈λ) where

  module A = P679.At lam ordλ succλ X X⊆Lλ ∅∈λ elem
  open A using ( Completeness; HierInStage; BoundInStage; inBound; Code; val )
  module F = P652.Frame652 lam ordλ succλ X X⊆Lλ ∅∈λ
  module Cy = F.Carry elem

  -- ===================================================================
  -- THE PACKING, AS LEMMAS WITH EXPLICIT CODOMAINS.  Probe700's,
  -- restated per the brief: the [LJ-1.700] review named the `_`
  -- codomain of hier-at-code as the likeliest wall
  -- (review-of-LJ-1-700-1.md:150-153).  same-at-codes is NOT restated:
  -- this route never reads SameAsGraph.
  -- ===================================================================

  toL : F.HS.ASt.SL → CS.S
  toL x = fst x , Lset→isL lam ordλ (fst x) (snd x)

  -- The table at the parameter code.  hierL lands in the class
  -- carrier (src/L/Hierarchy.lagda.md:621-622, under
  -- `open hPropStructure 𝒮ʟ` at :73), so fst of it is the stage the
  -- membership statements read.
  table : (cp : Code) (o : IsOrd (fst (val cp))) → S
  table cp o = fst (hierL (fst (toL (val cp))) (snd (toL (val cp))) o)

  -- EXPLICIT CODOMAIN.  HierInStage at the parameter code.
  hier-at-code : HierInStage
               → (cp : Code) (o : IsOrd (fst (val cp)))
               → ⟨ table cp o ∈ˢ Lset lam ⟩
  hier-at-code hier cp o =
    hier (toL (val cp)) o (snd (val cp))

  -- ===================================================================
  -- THE MISSING INPUT OF [LJ-1.700], NAMED.  The ambient 3-slot
  -- reading at (Lset δ, δ, the table): the brief's sketch
  -- ⟨ (Lset δ ∷ δ ∷ hierL δ ∷ []) ⊨ₚ matrix₃ ⟩ over the telescope
  -- HierInStage itself carries.
  -- ===================================================================

  AmbientAt-at-hier : Type (ℓ-suc ℓ)
  AmbientAt-at-hier =
    (δ : CS.S) (oδ : IsOrd (fst δ))
    → ⟨ fst δ ∈ˢ Lset lam ⟩
    → ⟨ ( Lset (fst δ) ∷ fst δ
          ∷ fst (hierL (fst δ) (δ .snd) oδ) ∷ [] )
        P652.⊨ₚ P667.matrix₃ ⟩

  -- mapFo val sends matrix₃-Code to embed matrix₃: its only constants
  -- come from slide, which is the absurd eliminator
  -- (Probe673.agda:68-69), and compose with val to the eliminator
  -- again.  atL is stated at embed φ, so this is the one path the
  -- matrix conjunct needs.
  val-matrix-embed :
    mapFo val A.A.matrix₃-Code ≡ embed P667.matrix₃
  val-matrix-embed =
      mapFo-comp A.A.slide val P667.matrix₃
    ∙ cong (λ h → mapFo h P667.matrix₃) (funExt (λ b → Empty.rec* b))

  -- The value slot's carrier witness.  eq transports the VALUE code's
  -- own membership onto the tower value, with the membership atom as
  -- the motive: the smallest family the transport can travel in.
  lset-val-mem :
    (ca cp : Code)
    (eq : fst (val ca) ≡ Lset (fst (val cp)))
    → ⟨ Lset (fst (val cp)) ∈ˢ Lset lam ⟩
  lset-val-mem ca cp eq =
    subst (λ z → ⟨ z ∈ˢ Lset lam ⟩) eq (snd (val ca))

  -- The three existential witnesses of BoundInStage, in the slot order
  -- value / parameter / witness.  The value slot is the tower value
  -- itself, NOT the code's image, so the ambient hypothesis applies
  -- with no satisfaction-level transport anywhere.
  wb wp wH : (ca cp : Code)
    (eq : fst (val ca) ≡ Lset (fst (val cp)))
    (hier : HierInStage) (ocp : IsOrd (fst (val cp)))
    → F.HS.ASt.SL
  wb ca cp eq hier ocp =
    ( Lset (fst (val cp))
    , lset-val-mem ca cp eq )
  wp ca cp eq hier ocp =
    (fst (val cp) , snd (val cp))
  wH ca cp eq hier ocp =
    (table cp ocp , hier-at-code hier cp ocp)

  -- The matrix conjunct, STEP 1: atL read backwards at the witness
  -- triple.  amb fires DIRECTLY: map fst of the triple reduces to amb's
  -- own reading vector, so no ambient satisfaction is restated anywhere.
  -- Ascribed at the embed side; step 2 moves the formula.
  matrix-embed-at :
    (amb : AmbientAt-at-hier) (hier : HierInStage)
    (ca cp : Code) (eq : fst (val ca) ≡ Lset (fst (val cp)))
    (ocp : IsOrd (fst (val cp)))
    → ⟨ ( wb ca cp eq hier ocp ∷ wp ca cp eq hier ocp
          ∷ wH ca cp eq hier ocp ∷ [] )
        F.HS.ASt.AbsL.⊨ᵐ embed P667.matrix₃ ⟩
  matrix-embed-at amb hier ca cp eq ocp =
    subst ⟨_⟩ (sym (Cy.atL P667.Δ₀-matrix₃ triple))
      (amb (toL (val cp)) ocp (snd (val cp)))
    where
    triple : F.HS.ASt.SL ^ 3
    triple =
      wb ca cp eq hier ocp ∷ wp ca cp eq hier ocp
        ∷ wH ca cp eq hier ocp ∷ []

  -- STEP 2: the formula transport, alone in its own definition so the
  -- two big satisfaction normal forms are never alive together.
  matrix-mapped-at :
    (amb : AmbientAt-at-hier) (hier : HierInStage)
    (ca cp : Code) (eq : fst (val ca) ≡ Lset (fst (val cp)))
    (ocp : IsOrd (fst (val cp)))
    → ⟨ ( wb ca cp eq hier ocp ∷ wp ca cp eq hier ocp
          ∷ wH ca cp eq hier ocp ∷ [] )
        F.HS.ASt.AbsL.⊨ᵐ mapFo val A.A.matrix₃-Code ⟩
  matrix-mapped-at amb hier ca cp eq ocp =
    subst (λ ψ → ⟨ ( wb ca cp eq hier ocp ∷ wp ca cp eq hier ocp
                     ∷ wH ca cp eq hier ocp ∷ [] )
                   F.HS.ASt.AbsL.⊨ᵐ ψ ⟩)
          (sym val-matrix-embed)
          (matrix-embed-at amb hier ca cp eq ocp)

  -- ===================================================================
  -- THE TERM.  BoundInStage ca cp = ⟨ [] AbsL.⊨ᵐ ∃̇ (mapFo val (inBound
  -- ca cp)) ⟩ has three existentials: the outer one (BoundInStage's
  -- own) takes the WITNESS, here the table; the inner two take the
  -- parameter and the value.  The value pin closes by sym eq, the
  -- parameter pin by refl.
  -- ===================================================================

  completeness-from-ambient :
    AmbientAt-at-hier → HierInStage → Completeness
  completeness-from-ambient amb hier ca cp ocp eq =
    ∣ wH ca cp eq hier ocp
    , ∣ wp ca cp eq hier ocp
      , ∣ wb ca cp eq hier ocp
        , ( matrix-mapped-at amb hier ca cp eq ocp
          , ( sym eq , refl ) )
        ∣₁ ∣₁ ∣₁
