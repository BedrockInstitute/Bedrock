  (X : S) (X⊆Lλ : (z : S) → ⟨ z ∈ˢ X ⟩ → ⟨ z ∈ˢ Lset lam ⟩)

  open A using ( Completeness; HierInStage; BoundInStage; inBound; Code; val )
  module F = P652.Frame652 lam ordλ succλ X X⊆Lλ ∅∈λ

  -- ===================================================================
  -- THE PACKING, AS LEMMAS WITH EXPLICIT CODOMAINS.  Probe700's,
  -- restated per the brief: the [LJ-1.700] review named the `_`
  -- codomain of hier-at-code as the likeliest wall

  toL : F.HS.ASt.SL → CS.S
  toL x = fst x , Lset→isL lam ordλ (fst x) (snd x)

  -- The table at the parameter code.  hierL lands in the class
  -- carrier (src/L/Hierarchy.lagda.md:621-622, under
  -- `open hPropStructure 𝒮ʟ` at :73), so fst of it is the stage the
  -- membership statements read.
  table : (cp : Code) (o : IsOrd (fst (val cp))) → S
  table cp o = fst (hierL (fst (toL (val cp))) (snd (toL (val cp))) o)

  -- EXPLICIT CODOMAIN.  HierInStage at the parameter code.
  hier-at-code : HierInStage
               → (cp : Code) (o : IsOrd (fst (val cp)))
               → ⟨ table cp o ∈ˢ Lset lam ⟩

    hier (toL (val cp)) o (snd (val cp))
