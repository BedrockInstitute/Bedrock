module Build (lam : S) (ordλ : IsOrd lam)
  (succλ : (d : S) → ⟨ d ∈ˢ lam ⟩ → ⟨ sucV d ∈ˢ lam ⟩)
  (X : S) (X⊆Lλ : (z : S) → ⟨ z ∈ˢ X ⟩ → ⟨ z ∈ˢ Lset lam ⟩)
  (∅∈λ : ⟨ ∅ ∈ˢ lam ⟩)
  (elem : P652.Frame652.A.Elementary lam ordλ succλ X X⊆Lλ ∅∈λ) where

  open Frame.Build lam ordλ succλ X X⊆Lλ ∅∈λ elem
    using ( Code; val; module F; module HS )

  -- The two code equations, as two single-coordinate substs.  The
  -- hypothesis, both families and the codomain are the literal
  -- ambient spelling, byte for byte the transcription's own body.
  amb : (δ : S) (ca cp : Code)
      → fst (val ca) ≡ Lset δ → fst (val cp) ≡ δ
      → (a : F.HS.ASt.SL)
      → ⟨ (fst (val ca) ∷ fst (val cp) ∷ fst a ∷ []) P652.⊨ₚ P667.matrix₃ ⟩
      → ⟨ (Lset δ ∷ δ ∷ fst a ∷ []) P652.⊨ₚ P667.matrix₃ ⟩
  amb δ ca cp ca≡Lδ cp≡δ a h =
    subst (λ u → ⟨ (u ∷ δ ∷ fst a ∷ []) P652.⊨ₚ P667.matrix₃ ⟩) ca≡Lδ
      (subst (λ u → ⟨ (fst (val ca) ∷ u ∷ fst a ∷ []) P652.⊨ₚ P667.matrix₃ ⟩)
             cp≡δ h)
