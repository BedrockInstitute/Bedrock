-- =====================================================================
-- THE DISCHARGE, AT A GENERIC `K` AND `γ'`.  ONE MEMBERSHIP IN.
--
-- `Ktr` IS NOT A SECOND ASK.  It is the record's own `transK`
-- (src/L/Condensation/TwelveAgree.lagda.md:262-264) one level up: the
-- record states transitivity over `S`, `EnvSupply.Fact` needs it over
-- `V ℓ` (src/L/Coding/EnvSupply.lagda.md:450, `isTransV` at
-- src/L/Constructible.lagda.md:83-84), and the `S` form cannot produce
-- the `V` form because `prK`'s intermediate `⁅ x , y ⁆` carries no `isL`
-- witness (src/L/Coding/EnvSupply.lagda.md:452-459).  So it is a
-- parameter HERE and it is PAID FROM THE TREE at the frame below, where
-- `layer-trans (Lset-layer lam)` discharges it and costs the caller
-- nothing.  `graphK` is the ONE hypothesis the frame does not pay.
--
-- NOT ONE FIELD IS WEAKENED and no `subst` is added at any field: every
-- one of the nine is a single application of an `EnvSupply.Fact` form
-- whose slot arguments make `graphK` fit DEFINITIONALLY, because
-- `lookup (suc i) (x ∷ xs)` reduces to `lookup i xs`.
-- =====================================================================

  (K : Fin (5 + n))
  (γ' : S ^ (11 + n))
  (Ktr : isTransV (fst (lookup (suc (suc (suc (suc (suc (suc K)))))) γ')))
  (graphK : ⟨ fst (lookup (suc zero) γ')
              ∈ fst (lookup (suc (suc (suc (suc (suc (suc K)))))) γ') ⟩)
  where

  module F = Fact (lookup (suc (suc (suc (suc (suc (suc K)))))) γ') Ktr

  -- The two slots the nine read, named once.
  Cslot Tslot : S
  Cslot = lookup (suc (suc zero)) γ'
  Tslot = lookup (suc zero) γ'

  nine : NineOfThirteen {n} K γ'
  nine = record
    { valK = λ k c ar a b yc c∈ shape hc →

          (suc (suc (suc (suc (suc (suc (suc (suc zero))))))))

    ; subK₁-imp = λ E yb ya yc b a ar c h →

          (suc (suc (suc (suc (suc (suc (suc (suc (suc zero)))))))))

-- =====================================================================
-- THE OBLIGATION, AT `KValue`'s FRAME.
--
-- THE TELESCOPE, ITEM BY ITEM.
--   `lam ordλ succλ ∅∈λ gam ordγ γ∈λ`  `KValue`'s seven
--     (src/L/Condensation.lagda.md:7380-7383).
--   `ω∈σ`                              `SupplyEnv`'s eighth parameter

--     the frame is `[LJ-1.551]`'s, whose cell 0 is `SE.B₀`; the nine
--     fields never consume it.

--   `graphK`                           THE ONE SUPPLIED MEMBERSHIP.

module Frame
  (lam : V ℓ) (ordλ : IsOrd lam)
  (succλ : (d : V ℓ) → ⟨ d ∈ lam ⟩ → ⟨ sucV d ∈ lam ⟩)
  (∅∈λ : ⟨ ∅ ∈ lam ⟩)
  (gam : V ℓ) (ordγ : IsOrd gam) (γ∈λ : ⟨ gam ∈ lam ⟩)
  (ω∈σ : ⟨ ω ∈ sucV gam ⟩) where

  module KV = KValue lam ordλ succλ ∅∈λ gam ordγ γ∈λ
  module SE = SupplyEnv lam ordλ succλ ∅∈λ gam ordγ γ∈λ ω∈σ

  γ‡ : (g1 g2 g3 g4 g5 : S) → Vec S 20
  γ‡ g1 g2 g3 g4 g5 = SE.B₀ ∷ g1 ∷ g2 ∷ g3 ∷ g4 ∷ g5 ∷ KV.Kenv

  -- The frame pays `Ktr`.  `lookup (suc^6 iK) (γ‡ ...)` reduces to
  -- `LsetS lam ordλ` (src/L/Condensation.lagda.md:7389-7390, :7397), so
  -- `fst` of it is `Lset lam` and the tree's own layer lemma closes it.
  Ktr‡ : (g1 g2 g3 g4 g5 : S)
    → isTransV (fst (lookup (suc (suc (suc (suc (suc (suc KV.iK))))))
                            (γ‡ g1 g2 g3 g4 g5)))
  Ktr‡ g1 g2 g3 g4 g5 = layer-trans (Lset-layer lam)

  graph-in-K-discharges : (g1 g2 g3 g4 g5 : S)
    → ⟨ fst (lookup (suc zero) (γ‡ g1 g2 g3 g4 g5))
        ∈ fst (lookup (suc (suc (suc (suc (suc (suc KV.iK))))))
                      (γ‡ g1 g2 g3 g4 g5)) ⟩
    → NineOfThirteen {n = 9} KV.iK (γ‡ g1 g2 g3 g4 g5)
  graph-in-K-discharges g1 g2 g3 g4 g5 graphK =
    Discharge.nine {n = 9} KV.iK (γ‡ g1 g2 g3 g4 g5)
      (Ktr‡ g1 g2 g3 g4 g5) graphK
