-- =====================================================================
-- 6.  WHY THE GAP FACT IS NOT MERELY UNBUILT.  IT IS FALSE AT THE
--     THEOREM'S OWN TELESCOPE.
--
-- `UnionKit` is `src/L/BoundedSubset.lagda.md:1145-1147`, and it is how
-- every one of the four consumer sites builds its `X`:
-- `X = Lset α ∪ ⁅ x ⁆s` (:1149-1150), where `x` is THE ARBITRARY
-- BOUNDED SUBSET the whole chapter is about.  `x` carries `x⊆Lα` and
-- nothing else: no `IsOrd x` anywhere in the telescope
-- (src/L/StageBound.lagda.md:68-69).
--
-- `X⊆M` (src/L/Hull.lagda.md:354-355) puts every member of `X` into the
-- hull.  So `AllHullOrd` says `x` is an ordinal, and it says every
-- member of `Lset α` is one too, which makes `Lset α` itself an
-- ordinal.  Neither is a hypothesis the chapter has, and the second is
-- refuted by any non-transitive member of the stage.
-- =====================================================================

module AtTheorem (α lam x : S) (ordα : IsOrd α) (ordλ : IsOrd lam)
  (α∈λ : ⟨ α ∈ˢ lam ⟩) (x⊆Lα : (z : S) → ⟨ z ∈ˢ x ⟩ → ⟨ z ∈ˢ Lset α ⟩)
  (x∈Lλ : ⟨ x ∈ˢ Lset lam ⟩) (α∉ω : ⟨ α ∈ˢ ω ⟩ → Empty.⊥)
  (succλ : (d : S) → ⟨ d ∈ˢ lam ⟩ → ⟨ sucV d ∈ˢ lam ⟩) where

  module UK = UnionKit α lam x ordα ordλ α∈λ x⊆Lα x∈Lλ α∉ω
  module Here = AtSite lam ordλ succλ UK.X UK.X⊆Lλ UK.∅∈λ

  -- The arbitrary bounded subset would have to be an ordinal.
  gap-fact-forces-x-ordinal : Here.AllHullOrd → IsOrd x
  gap-fact-forces-x-ordinal f = f x (Here.Site.H.X⊆M x UK.x∈X)

  -- And every stage below the hull's own stage would be an ordinal.
  gap-fact-makes-the-stage-an-ordinal : Here.AllHullOrd → IsOrd (Lset α)
  gap-fact-makes-the-stage-an-ordinal f =
    layer-trans (Lset-layer α)
    , λ z z∈Lα → f z (Here.Site.H.X⊆M z (UK.Lα∈X z z∈Lα)) .fst

  (succλ : (d : S) → ⟨ d ∈ˢ lam ⟩ → ⟨ sucV d ∈ˢ lam ⟩)
  (X : S) (X⊆L : (x : S) → ⟨ x ∈ˢ X ⟩ → ⟨ x ∈ˢ Lset lam ⟩)
  (∅∈λ : ⟨ ∅ ∈ˢ lam ⟩)

  → AtSite.A649.PiCommuteLset lam ordλ succλ X X⊆L ∅∈λ
  → (δ : S) → IsOrd δ

  AtSite.levelin-from-keystone lam ordλ succλ X X⊆L ∅∈λ

the-obligation-is-the-unordinal-keystone : (lam : S) (ordλ : IsOrd lam)
  (succλ : (d : S) → ⟨ d ∈ˢ lam ⟩ → ⟨ sucV d ∈ˢ lam ⟩)
