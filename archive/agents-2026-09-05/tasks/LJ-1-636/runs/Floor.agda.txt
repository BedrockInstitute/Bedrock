SqParam : S → Type (ℓ-suc ℓ)
SqParam α₀ =
  (δ : S) → ⟨ δ ∈ˢ sucV α₀ ⟩ → (⟨ δ ∈ˢ ω ⟩ → Empty.⊥)
  → Σ[ f ∈ (⟪ δ ⟫ × ⟪ δ ⟫ → ⟪ δ ⟫) ]
      ((u v : ⟪ δ ⟫ × ⟪ δ ⟫) → f u ≡ f v → u ≡ v)

module Frame (α₀ : S) (oα₀ : IsOrd α₀) (sq : SqParam α₀) where

  module SC = L.StageCardinal {ℓ} lem α₀ oα₀ sq

  floor : (α : S) → IsOrd α → ⟨ α ∈ˢ sucV α₀ ⟩
        → (⟨ α ∈ˢ ω ⟩ → Empty.⊥) → SC.Upper.P α
  floor α oα α∈suc infα _ _ _ = SC.Upper.stage-card-upper α oα α∈suc infα
