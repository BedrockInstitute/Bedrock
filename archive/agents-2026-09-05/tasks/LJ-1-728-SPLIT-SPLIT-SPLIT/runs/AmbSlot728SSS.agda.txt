module Build (lam : S) (ordλ : IsOrd lam)
  (succλ : (d : S) → ⟨ d ∈ˢ lam ⟩ → ⟨ sucV d ∈ˢ lam ⟩)
  (X : S) (X⊆Lλ : (z : S) → ⟨ z ∈ˢ X ⟩ → ⟨ z ∈ˢ Lset lam ⟩)
  (∅∈λ : ⟨ ∅ ∈ˢ lam ⟩)
  (elem : P652.Frame652.A.Elementary lam ordλ succλ X X⊆Lλ ∅∈λ) where

  open Frame.Build lam ordλ succλ X X⊆Lλ ∅∈λ elem
  open ConvM.Build lam ordλ succλ X X⊆Lλ ∅∈λ elem
    using ( conv )

  -- The two code equations, as two single-coordinate substs over
  -- AmbientAt.
  amb : (δ : S) (ca cp : Code)
      → fst (val ca) ≡ Lset δ → fst (val cp) ≡ δ
      → (a : F.HS.ASt.SL)
      → AmbientAt (fst (val ca)) (fst (val cp)) (fst a)
      → AmbientAt (Lset δ) δ (fst a)
  amb δ ca cp ca≡Lδ cp≡δ a h =
    subst (λ u → AmbientAt u δ (fst a)) ca≡Lδ
      (subst (λ u → AmbientAt (fst (val ca)) u (fst a)) cp≡δ h)

  -- The witness triple: the codes' values carried to (Lset δ, δ),
  -- hull membership read at HS.M, the parameter membership passed
  -- through.
  mkWit : (δ : S) → ⟨ Lset δ ∈ˢ HS.M ⟩
        → (ca cp : Code)
        → fst (val ca) ≡ Lset δ → fst (val cp) ≡ δ
        → (a : F.HS.ASt.SL)
        → HullM a → SatIn a ca cp
        → Σ[ z ∈ S ] (⟨ z ∈ˢ HS.M ⟩ × ⟨ Lset δ ∈ˢ HS.M ⟩
                     × AmbientAt (Lset δ) δ z)
  mkWit δ Lδ∈M ca cp ca≡Lδ cp≡δ a a∈H sat =
    fst a , a∈H , Lδ∈M , amb δ ca cp ca≡Lδ cp≡δ a (conv ca cp a sat)
