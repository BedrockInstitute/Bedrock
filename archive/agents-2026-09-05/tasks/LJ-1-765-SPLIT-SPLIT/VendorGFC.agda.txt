  GroundedFromComplete : Type (ℓ-suc ℓ)
  GroundedFromComplete =
    Completeness
    → ((a p z : S)
       → ⟨ (a ∷ p ∷ z ∷ []) P652.⊨ₚ P667.matrix₃ ⟩ → IsOrd p → a ≡ Lset p)
    → (δ : S) → IsOrd δ → ⟨ δ ∈ˢ HS.M ⟩ → ⟨ Lset δ ∈ˢ HS.M ⟩
    → ∥ Σ[ z ∈ S ] (⟨ z ∈ˢ HS.M ⟩ × ⟨ Lset δ ∈ˢ HS.M ⟩
                  × ⟨ (Lset δ ∷ δ ∷ z ∷ []) P652.⊨ₚ P667.matrix₃ ⟩) ∥₁

  -- THE CONSTRUCTED CONVERSION at B6's measured-green shape
  -- (SSS runs/bisect-6.out, 140.69 s): the domain the inline
  -- spelling, the codomain a solved meta, the definition unapplied.
  conv0 : (ca cp : Code) (a : F.HS.ASt.SL)
        → ⟨ (a ∷ []) F.HS.ASt.AbsL.⊨ᵐ (mapFo val (A.inBound ca cp)) ⟩
        → _
  conv0 = hull-convert-at-matrix

  -- The two code equations, as two single-coordinate substs.  The
  -- hypothesis, both families and the codomain are the literal
  -- Convert-codomain spelling, so h's argument check is
  -- string-identical and both subst results are computed, never
  -- compared.
  amb : (δ : S) (ca cp : Code)
      → fst (val ca) ≡ Lset δ → fst (val cp) ≡ δ
      → (a : F.HS.ASt.SL)
      → ⟨ (fst (val ca) ∷ fst (val cp) ∷ fst a ∷ []) P652.⊨ₚ P667.matrix₃ ⟩
      → ⟨ (Lset δ ∷ δ ∷ fst a ∷ []) P652.⊨ₚ P667.matrix₃ ⟩
  amb δ ca cp ca≡Lδ cp≡δ a h =
    subst (λ u → ⟨ (u ∷ δ ∷ fst a ∷ []) P652.⊨ₚ P667.matrix₃ ⟩) ca≡Lδ
      (subst (λ u → ⟨ (fst (val ca) ∷ u ∷ fst a ∷ []) P652.⊨ₚ P667.matrix₃ ⟩)
             cp≡δ h)

  -- The witness triple: the codes' values carried to (Lset δ, δ),
  -- hull membership read at HS.M, the parameter membership passed
  -- through.  The Sigma component is the obligation's own literal
  -- spelling.
  mkWit : (δ : S) → ⟨ Lset δ ∈ˢ HS.M ⟩
        → (ca cp : Code)
        → fst (val ca) ≡ Lset δ → fst (val cp) ≡ δ
        → (a : F.HS.ASt.SL)
        → HullM a → SatIn a ca cp
        → Σ[ z ∈ S ] (⟨ z ∈ˢ HS.M ⟩ × ⟨ Lset δ ∈ˢ HS.M ⟩
                     × ⟨ (Lset δ ∷ δ ∷ z ∷ []) P652.⊨ₚ P667.matrix₃ ⟩)
  mkWit δ Lδ∈M ca cp ca≡Lδ cp≡δ a a∈H sat =
    fst a , a∈H , Lδ∈M , amb δ ca cp ca≡Lδ cp≡δ a (conv0 ca cp a sat)

  -- THE TERM.  Hull membership gives both codes; Completeness lifts
  -- the pair to the stage existential; hull-closed (the loaded
  -- hull half) produces the witness; the constructed conversion and
  -- the two transports are mkWit's business.  The soundness argument
  -- is clause (iii)'s own; the composition does not need it and does

  grounded-from-complete : GroundedFromComplete

      PT.rec squash₁

                 mkWit δ Lδ∈M ca cp ca≡Lδ cp≡δ a a∈H sat })
               (hullClosed comp δ ordδ ca cp ca≡Lδ cp≡δ) })
        (codeOf (Lset δ) Lδ∈M) })
      (codeOf δ δ∈M)
