  -- The Bridge's vector decode, at the carrier A:  every environment
  -- set member is the graph of a carrier vector.
  module DBA = DefOf (fst A)

  envSet-graph : (n : ℕ) (z : S) → ⟨ z ∈ˢ envSet A n ⟩
    → ∥ Σ[ δ ∈ DBA.SM ^ n ] (fst z ≡ graph A δ) ∥₁
  envSet-graph n z hz = PT.map
    (λ { (g , hzg) → (λ i → (( ⟪ fst A ⟫↪ (g i) , wL (g i) ) , memS (g i))) , hzg })
    (envSet-out A n z hz)
    where
    memg : (m : ⟪ fst A ⟫) → ⟨ ⟪ fst A ⟫↪ m ∈ fst A ⟩
    memg m = ∈∈ₛ {a = ⟪ fst A ⟫↪ m} {b = fst A} .snd (∈ₛ⟪ fst A ⟫↪ m)
    wL : (m : ⟪ fst A ⟫) → ⟨ isL (⟪ fst A ⟫↪ m) ⟩
    wL m = isL-trans {x = fst A} {y = ⟪ fst A ⟫↪ m} (memg m) (snd A)
    memS : (m : ⟪ fst A ⟫) → (( ⟪ fst A ⟫↪ m , wL m ) ∈ᶜ DBA.M)
    memS m = memg m

  -- The cons-graph route:  a consAtL fact pins the extended
  -- environment's value to the graph of the consed vector, and the
  -- Bridge lands that graph in the environment set.
  consAt-route : ∀ {n} (δ : DBA.SM ^ n) (x : DBA.SM) (e′ z : S)
    → fst z ≡ graph δ
    → ⟨ (e′ ∷ x ∷ z ∷ []) ⊨ consAtL zero (suc zero) (suc (suc zero)) ⟩
    → ⟨ e′ ∈ˢ envSet A (suc n) ⟩
  consAt-route δ x e′ z hzg hc =
    graph-envSet A (x ∷ δ) e′ hfe
    where
    hfe : fst e′ ≡ graph (x ∷ δ)
    hfe = consAtL-out A δ x (e′ ∷ x ∷ z ∷ []) zero (suc zero) (suc (suc zero))
            hzg refl hc

  -- The bounded description.
  Bd : ∀ {n} → Formula ⟪ fst A ⟫ n → Formula S 1
  Bd (t ∈̇ u) =
    ∃̇∈ (con Om) (∃̇∈ (con Om)
    ( dIs (mapTm (asConst A) t) (suc zero) (suc (suc zero))
    ∧̇ ( dIs (mapTm (asConst A) u) zero (suc (suc zero))
         ∧̇ (var (suc zero) ∈̇ var zero) )))
  Bd (t ≐ u) =
    ∃̇∈ (con Om) (∃̇∈ (con Om)
    ( dIs (mapTm (asConst A) t) (suc zero) (suc (suc zero))
    ∧̇ ( dIs (mapTm (asConst A) u) zero (suc (suc zero))
         ∧̇ (var (suc zero) ≐ var zero) )))
  Bd (a ∧̇ b) = cond A (mapFo (asConst A) (a ∧̇ b))
  Bd (a ∨̇ b) = cond A (mapFo (asConst A) (a ∨̇ b))
  Bd (a ⇒̇ b) = cond A (mapFo (asConst A) (a ⇒̇ b))
  Bd (¬̇ a) = cond A (mapFo (asConst A) (¬̇ a))
  Bd ⊤̇ = cond A (mapFo (asConst A) ⊤̇)
  Bd ⊥̇ = cond A (mapFo (asConst A) ⊥̇)
  Bd (∃̇ a) =
    ∃̇∈ (con A) (∃̇∈ (con (envSet A (suc _)))
    ( consAtL zero (suc zero) (suc (suc zero))
    ∧̇ (var zero ∈̇ con (Sat A (mapFo (asConst A) a))) ))
  Bd (∀̇ a) =
    ∀̇∈ (con A) (∀̇∈ (con (envSet A (suc _)))
    ( consAtL zero (suc zero) (suc (suc zero))
    ⇒̇ (var zero ∈̇ con (Sat A (mapFo (asConst A) a))) ))
  Bd (∀̇∈ t a) =
    ∀̇∈ (con Om) ( dIs (mapTm (asConst A) t) zero (suc zero)
    ⇒̇ ∀̇∈ (con A) ( (var zero ∈̇ var (suc zero))
    ⇒̇ ∀̇∈ (con (envSet A (suc _)))
      ( consAtL zero (suc zero) (suc (suc (suc zero)))
      ⇒̇ (var zero ∈̇ con (Sat A (mapFo (asConst A) a))) ) ))
  Bd (∃̇∈ t a) =
    ∃̇∈ (con Om) ( dIs (mapTm (asConst A) t) zero (suc zero)
    ∧̇ ∃̇∈ (con A) ( (var zero ∈̇ var (suc zero))
    ∧̇ ∃̇∈ (con (envSet A (suc _)))
      ( consAtL zero (suc zero) (suc (suc (suc zero)))
      ∧̇ (var zero ∈̇ con (Sat A (mapFo (asConst A) a))) ) ))

  bdΔ₀ : ∀ {n} (ψ : Formula ⟪ fst A ⟫ n) → Δ₀ (Bd ψ)
  bdΔ₀ (t ∈̇ u) =
    δ-∃∈ (δ-∃∈ (δ-∧ (dIsΔ₀ (mapTm (asConst A) t) (suc zero) (suc (suc zero)))
    (δ-∧ (dIsΔ₀ (mapTm (asConst A) u) zero (suc (suc zero))) δ-∈)))
  bdΔ₀ (t ≐ u) =
    δ-∃∈ (δ-∃∈ (δ-∧ (dIsΔ₀ (mapTm (asConst A) t) (suc zero) (suc (suc zero)))
    (δ-∧ (dIsΔ₀ (mapTm (asConst A) u) zero (suc (suc zero))) δ-≐)))
  bdΔ₀ (a ∧̇ b) = δ-∧ δ-∈ δ-∈
  bdΔ₀ (a ∨̇ b) = δ-∨ δ-∈ δ-∈
  bdΔ₀ (a ⇒̇ b) = δ-⇒ δ-∈ δ-∈
  bdΔ₀ (¬̇ a) = δ-¬ δ-∈
  bdΔ₀ ⊤̇ = δ-⊤
  bdΔ₀ ⊥̇ = δ-⊥
  bdΔ₀ (∃̇ a) =
    δ-∃∈ (δ-∃∈ (δ-∧ (Δ₀-consAtL zero (suc zero) (suc (suc zero))) δ-∈))
  bdΔ₀ (∀̇ a) =
    δ-∀∈ (δ-∀∈ (δ-⇒ (Δ₀-consAtL zero (suc zero) (suc (suc zero))) δ-∈))
  bdΔ₀ (∀̇∈ t a) =
    δ-∀∈ (δ-⇒ (dIsΔ₀ (mapTm (asConst A) t) zero (suc zero))
    (δ-∀∈ (δ-⇒ δ-∈
    (δ-∀∈ (δ-⇒ (Δ₀-consAtL zero (suc zero) (suc (suc (suc zero)))) δ-∈)))))
  bdΔ₀ (∃̇∈ t a) =
    δ-∃∈ (δ-∧ (dIsΔ₀ (mapTm (asConst A) t) zero (suc zero))
    (δ-∃∈ (δ-∧ δ-∈
    (δ-∃∈ (δ-∧ (Δ₀-consAtL zero (suc zero) (suc (suc (suc zero)))) δ-∈)))))


  -- The per-formula placement needs:  the immediate subformulas'
  -- placements at the carve stage.
  NeedsLeafAt : (β : V ℓ) (oβ : IsOrd β) → ∀ {k} → Formula S k → Type (ℓ-suc ℓ)
  NeedsLeafAt β oβ a = ⟨ Sat A a ∈ˢ LsetS β oβ ⟩
  NeedsAt : (β : V ℓ) (oβ : IsOrd β) → ∀ {n} (ψ : Formula ⟪ fst A ⟫ n) → Type (ℓ-suc ℓ)
  NeedsAt β oβ (t ∈̇ u) = Lift Unit
  NeedsAt β oβ (t ≐ u) = Lift Unit
  NeedsAt β oβ ⊤̇ = Lift Unit
  NeedsAt β oβ ⊥̇ = Lift Unit
  NeedsAt β oβ (a ∧̇ b) = NeedsLeafAt β oβ (mapFo (asConst A) a) × NeedsLeafAt β oβ (mapFo (asConst A) b)
  NeedsAt β oβ (a ∨̇ b) = NeedsLeafAt β oβ (mapFo (asConst A) a) × NeedsLeafAt β oβ (mapFo (asConst A) b)
  NeedsAt β oβ (a ⇒̇ b) = NeedsLeafAt β oβ (mapFo (asConst A) a) × NeedsLeafAt β oβ (mapFo (asConst A) b)
  NeedsAt β oβ (¬̇ a) = NeedsLeafAt β oβ (mapFo (asConst A) a)
  NeedsAt β oβ (∃̇ a) = NeedsLeafAt β oβ (mapFo (asConst A) a)
  NeedsAt β oβ (∀̇ a) = NeedsLeafAt β oβ (mapFo (asConst A) a)
  NeedsAt β oβ (∀̇∈ t a) = NeedsLeafAt β oβ (mapFo (asConst A) a)
  NeedsAt β oβ (∃̇∈ t a) = NeedsLeafAt β oβ (mapFo (asConst A) a)

  -- The bounded-fo certificate, by recursion on the description.
  bddTm : (β : V ℓ) → ((k : ℕ) → BelowAt β (numS k))
        → ((m′ : ⟪ fst A ⟫) → BelowAt β (asConst A m′))
        → ∀ {n} (τ : Term ⟪ fst A ⟫ n)
        → BoundedTm (BelowAt β) (mapTm (asConst A) τ)
  bddTm β rNβ rCβ (var i) = _
  bddTm β rNβ rCβ (con m) = rCβ m

  bddDIs : (β : V ℓ) → (bddOm : BelowAt β Om)
         → ((k : ℕ) → BelowAt β (numS k))
         → ((m′ : ⟪ fst A ⟫) → BelowAt β (asConst A m′))
         → ∀ {n m} (τ : Term ⟪ fst A ⟫ n) (v e : Fin m)
         → BoundedFo (BelowAt β) (dIs (mapTm (asConst A) τ) v e)
  bddDIs β bddOm rNβ rCβ (var i) v e = (bddOm , (rNβ (toℕ i) , _))
  bddDIs β bddOm rNβ rCβ (con m) v e = rCβ m

  bddBd : (β : V ℓ) (oβ : IsOrd β) → ∀ {n} (ψ : Formula ⟪ fst A ⟫ n)
    (bddOm : BelowAt β Om)
    (rAβ : BelowAt β A)
    (rEβ : (k : ℕ) → BelowAt β (envSet A k))
    (rNβ : (k : ℕ) → BelowAt β (numS k))
    (rCβ : (m′ : ⟪ fst A ⟫) → BelowAt β (asConst A m′))
    → NeedsAt β oβ ψ
    → BoundedFo (BelowAt β) (Bd ψ)
  bddBd β oβ (t ∈̇ u) bddOm rAβ rEβ rNβ rCβ _ =
    ( bddOm , (bddOm , (bddDIs β bddOm rNβ rCβ t , (bddDIs β bddOm rNβ rCβ u , _))) )
  bddBd β oβ (t ≐ u) bddOm rAβ rEβ rNβ rCβ _ =
    ( bddOm , (bddOm , (bddDIs β bddOm rNβ rCβ t , (bddDIs β bddOm rNβ rCβ u , _))) )
  bddBd β oβ (a ∧̇ b) bddOm rAβ rEβ rNβ rCβ (ha , hb) = ((_ , ha) , (_ , hb))
  bddBd β oβ (a ∨̇ b) bddOm rAβ rEβ rNβ rCβ (ha , hb) = ((_ , ha) , (_ , hb))
  bddBd β oβ (a ⇒̇ b) bddOm rAβ rEβ rNβ rCβ (ha , hb) = ((_ , ha) , (_ , hb))
  bddBd β oβ (¬̇ a) bddOm rAβ rEβ rNβ rCβ h = (_ , h)
  bddBd β oβ ⊤̇ bddOm rAβ rEβ rNβ rCβ _ = _
  bddBd β oβ ⊥̇ bddOm rAβ rEβ rNβ rCβ _ = _
  bddBd β oβ (∃̇ a) bddOm rAβ rEβ rNβ rCβ h = (rAβ , (rEβ _ , (_ , ((_ , h)))))
  bddBd β oβ (∀̇ a) bddOm rAβ rEβ rNβ rCβ h = (rAβ , (rEβ _ , (_ , ((_ , h)))))
  bddBd β oβ (∀̇∈ t a) bddOm rAβ rEβ rNβ rCβ h =
    ( bddOm , (bddDIs β bddOm rNβ rCβ t
    , (rAβ , ((_ , _) , (rEβ _ , (_ , ((_ , h))))))) )
  bddBd β oβ (∃̇∈ t a) bddOm rAβ rEβ rNβ rCβ h =
    ( bddOm , (bddDIs β bddOm rNβ rCβ t
    , (rAβ , ((_ , _) , (rEβ _ , (_ , ((_ , h))))))) )

  -- tmIs and dIs agree at every environment;  the con-case is literal.
  tmIs→dIs : ∀ {n m} (t : Term S n) (v e : Fin m) (γ : S ^ m)
           → ⟨ γ ⊨ tmIs t v e ⟩ → ⟨ γ ⊨ dIs t v e ⟩
  tmIs→dIs (var i) v e γ h = dIs-var-in i v e γ (tmIs-var-out i γ v e h)
  tmIs→dIs (con c) v e γ h = h

  dIs→tmIs : ∀ {n m} (t : Term S n) (v e : Fin m) (γ : S ^ m)
           → ⟨ γ ⊨ dIs t v e ⟩ → ⟨ γ ⊨ tmIs t v e ⟩
  dIs→tmIs (var i) v e γ h = tmIs-var-in i γ v e (dIs-var-out i v e γ h)
  dIs→tmIs (con c) v e γ h = h

  -- ===================================================================
  -- SECTION 4c.  THE ADEQUACY, PER CONSTRUCTOR.
  --
  -- Forward (cond to Bd) discharges the bounded witnesses:  the atom
  -- escorts through the stage set, the quantifier extenders through
  -- envSet A (suc n) by the cons-graph route.  Backward is extraction,
  -- and the propositional clauses are the identity, since there Bd is
  -- cond itself.
  -- ===================================================================

  -- The bounded-existential introduction, once and for all:  a
  -- witness, its bound-membership, and the body.
  bnd∈ : ∀ {m} (Ωc : S) (γ : S ^ m) {φ : Formula S (suc m)}
       → (x : S) → ⟨ x ∈ˢ Ωc ⟩ → ⟨ (x ∷ γ) ⊨ φ ⟩
       → ⟨ γ ⊨ ∃̇∈ (con Ωc) φ ⟩
  bnd∈ Ωc γ x hx hbody = ∣ x , (hx , hbody) ∣₁

  -- A tmIs fact bounds the v-slot's member:  the var case through the
  -- tagged pair inside the e-slot's value, the con case through the
  -- constant.
  slot∈Om : ∀ {n m} (τ : Term ⟪ fst A ⟫ n) (v e : Fin m) (γ : S ^ m)
    (hzn : ⟨ lookup e γ ∈ˢ envSet A n ⟩) (znL : ⟨ isL (fst (lookup e γ)) ⟩)
    → ⟨ γ ⊨ tmIs (mapTm (asConst A) τ) v e ⟩
    → ⟨ lookup v γ ∈ˢ Om ⟩
  slot∈Om (var i) v e γ hzn znL h =
    slot-bounded _ (lookup e γ) znL hzn (toℕ i) (fst (lookup v γ))
    (tmIs-var-out i γ v e h)
  slot∈Om (con m) v e γ hzn znL h =
    asConst-bounded m (fst (lookup v γ)) h

  atom-adeq : ∀ {n} (τ₁ τ₂ : Term ⟪ fst A ⟫ n)
    (z : S) (zL : ⟨ isL (fst z) ⟩) (hz : ⟨ z ∈ˢ envSet A n ⟩)
    → ⟨ (z ∷ []) ⊨ cond A (mapFo (asConst A) (τ₁ ∈̇ τ₂)) ⟩
    → ⟨ (z ∷ []) ⊨ Bd (τ₁ ∈̇ τ₂) ⟩
  atom-adeq τ₁ τ₂ z zL hz sat =
    PT.map (λ { (v , (w , (h1 , (h2 , hvw)))) →
    v , ( slot∈Om τ₁ (suc zero) (suc (suc zero)) (w ∷ v ∷ z ∷ []) hz zL h1
    , ∣ w , ( slot∈Om τ₂ zero (suc (suc zero)) (w ∷ v ∷ z ∷ []) hz zL h2
    , ( tmIs→dIs (mapTm (asConst A) τ₁) (suc zero) (suc (suc zero))
                   (w ∷ v ∷ z ∷ []) h1
      , ( tmIs→dIs (mapTm (asConst A) τ₂) zero (suc (suc zero))
                     (w ∷ v ∷ z ∷ []) h2
      , hvw ) ) ) ∣₁ ) })
    (cond∈-out A (mapTm (asConst A) τ₁) (mapTm (asConst A) τ₂) z sat)

  atom-adeq-bwd : ∀ {n} (τ₁ τ₂ : Term ⟪ fst A ⟫ n)
    (z : S) (zL : ⟨ isL (fst z) ⟩) (hz : ⟨ z ∈ˢ envSet A n ⟩)
    → ⟨ (z ∷ []) ⊨ Bd (τ₁ ∈̇ τ₂) ⟩
    → ⟨ (z ∷ []) ⊨ cond A (mapFo (asConst A) (τ₁ ∈̇ τ₂)) ⟩
  atom-adeq-bwd τ₁ τ₂ z zL hz sat =
    cond∈-in A (mapTm (asConst A) τ₁) (mapTm (asConst A) τ₂) z
    (PT.map (λ { (v , hv) →
      PT.map (λ { (w , hwp) →
      v , (w , ( dIs→tmIs (mapTm (asConst A) τ₁) (suc zero) (suc (suc zero))
                               (w ∷ v ∷ z ∷ []) (hwp .snd .fst)
                   , ( dIs→tmIs (mapTm (asConst A) τ₂) zero (suc (suc zero))
                                (w ∷ v ∷ z ∷ []) (hwp .snd .snd .fst)
                   , hwp .snd .snd .snd ) ) ) })
      (hv .snd) })
    sat)

  atom-adeq≐ : ∀ {n} (τ₁ τ₂ : Term ⟪ fst A ⟫ n)
    (z : S) (zL : ⟨ isL (fst z) ⟩) (hz : ⟨ z ∈ˢ envSet A n ⟩)
    → ⟨ (z ∷ []) ⊨ cond A (mapFo (asConst A) (τ₁ ≐ τ₂)) ⟩
    → ⟨ (z ∷ []) ⊨ Bd (τ₁ ≐ τ₂) ⟩
  atom-adeq≐ τ₁ τ₂ z zL hz sat =
    PT.map (λ { (v , (w , (h1 , (h2 , hvw)))) →
    v , ( slot∈Om τ₁ (suc zero) (suc (suc zero)) (w ∷ v ∷ z ∷ []) hz zL h1
    , ∣ w , ( slot∈Om τ₂ zero (suc (suc zero)) (w ∷ v ∷ z ∷ []) hz zL h2
    , ( tmIs→dIs (mapTm (asConst A) τ₁) (suc zero) (suc (suc zero))
                   (w ∷ v ∷ z ∷ []) h1
      , ( tmIs→dIs (mapTm (asConst A) τ₂) zero (suc (suc zero))
                     (w ∷ v ∷ z ∷ []) h2
      , hvw ) ) ) ∣₁ ) })
    (cond≐-out A (mapTm (asConst A) τ₁) (mapTm (asConst A) τ₂) z sat)

  atom-adeq≐-bwd : ∀ {n} (τ₁ τ₂ : Term ⟪ fst A ⟫ n)
    (z : S) (zL : ⟨ isL (fst z) ⟩) (hz : ⟨ z ∈ˢ envSet A n ⟩)
    → ⟨ (z ∷ []) ⊨ Bd (τ₁ ≐ τ₂) ⟩
    → ⟨ (z ∷ []) ⊨ cond A (mapFo (asConst A) (τ₁ ≐ τ₂)) ⟩
  atom-adeq≐-bwd τ₁ τ₂ z zL hz sat =
    cond≐-in A (mapTm (asConst A) τ₁) (mapTm (asConst A) τ₂) z
    (PT.map (λ { (v , hv) →
      PT.map (λ { (w , hwp) →
      v , (w , ( dIs→tmIs (mapTm (asConst A) τ₁) (suc zero) (suc (suc zero))
                               (w ∷ v ∷ z ∷ []) (hwp .snd .fst)
                   , ( dIs→tmIs (mapTm (asConst A) τ₂) zero (suc (suc zero))
                                (w ∷ v ∷ z ∷ []) (hwp .snd .snd .fst)
                   , hwp .snd .snd .snd ) ) ) })
      (hv .snd) })
    sat)


  -- ∃̇:  forward bounds the extender by envSet A (suc n) through the
  -- cons-graph route;  backward drops the bound.
  q-adeq∃ : ∀ {n} (a : Formula ⟪ fst A ⟫ (suc n))
    (z : S) (zL : ⟨ isL (fst z) ⟩) (hz : ⟨ z ∈ˢ envSet A n ⟩)
    → ⟨ (z ∷ []) ⊨ cond A (mapFo (asConst A) (∃̇ a)) ⟩
    → ⟨ (z ∷ []) ⊨ Bd (∃̇ a) ⟩
  q-adeq∃ a z zL hz sat = PT.rec (snd GOALΩ) outer sat
    where
    GOALΩ : Ω
    GOALΩ = ((z ∷ []) ⊨ Bd (∃̇ a))
    outer : Σ[ x ∈ S ] (⟨ x ∈ˢ A ⟩
            × ∥ Σ[ e′ ∈ S ] (⟨ (e′ ∷ x ∷ z ∷ [])
                ⊨ consAtL zero (suc zero) (suc (suc zero)) ⟩
              × ⟨ fst e′ ∈ fst (Sat A (mapFo (asConst A) a)) ⟩) ∥₁)
      → ⟨ GOALΩ ⟩
    outer (x , (hxB , hinner)) =
      PT.rec (snd GOALΩ)
        (λ { (e′ , (hc , hr)) →
          PT.rec (snd GOALΩ)
            (λ { (δ , hzg) →
              ( ∣ x , ( hxB
              , (∣ e′ , ( (consAt-route δ (x , hxB) e′ z hzg hc)
                        , (hc , hr) ) ∣₁) ) ∣₁ ) })
            (envSet-graph _ z hz) })
        hinner

  q-adeq∃-bwd : ∀ {n} (a : Formula ⟪ fst A ⟫ (suc n))
    (z : S) (zL : ⟨ isL (fst z) ⟩) (hz : ⟨ z ∈ˢ envSet A n ⟩)
    → ⟨ (z ∷ []) ⊨ Bd (∃̇ a) ⟩
    → ⟨ (z ∷ []) ⊨ cond A (mapFo (asConst A) (∃̇ a)) ⟩
  q-adeq∃-bwd a z zL hz sat =
    cond∃-in A (mapFo (asConst A) a) z
    (PT.map (λ { (x , (hxB , hinner)) →
      PT.map (λ { (e′ , (hE′ , hc , hr)) →
      x , (hxB , (∣ e′ , (hc , hr) ∣₁)) })
      hinner })
    sat)

  -- ∀̇:  forward is a direct application (cond's extender is already
  -- universal);  backward bounds the extender through the route.
  q-adeq∀ : ∀ {n} (a : Formula ⟪ fst A ⟫ (suc n))
    (z : S) (zL : ⟨ isL (fst z) ⟩) (hz : ⟨ z ∈ˢ envSet A n ⟩)
    → ⟨ (z ∷ []) ⊨ cond A (mapFo (asConst A) (∀̇ a)) ⟩
    → ⟨ (z ∷ []) ⊨ Bd (∀̇ a) ⟩
  q-adeq∀ a z zL hz sat = λ x hxB e′ hE′ hc →
    cond∀-out A (mapFo (asConst A) a) z sat x e′ hxB hc
  q-adeq∀-bwd : ∀ {n} (a : Formula ⟪ fst A ⟫ (suc n))
    (z : S) (zL : ⟨ isL (fst z) ⟩) (hz : ⟨ z ∈ˢ envSet A n ⟩)
    → ⟨ (z ∷ []) ⊨ Bd (∀̇ a) ⟩
    → ⟨ (z ∷ []) ⊨ cond A (mapFo (asConst A) (∀̇ a)) ⟩
  q-adeq∀-bwd a z zL hz sat = λ x hxB e′ hc →
    PT.rec (snd ((e′ ∷ x ∷ z ∷ [])
              ⊨ (var zero ∈̇ con (Sat A (mapFo (asConst A) a)))))
    (λ { (δ , hzg) →
      sat x hxB e′ (consAt-route δ (x , hxB) e′ z hzg hc) hc })
    (envSet-graph _ z hz)


  -- The two bounded-quantifier clauses, ∀̇∈ and ∃̇∈:  the tmIs guard's
  -- witness is bounded by the stage set, the environment extender by
  -- envSet A (suc n) through the cons-graph route.
  -- The ∀̇∈ body of the bounded description, named for the backward
  -- adequacy's truncation target.
  BdBody∀∈ : ∀ {n} (τ : Term ⟪ fst A ⟫ n) (a : Formula ⟪ fst A ⟫ (suc n))
           → Formula S 2
  BdBody∀∈ τ a =
    dIs (mapTm (asConst A) τ) zero (suc zero)
    ⇒̇ ∀̇∈ (con A) ( (var zero ∈̇ var (suc zero))
    ⇒̇ ∀̇∈ (con (envSet A _))
    ( consAtL zero (suc zero) (suc (suc (suc zero)))
    ⇒̇ (var zero ∈̇ con (Sat A (mapFo (asConst A) a))) ) )

  q-adeq∀∈ : ∀ {n} (τ : Term ⟪ fst A ⟫ n) (a : Formula ⟪ fst A ⟫ (suc n))
    (z : S) (zL : ⟨ isL (fst z) ⟩) (hz : ⟨ z ∈ˢ envSet A n ⟩)
    → ⟨ (z ∷ []) ⊨ cond A (mapFo (asConst A) (∀̇∈ τ a)) ⟩
    → ⟨ (z ∷ []) ⊨ Bd (∀̇∈ τ a) ⟩
  q-adeq∀∈ τ a z zL hz sat =
    λ w hw∈Om hdis x hxA hxw e′ hE′ hcons →
    cond∀∈-out A (mapTm (asConst A) τ) (mapFo (asConst A) a) z sat w
      (dIs→tmIs (mapTm (asConst A) τ) zero (suc zero) (w ∷ z ∷ []) hdis)
      x e′ hxA hxw hcons

  q-adeq∀∈-bwd : ∀ {n} (τ : Term ⟪ fst A ⟫ n) (a : Formula ⟪ fst A ⟫ (suc n))
    (z : S) (zL : ⟨ isL (fst z) ⟩) (hz : ⟨ z ∈ˢ envSet A n ⟩)
    → ⟨ (z ∷ []) ⊨ Bd (∀̇∈ τ a) ⟩
    → ⟨ (z ∷ []) ⊨ cond A (mapFo (asConst A) (∀̇∈ τ a)) ⟩
  q-adeq∀∈-bwd τ a z zL hz sat =
    λ w hwft x hxA hxw e′ hcons →
    PT.rec (snd ((w ∷ z ∷ []) ⊨ BdBody∀∈ τ a))
      (λ { (δ , hzg) →
      sat w (slot∈Om τ zero (suc zero) (w ∷ z ∷ []) hz zL hwft)
            (tmIs→dIs (mapTm (asConst A) τ) zero (suc zero) (w ∷ z ∷ []) hwft)
            x hxA hxw e′
            (consAt-route δ (x , hxA) e′ z hzg hcons)
            hcons })
      (envSet-graph _ z hz)

  q-adeq∃∈ : ∀ {n} (τ : Term ⟪ fst A ⟫ n) (a : Formula ⟪ fst A ⟫ (suc n))
    (z : S) (zL : ⟨ isL (fst z) ⟩) (hz : ⟨ z ∈ˢ envSet A n ⟩)
    → ⟨ (z ∷ []) ⊨ cond A (mapFo (asConst A) (∃̇∈ τ a)) ⟩
    → ⟨ (z ∷ []) ⊨ Bd (∃̇∈ τ a) ⟩
  q-adeq∃∈ τ a z zL hz sat = PT.rec (snd GOALΩ) outerW sat
    where
    GOALΩ : Ω
    GOALΩ = ((z ∷ []) ⊨ Bd (∃̇∈ τ a))
    stepE : (w : S)
           → ⟨ (w ∷ z ∷ []) ⊨ tmIs (mapTm (asConst A) τ) zero (suc zero) ⟩
           → (δ : DBA.SM ^ _) → fst z ≡ graph A δ
           → (x : S) → ⟨ x ∈ˢ A ⟩ → ⟨ x ∈ˢ w ⟩
           → (e′ : S)
           → ⟨ (e′ ∷ x ∷ w ∷ z ∷ [])
             ⊨ consAtL zero (suc zero) (suc (suc (suc zero))) ⟩
           → ⟨ fst e′ ∈ fst (Sat A (mapFo (asConst A) a)) ⟩
           → ⟨ GOALΩ ⟩
    stepE w hwft δ hzg x hxA hxw e′ hcons hr =
      bnd∈ Om (z ∷ []) w
        (slot∈Om τ zero (suc zero) (w ∷ z ∷ []) hz zL hwft)
        ( tmIs→dIs (mapTm (asConst A) τ) zero (suc zero) (w ∷ z ∷ []) hwft
        , bnd∈ A (w ∷ z ∷ []) x hxA
            ( hxw
            , bnd∈ (envSet A _) (x ∷ w ∷ z ∷ []) e′
                (consAt-route δ (x , hxA) e′ z hzg hcons)
                (hcons , hr) ) )
    stepX : (w : S)
           → ⟨ (w ∷ z ∷ []) ⊨ tmIs (mapTm (asConst A) τ) zero (suc zero) ⟩
           → (δ : DBA.SM ^ _) → fst z ≡ graph A δ
           → (x : S) → ⟨ x ∈ˢ A ⟩ → ⟨ x ∈ˢ w ⟩
           → ∥ Σ[ e′ ∈ S ]
               (⟨ (e′ ∷ x ∷ w ∷ z ∷ [])
               ⊨ consAtL zero (suc zero) (suc (suc (suc zero))) ⟩
               × ⟨ fst e′ ∈ fst (Sat A (mapFo (asConst A) a)) ⟩) ∥₁
           → ⟨ GOALΩ ⟩
    stepX w hwft δ hzg x hxA hxw he∥ =
      PT.rec (snd GOALΩ)
        (λ { (e′ , (hcons , hr)) → stepE w hwft δ hzg x hxA hxw e′ hcons hr })
        he∥
    stepD : (w : S)
           → ⟨ (w ∷ z ∷ []) ⊨ tmIs (mapTm (asConst A) τ) zero (suc zero) ⟩
           → ∥ Σ[ x ∈ S ] ((⟨ x ∈ˢ A ⟩ × ⟨ x ∈ˢ w ⟩)
                × ∥ Σ[ e′ ∈ S ]
                    (⟨ (e′ ∷ x ∷ w ∷ z ∷ [])
                    ⊨ consAtL zero (suc zero) (suc (suc (suc zero))) ⟩
                    × ⟨ fst e′ ∈ fst (Sat A (mapFo (asConst A) a)) ⟩) ∥₁) ∥₁
           → (δ : DBA.SM ^ _) → fst z ≡ graph A δ
           → ⟨ GOALΩ ⟩
    stepD w hwft hx∥ δ hzg =
      PT.rec (snd GOALΩ) (λ { (x , ((hxA , hxw) , he∥)) →
        stepX w hwft δ hzg x hxA hxw he∥ }) hx∥
    outerW : (Σ[ w ∈ S ]
              (⟨ (w ∷ z ∷ []) ⊨ tmIs (mapTm (asConst A) τ) zero (suc zero) ⟩
              × ∥ Σ[ x ∈ S ] ((⟨ x ∈ˢ A ⟩ × ⟨ x ∈ˢ w ⟩)
                × ∥ Σ[ e′ ∈ S ]
                    (⟨ (e′ ∷ x ∷ w ∷ z ∷ [])
                    ⊨ consAtL zero (suc zero) (suc (suc (suc zero))) ⟩
                    × ⟨ fst e′ ∈ fst (Sat A (mapFo (asConst A) a)) ⟩) ∥₁) ∥₁))
            → ⟨ GOALΩ ⟩
    outerW (w , (hwft , hx∥)) =
      PT.rec (snd GOALΩ) (λ { (δ , hzg) → stepD w hwft hx∥ δ hzg })
        (envSet-graph _ z hz)

  q-adeq∃∈-bwd : ∀ {n} (τ : Term ⟪ fst A ⟫ n) (a : Formula ⟪ fst A ⟫ (suc n))
    (z : S) (zL : ⟨ isL (fst z) ⟩) (hz : ⟨ z ∈ˢ envSet A n ⟩)
    → ⟨ (z ∷ []) ⊨ Bd (∃̇∈ τ a) ⟩
    → ⟨ (z ∷ []) ⊨ cond A (mapFo (asConst A) (∃̇∈ τ a)) ⟩
  q-adeq∃∈-bwd τ a z zL hz sat =
    cond∃∈-in A (mapTm (asConst A) τ) (mapFo (asConst A) a) z
    (PT.map (λ { (w , (hw∈Om , (hdis , hx∥))) →
      PT.map (λ { (x , ((hxA , hxw) , he∥)) →
      PT.map (λ { (e′ , (hE′ , hcons , hr)) →
            w , ( dIs→tmIs (mapTm (asConst A) τ) zero (suc zero)
                          (w ∷ z ∷ []) hdis
                , bnd∈ A (w ∷ z ∷ []) x hxA
                    (hxw , bnd∈ (envSet A _) (x ∷ w ∷ z ∷ []) e′ hE′
                      (hcons , hr)) ) })
      he∥ })
      hx∥ })
    sat)

  -- ===================================================================
  -- SECTION 5.  THE CLIMB.
  --
  -- One number per formula, `size psi`, carries the successor-count
  -- above sigma0 the placement costs.  The recursion is stated at an
  -- arbitrary room depth d;  the top call is d = zero, and
  -- closedomega absorbs the whole iterate into gamma.
  -- ===================================================================

  sizeψ : ∀ {n} → Formula ⟪ fst A ⟫ n → ℕ
  sizeψ (t ∈̇ u) = nsuc nzero
  sizeψ (t ≐ u) = nsuc nzero
  sizeψ (a ∧̇ b) = nsuc (sizeψ a + sizeψ b)
  sizeψ (a ∨̇ b) = nsuc (sizeψ a + sizeψ b)
  sizeψ (a ⇒̇ b) = nsuc (sizeψ a + sizeψ b)
  sizeψ (¬̇ a) = nsuc (sizeψ a)
  sizeψ ⊤̇ = nsuc nzero
  sizeψ ⊥̇ = nsuc nzero
  sizeψ (∃̇ a) = nsuc (sizeψ a)
  sizeψ (∀̇ a) = nsuc (sizeψ a)
  sizeψ (∀̇∈ t a) = nsuc (sizeψ a)
  sizeψ (∃̇∈ t a) = nsuc (sizeψ a)

  size≥1 : ∀ {n} (ψ : Formula ⟪ fst A ⟫ n) → nsuc nzero ≤ⁿ sizeψ ψ
  size≥1 (t ∈̇ u) = ≤r
  size≥1 (t ≐ u) = ≤r
  size≥1 (a ∧̇ b) = ≤s (m≤ⁿ+ (sizeψ a) (sizeψ b))
  size≥1 (a ∨̇ b) = ≤s (m≤ⁿ+ (sizeψ a) (sizeψ b))
  size≥1 (a ⇒̇ b) = ≤s (m≤ⁿ+ (sizeψ a) (sizeψ b))
  size≥1 (¬̇ a) = ≤s ≤r
  size≥1 ⊤̇ = ≤r
  size≥1 ⊥̇ = ≤r
  size≥1 (∃̇ a) = ≤s ≤r
  size≥1 (∀̇ a) = ≤s ≤r
  size≥1 (∀̇∈ t a) = ≤s ≤r
  size≥1 (∃̇∈ t a) = ≤s ≤r

  -- The room and the stage-bound certificate, lifted to any stage at
  -- or above the room depth:  the same four facts the clause needs.
  roomAt : (d s : ℕ) → Room (sucIter d σ₀) (sucIter-ord d oσ₀)
         → Room (sucIter (d + s) σ₀) (sucIter-ord (d + s) oσ₀)
  roomAt d s = Room-lift d s

  omBelow : (d s : ℕ) → ⟨ Lset σ₀ ∈ˢᵥ Lset (sucIter (d + nsuc s) σ₀) ⟩
  omBelow d s =
    LsetV-lift≤ (Lset σ₀) (nsuc nzero) (d + nsuc s)
    (≤ⁿ-trans (≤s ≤r) (≤ⁿ-1ge-alternate d s))
    Om∈sucσ₀
    where
    ≤ⁿ-1ge-alternate : (d s : ℕ) → nsuc nzero ≤ⁿ d + nsuc s
    ≤ⁿ-1ge-alternate d s =
      subst (λ w → nsuc nzero ≤ⁿ w) (sym (plus-suc d s)) (≤s (m≤ⁿ+ d s))


  un≤' : (d s : ℕ) → nsuc (d + s) ≤ⁿ d + nsuc s
  un≤' d s =
    subst (λ w → nsuc (d + s) ≤ⁿ w) (sym (plus-suc d s)) ≤r

  -- The climb.  One clause per constructor;  the shape is uniform:
  -- lift the room to the carve stage, take the certificate and the
  -- adequacy from Section 4, and place.
  R : (d : ℕ) {n : ℕ} (ψ : Formula ⟪ fst A ⟫ n)
    → Room (sucIter d σ₀) (sucIter-ord d oσ₀)

      ∈ˢ LsetS (sucIter (nsuc (d + sizeψ ψ)) σ₀)

    where

    χ = mapFo (asConst A) (t ∈̇ u)
