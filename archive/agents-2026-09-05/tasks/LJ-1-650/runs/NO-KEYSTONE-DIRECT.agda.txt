           (succλ : (d : SV.S) → ⟨ d ∈ˢ lam ⟩ → ⟨ sucV d ∈ˢ lam ⟩)

           (X⊆Lλ : (z : SV.S) → ⟨ z ∈ˢ X ⟩ → ⟨ z ∈ˢ Lset lam ⟩)
           (∅∈λ : ⟨ ∅ ∈ˢ lam ⟩) where

  module Q = P.Coded lam ordλ succλ X X⊆Lλ ∅∈λ

  coded-cover-from-keystone : Q.LsetCodeOrd → Q.CodedCover
