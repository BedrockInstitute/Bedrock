-- Telescope copied from src/L/BoundedSubset.lagda.md:903-905, the same
-- six parameters the predecessor carries (Probe474.agda:88-91).

module HullStage (lam : S) (ordλ : IsOrd lam)
  (succλ : (d : S) → ⟨ d ∈ˢ lam ⟩ → ⟨ sucV d ∈ˢ lam ⟩)
  (X : S) (X⊆L : (x : S) → ⟨ x ∈ˢ X ⟩ → ⟨ x ∈ˢ Lset lam ⟩)
  (∅∈λ : ⟨ ∅ ∈ˢ lam ⟩) where

  module ASt = AtStage lam ordλ

  module H = ASt.Hull X X⊆L ∅∈λ

  -- The predecessor's own instance.  feed and lset-codes are taken, not
  -- rebuilt: Probe474.agda:124-131.
  module P = Q.HullStage lam ordλ succλ X X⊆L ∅∈λ

  open H.T using ( Code; val; vals; Sat; val-wit; _⊨₀_ )

  -- =====================================================================
  -- THE OBLIGATION, stated.
  -- =====================================================================

  LsetCodeOrd : Type (ℓ-suc ℓ)
  LsetCodeOrd =
    (c : Code) → IsOrd (fst (val c))
    → Σ[ d ∈ Code ] (fst (val d) ≡ Lset (fst (val c)))

  -- =====================================================================
  -- STEP ONE.  The ordinality join [LJ-1.479] could not find.
  -- It asked for IsOrd from hull membership and stopped
  -- (lj-1.479-report.md:84-88).  With IsOrd GIVEN, the converse is one
  -- line: ord∈Lset→∈ at src/L/Ordinal/Stages.lagda.md:265-268.
  -- =====================================================================

  code-ord-mem : (c : Code) → IsOrd (fst (val c)) → ⟨ fst (val c) ∈ˢ lam ⟩
  code-ord-mem c oc = ord∈Lset→∈ lam ordλ (fst (val c)) oc (snd (val c))

  -- =====================================================================
  -- STEP TWO.  The wit read-off.  What val (feed c) is, once the stage
  -- is known to satisfy the packaged graph there.
  -- =====================================================================

  env : Code → Vec ASt.SL (suc N)
  env c = vals (c ∷ P.lset-codes)

  -- RESIDUE ONE.  The stage believes the graph of the tower at the code.
  StageSat : Type (ℓ-suc ℓ)
  StageSat = (c : Code) → IsOrd (fst (val c)) → Sat (suc N) Q.packaged (env c)

  -- RESIDUE TWO.  A stage witness of the packaged graph IS the tower.
  ReadOff : Type (ℓ-suc ℓ)
  ReadOff = (c : Code) (a : ASt.SL) → IsOrd (fst (val c))
          → ⟨ (a ∷ env c) ⊨₀ Q.packaged ⟩
          → fst a ≡ Lset (fst (val c))

  -- The reduction, typechecked: the obligation is exactly those two.
  join : StageSat → ReadOff → LsetCodeOrd
  join sat read c oc = P.feed c , eq
    where
    w : Sat (suc N) Q.packaged (env c)
    w = sat c oc
    least : Σ[ a ∈ ASt.SL ] _
    least = leastOf ASt.wL {ℓ'' = ℓ-suc ℓ} lem
              (λ a → (a ∷ env c) ⊨₀ Q.packaged) w
    eq : fst (val (P.feed c)) ≡ Lset (fst (val c))
    eq = cong fst (val-wit (suc N) Q.packaged (c ∷ P.lset-codes) w)
       ∙ read c (least .fst) oc (least .snd .fst)
