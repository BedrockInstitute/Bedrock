module Build (lam : S) (ordλ : IsOrd lam)
  (succλ : (d : S) → ⟨ d ∈ˢ lam ⟩ → ⟨ sucV d ∈ˢ lam ⟩)
  (X : S) (X⊆Lλ : (z : S) → ⟨ z ∈ˢ X ⟩ → ⟨ z ∈ˢ Lset lam ⟩)
  (∅∈λ : ⟨ ∅ ∈ˢ lam ⟩)
  (elem : P652.Frame652.A.Elementary lam ordλ succλ X X⊆Lλ ∅∈λ) where

  open Frame.Build lam ordλ succλ X X⊆Lλ ∅∈λ elem
    using ( module F; module HS )

  -- THE GENERIC CARRY, opened once at this frame (W2), as in the
  -- obligation file.

  -- STEP 2, THE CONVERSION ALONE.  h is a variable, so the check of
  -- conv h = h is exactly one defeq run between the raw codomain and
  -- the goal: the π-compute reduction on the three entries, no carry
  -- body anywhere.  The membership hypotheses are NAMED binders (the
  -- spelled domain mentions the pair entries, and an unnamed arrow
  -- binds nothing).
  conv :
    (x₁ x₂ x₃ : S)
    (m₁ : ⟨ x₁ ∈ˢ HS.M ⟩) (m₂ : ⟨ x₂ ∈ˢ HS.M ⟩) (m₃ : ⟨ x₃ ∈ˢ HS.M ⟩)
    → ⟨ map fst (map Cy.CIso.I.g
                  ((x₁ , m₁) ∷ (x₂ , m₂) ∷ (x₃ , m₃) ∷ []))
         P652.⊨ₚ P667.matrix₃ ⟩
    → ⟨ (HS.C.π x₁ ∷ HS.C.π x₂ ∷ HS.C.π x₃ ∷ [])
         P652.⊨ₚ P667.matrix₃ ⟩
  conv x₁ x₂ x₃ m₁ m₂ m₃ h = h
