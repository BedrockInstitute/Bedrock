-- =====================================================================
-- 1.  THE CARVE.  [LJ-1.490]'s (Probe490.agda:105-116), at [LJ-1.521]'s
--     `rankFo`.  The two formulas are the same formula: [LJ-1.521]'s
--     change to [LJ-1.497]'s spelling is naming only, its own file head
--     says so (Probe521.agda:28-31), and `assignAt`/`supAt` unfold to
--     [LJ-1.490]'s bodies.  `bnd` stays a parameter.
-- =====================================================================

rank-graph :
    (Q a bnd : S)
  → Σ[ G ∈ S ] ((z : S) → (z ∈ˢ G)
      ≡ ((z ∈ˢ bnd) ⊓ ((z ∷ []) ⊨ P521.rankFo Q a)))
rank-graph Q a bnd = hasSeparationL bnd (P521.rankFo Q a) .fst

rank-graph-out :
    (Q a bnd z : S)
  → ⟨ z ∈ˢ fst (rank-graph Q a bnd) ⟩
  → ⟨ z ∈ˢ bnd ⟩ × ⟨ (z ∷ []) ⊨ P521.rankFo Q a ⟩
rank-graph-out Q a bnd z h =
  subst ⟨_⟩ (snd (rank-graph Q a bnd) z) h

-- =====================================================================
-- 2.  W3, THE INSTANTIATION.  `rankFo-adequate′` at
--     `Q := ord-set-witness a oa .fst`, with the hypothesis discharged by
--     the witness.  This is runs/W3.agda, repeated here because a file
--     cannot import the slice it was cut from.
-- =====================================================================

ordQ : (a : S) → IsOrd (fst a) → S
ordQ a oa = P521.ord-set-witness a oa .fst

adequate-at-witness :
    (a : S) (oa : IsOrd (fst a)) (z : S)
  → ⟨ (z ∷ []) ⊨ P521.rankFo (ordQ a oa) a ⟩
  → Σ[ m ∈ S ] Σ[ mx ∈ ⟨ fst m ∈ fst a ⟩ ]
      (fst z ≡ pr (fst m) (fst (P521.rank-at′ a oa m mx)))
adequate-at-witness a oa z =
  P521.rankFo-adequate′ (ordQ a oa) a oa z (P521.ord-set-witness a oa .snd)

-- =====================================================================
-- 3.  THE RANK IS INDIFFERENT TO BOTH ARGUMENTS THAT CAN DIFFER.  The
--     member is determined only up to `↾-reflects`, and the membership
--     proof is a proposition.  This is [LJ-1.521]'s `isPropTgt` design
--     (Probe521.agda:964-1000) at a weaker conclusion: THE MOTIVE DOES
--     NOT NAME THE PATH.  Writing it as `subst`/`ΣPathP` over the path
--     puts the path inside a type family, the elaborator normalizes it,
--     and [LJ-1.521] measured that shape at more than ten minutes
--     (agents/tasks/LJ-1-521/runs/full-try4.out).
-- =====================================================================

rank-at′-irr :
    (a : S) (oa : IsOrd (fst a)) (m m' : S) → m ≡ m'
  → (mx : ⟨ fst m ∈ fst a ⟩) (mx' : ⟨ fst m' ∈ fst a ⟩)
  → fst (P521.rank-at′ a oa m mx) ≡ fst (P521.rank-at′ a oa m' mx')
rank-at′-irr a oa m m' q = atSame q
  where
  Mot : (m'' : S) → m ≡ m'' → Type (ℓ-suc ℓ)
  Mot m'' _ =
      (u : ⟨ fst m ∈ fst a ⟩) (u' : ⟨ fst m'' ∈ fst a ⟩)
    → fst (P521.rank-at′ a oa m u) ≡ fst (P521.rank-at′ a oa m'' u')

  base : Mot m refl
  base u u' = cong (λ v → fst (P521.rank-at′ a oa m v))
                   (snd (fst m ∈ fst a) u u')

  atSame : {m'' : S} (p : m ≡ m'') → Mot m'' p
  atSame = J Mot base

-- =====================================================================
-- 3.1  THE READING.  A pair in the carve is a member-and-rank pair, with
--      `Q` determined.  Three delivered steps and no adapter:
--      `prʟ-fst` (src/L/Coding/Model.lagda.md:329-330) to enter the
--      carve, `rank-graph-out` to leave it with satisfaction,
--      `adequate-at-witness` to read the pair, `pr-inj`
--      (src/V/Coding.lagda.md:178-179) to split it.
-- =====================================================================

module Carve (a : S) (oa : IsOrd (fst a)) (bnd : S) where

  Q : S
  Q = ordQ a oa

  G : S
  G = fst (rank-graph Q a bnd)

  sat : (x y : S) → ⟨ pr (fst x) (fst y) ∈ fst G ⟩
      → ⟨ (prʟ x y ∷ []) ⊨ P521.rankFo Q a ⟩
  sat x y h = rank-graph-out Q a bnd (prʟ x y) inG .snd
    where
    inG : ⟨ prʟ x y ∈ˢ G ⟩
    inG = subst (λ w → ⟨ w ∈ fst G ⟩) (sym (prʟ-fst x y)) h

  read : (x y : S) → ⟨ pr (fst x) (fst y) ∈ fst G ⟩
       → Σ[ m ∈ S ] Σ[ mx ∈ ⟨ fst m ∈ fst a ⟩ ]
           (pr (fst x) (fst y) ≡ pr (fst m) (fst (P521.rank-at′ a oa m mx)))
  read x y h =
    t .fst , (t .snd .fst , sym (prʟ-fst x y) ∙ t .snd .snd)
    where
    t = adequate-at-witness a oa (prʟ x y) (sat x y h)

  -- the argument is the member, up to the carrier
  arg : (x y : S) (h : ⟨ pr (fst x) (fst y) ∈ fst G ⟩)
      → fst x ≡ fst (read x y h .fst)
  arg x y h = pr-inj (read x y h .snd .snd) .fst

  -- the value is the rank at that member
  val : (x y : S) (h : ⟨ pr (fst x) (fst y) ∈ fst G ⟩)
      → fst y ≡ fst (P521.rank-at′ a oa (read x y h .fst)
                                        (read x y h .snd .fst))
  val x y h = pr-inj (read x y h .snd .snd) .snd

-- =====================================================================
-- 4.  THE OBLIGATION.  `InjCode`'s FIRST conjunct
--     (src/L/Cardinal.lagda.md:225), over the carve, at the determined
--     `Q` and at a FREE `bnd`.  `svAt-in`
--     (src/L/Coding/Model.lagda.md:238-240) reduces it to: two values
--     recorded at the same argument agree.
-- =====================================================================

-- the conjunct's type, read out of `InjCode` so that nothing can drift
SvAtOf : S → S → Type (ℓ-suc ℓ)
SvAtOf F a = ⟨ (F ∷ a ∷ []) ⊨ svAt zero ⟩

svAt-is-first-of-InjCode : (F a b : S) → InjCode F a b → SvAtOf F a
svAt-is-first-of-InjCode F a b = fst

module Sv (a : S) (oa : IsOrd (fst a)) (bnd : S) where

  module C = Carve a oa bnd

  G : S
  G = C.G

  Hold : S → S → Type (ℓ-suc ℓ)
  Hold x y = ⟨ pr (fst x) (fst y) ∈ fst G ⟩

  -- the two members carry the same carrier, so they are the same element
  -- of the model: `↾-reflects` (src/FOL/ZFStructure.lagda.md:164-167)
  same : (x y y' : S) (h : Hold x y) (h' : Hold x y')
       → C.read x y h .fst ≡ C.read x y' h' .fst
  same x y y' h h' =
    ↾-reflects {𝒮 = 𝒮ᵥ} {M = isL} (sym (C.arg x y h) ∙ C.arg x y' h')

  agree : (x y y' : S) (h : Hold x y) (h' : Hold x y')
        → fst (P521.rank-at′ a oa (C.read x y h .fst) (C.read x y h .snd .fst))
        ≡ fst (P521.rank-at′ a oa (C.read x y' h' .fst)
                                  (C.read x y' h' .snd .fst))
  agree x y y' h h' =
    rank-at′-irr a oa (C.read x y h .fst) (C.read x y' h' .fst)
      (same x y y' h h') (C.read x y h .snd .fst) (C.read x y' h' .snd .fst)
