       → ⟨ (a ∷ p ∷ z ∷ []) P652.⊨ₚ P667.matrix₃ ⟩ → IsOrd p → a ≡ Lset p)

  grounded-from-complete : GroundedFromComplete
  grounded-from-complete comp sound δ ordδ δ∈M Lδ∈M =
    PT.rec squash₁
      (λ { (cp , cp≡δ) →
      PT.rec squash₁
        (λ { (ca , ca≡Lδ) → tail comp ca ca≡Lδ cp cp≡δ })
        (hull-member (Lset δ) Lδ∈M) })
      (hull-member δ δ∈M)

          (comp ca cp ordcp codeq))

      codeq : fst (val ca) ≡ Lset (fst (val cp))
